# County support for pretenders. Support lives on the county title as
# var:pretender_support (1-3) and var:pretender_backer (the character); the
# backer keeps a variable list pretender_supported_counties of those counties.
#
# Verified against the installed game (1.19):
# - geographical_region = custom_roman_full_borders:
#   map_data/geographical_regions/geographical_region.txt:1544
# - add/has/remove_county_modifier, title_province: casus_belli_types/07_ep3_wars.txt
#   (ep3_laamp_peasant_war)
# - add_to_variable_list / remove_list_variable / is_target_in_variable_list:
#   dozens of uses under common/ and events/

pretender_raise_support_effect = {
	$COUNTY$ = {
		save_temporary_scope_as = support_county
		if = {
			limit = {
				title_province = { geographical_region = custom_roman_full_borders }
			}
			if = {
				# Another living pretender holds this county: wear them down first.
				limit = {
					has_variable = pretender_backer
					NOT = { var:pretender_backer = $PRETENDER$ }
					var:pretender_backer = { is_alive = yes }
				}
				pretender_lower_support_effect = { COUNTY = scope:support_county }
			}
			else = {
				if = {
					limit = {
						has_variable = pretender_backer
						NOT = { var:pretender_backer = $PRETENDER$ }
					}
					# The old backer is dead: the county is up for grabs.
					remove_variable = pretender_support
				}
				set_variable = { name = pretender_backer value = $PRETENDER$ }
				if = {
					limit = { NOT = { has_variable = pretender_support } }
					set_variable = { name = pretender_support value = 0 }
				}
				change_variable = { name = pretender_support add = 1 }
				if = {
					limit = { var:pretender_support > 3 }
					set_variable = { name = pretender_support value = 3 }
				}
				$PRETENDER$ = {
					if = {
						limit = {
							NOT = {
								is_target_in_variable_list = {
									name = pretender_supported_counties
									target = scope:support_county
								}
							}
						}
						add_to_variable_list = {
							name = pretender_supported_counties
							target = scope:support_county
						}
					}
				}
				pretender_sync_county_modifier_effect = yes
			}
		}
	}
}

pretender_lower_support_effect = {
	$COUNTY$ = {
		save_temporary_scope_as = fading_county
		if = {
			limit = { has_variable = pretender_support }
			change_variable = { name = pretender_support add = -1 }
			if = {
				limit = { var:pretender_support <= 0 }
				var:pretender_backer ?= {
					remove_list_variable = {
						name = pretender_supported_counties
						target = scope:fading_county
					}
				}
				remove_variable = pretender_backer
				remove_variable = pretender_support
			}
			pretender_sync_county_modifier_effect = yes
		}
	}
}

pretender_clear_support_effect = {
	$COUNTY$ = {
		if = {
			limit = { has_variable = pretender_support }
			set_variable = { name = pretender_support value = 1 }
			pretender_lower_support_effect = { COUNTY = this }
		}
	}
}

# County scope. Show exactly the tier modifier matching var:pretender_support.
pretender_sync_county_modifier_effect = {
	if = {
		limit = { has_county_modifier = pretender_support_sympathetic_modifier }
		remove_county_modifier = pretender_support_sympathetic_modifier
	}
	if = {
		limit = { has_county_modifier = pretender_support_supportive_modifier }
		remove_county_modifier = pretender_support_supportive_modifier
	}
	if = {
		limit = { has_county_modifier = pretender_support_fervent_modifier }
		remove_county_modifier = pretender_support_fervent_modifier
	}
	if = {
		limit = { has_variable = pretender_support }
		if = {
			limit = { var:pretender_support = 1 }
			add_county_modifier = { modifier = pretender_support_sympathetic_modifier }
		}
		else_if = {
			limit = { var:pretender_support = 2 }
			add_county_modifier = { modifier = pretender_support_supportive_modifier }
		}
		else = {
			add_county_modifier = { modifier = pretender_support_fervent_modifier }
		}
	}
}
