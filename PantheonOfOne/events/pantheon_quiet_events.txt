namespace = pantheon

# 2.9: the quiet-prayer timer, every pantheon_quiet_interval_days (26 days; Rare 52, Frequent 13; Off keeps the Normal interval).
# Started by pantheon_take_throne_effect and, for a 2.8 save, by the quarterly pantheon.0010, like the popup timer pantheon.0050.
# Hidden, and silent: it never shows anything. It does three things, in this order:
#   1. prune (ALWAYS, also under the Off rule and before the cap check, so a faded or invalid record closes with the window shut);
#   2. unless prayers are Off, open one quiet prayer if the god's list holds fewer than 18;
#   3. reschedule itself (under Off too, so a later rule change takes effect without restarting).
pantheon.0051 = {
	type = character_event
	hidden = yes
	trigger = { has_trait = pantheon_divine }
	immediate = {
		# the heartbeat: while this chain runs, the quarterly pantheon.0010 never starts a second one
		set_variable = { name = pantheon_quiet_alive value = yes days = 120 }
		pantheon_quiet_prune_effect = yes
		if = {
			limit = { NOT = { has_game_rule = pantheon_prayers_off } }
			if = {
				limit = {
					OR = {
						NOT = { has_variable_list = pantheon_quiet_prayers }
						AND = { has_variable_list = pantheon_quiet_prayers NOT = { variable_list_size = { name = pantheon_quiet_prayers value >= 18 } } }
					}
				}
				pantheon_quiet_pick_effect = yes
			}
		}
		# a saved scope is inherited by this timer chain; never let the next pass see a stale candidate
		pantheon_prayer_clear_delivery_effect = yes
		trigger_event = { id = pantheon.0051 days = pantheon_quiet_interval_days }
	}
}
