# Pantheon of One: the Divine Will menu. Entered from pantheon_divine_will_interaction.
# scope:pantheon_target is the right-clicked follower. Root is the god.
# No trait names appear here (the 2.0 icon-flood rule): boons are tracked with timed variables.
namespace = pantheon_menu

pantheon_menu.0001 = {
	type = character_event
	title = pantheon_menu.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:pantheon_target = root }
				desc = pantheon_menu.0001.desc_self
			}
			desc = pantheon_menu.0001.desc
		}
		desc = pantheon_menu.0001.worship
		triggered_desc = {
			trigger = {
				NOT = { scope:pantheon_target = root }
				scope:pantheon_target = { pantheon_wavering_trigger = yes }
			}
			desc = pantheon_menu.0001.devotion_wavering
		}
		triggered_desc = {
			trigger = {
				NOT = { scope:pantheon_target = root }
				scope:pantheon_target = {
					has_variable = pantheon_devotion
					NOT = { pantheon_wavering_trigger = yes }
					NOT = { pantheon_devout_trigger = yes }
				}
			}
			desc = pantheon_menu.0001.devotion_faithful
		}
		triggered_desc = {
			trigger = {
				NOT = { scope:pantheon_target = root }
				scope:pantheon_target = { pantheon_devout_trigger = yes }
			}
			desc = pantheon_menu.0001.devotion_devout
		}
	}
	theme = faith
	left_portrait = {
		character = scope:pantheon_target
		animation = personality_zealous
	}

	# ---- The god's own options (on yourself only)

	option = {
		name = pantheon_menu.0001.throne
		trigger = {
			scope:pantheon_target = root
			NOT = { has_variable = pantheon_enthroned }
			is_landed = yes
		}
		custom_tooltip = pantheon_menu.0001.throne.tt
		hidden_effect = { pantheon_take_throne_effect = yes }
		trigger_event = pantheon_menu.0001
	}

	option = {
		name = pantheon_menu.0001.proclaim
		trigger = {
			scope:pantheon_target = root
			NOT = { rite = faith.main_rite }
		}
		custom_tooltip = pantheon_menu.0001.proclaim.tt
		hidden_effect = { pantheon_proclaim_effect = yes }
		trigger_event = pantheon_menu.0001
	}

	option = {
		name = pantheon_menu.0001.forswear
		trigger = {
			scope:pantheon_target = root
			NOT = { has_character_modifier = pantheon_forsworn }
		}
		custom_tooltip = pantheon_menu.0001.forswear.tt
		hidden_effect = { add_character_modifier = pantheon_forsworn }
		trigger_event = pantheon_menu.0001
	}

	option = {
		name = pantheon_menu.0001.embrace
		trigger = {
			scope:pantheon_target = root
			has_character_modifier = pantheon_forsworn
		}
		custom_tooltip = pantheon_menu.0001.embrace.tt
		hidden_effect = { remove_character_modifier = pantheon_forsworn }
		trigger_event = pantheon_menu.0001
	}

	option = {
		name = pantheon_menu.0001.blessed
		trigger = {
			scope:pantheon_target = root
			has_variable_list = pantheon_blessed
		}
		trigger_event = pantheon_menu.0400
	}

	# ---- Boons (Worship flows only once you are enthroned)

	option = {
		name = pantheon_menu.0001.traits
		trigger = {
			root = { has_variable = pantheon_enthroned }
			scope:pantheon_target = { NOT = { has_variable = pantheon_boon_trait } }
			scope:pantheon_target = { custom_tooltip = { text = pantheon_not_a_demigod_tt NOT = { has_trait = pantheon_demigod } } }
		}
		trigger_event = pantheon_menu.0100
	}

	option = {
		name = pantheon_menu.0001.realm
		trigger = {
			root = { has_variable = pantheon_enthroned }
			scope:pantheon_target = { is_landed = yes }
			scope:pantheon_target = { custom_tooltip = { text = pantheon_not_a_demigod_tt NOT = { has_trait = pantheon_demigod } } }
		}
		trigger_event = pantheon_menu.0200
	}

	option = {
		name = pantheon_menu.0001.life
		trigger = {
			root = { has_variable = pantheon_enthroned }
		}
		trigger_event = pantheon_menu.0300
	}

	option = {
		name = pantheon_menu.0001.host
		trigger = {
			custom_tooltip = { text = pantheon_needs_worship_host_tt var:pantheon_worship >= pantheon_cost_host }
			root = { has_variable = pantheon_enthroned }
			scope:pantheon_target = {
				is_landed = yes
				is_at_war = yes
				any_character_war = { NOT = { has_variable = pantheon_host_called } }
			}
			scope:pantheon_target = { custom_tooltip = { text = pantheon_not_a_demigod_tt NOT = { has_trait = pantheon_demigod } } }
		}
		show_as_unavailable = {
			root = { has_variable = pantheon_enthroned }
			scope:pantheon_target = {
				is_landed = yes
				is_at_war = yes
				any_character_war = { NOT = { has_variable = pantheon_host_called } }
			}
			scope:pantheon_target = { custom_tooltip = { text = pantheon_not_a_demigod_tt NOT = { has_trait = pantheon_demigod } } }
		}
		custom_tooltip = pantheon_menu.0001.host.tt
		hidden_effect = {
			pantheon_call_host_effect = yes
			pantheon_change_devotion_effect = { AMOUNT = 10 }
		}
		trigger_event = pantheon_menu.0001
	}

	option = {
		name = pantheon_menu.0001.close
	}
}
