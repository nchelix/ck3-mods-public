# Pantheon of One 1.4: Wrath. Root is the god; scope:pantheon_victim the one judged (saved by the route, else taken from
# var:pantheon_wrath_target). At most three options per page may grey out (show_as_unavailable); a fourth would vanish
# (memory ck3-event-option-visibility).
# 2.3 (R18, R19): every page carries its route's context as the persistent saved value scope:pantheon_wrath_ctx (1 sworn
# oath, 2 Retribution call, 3 Judgement interaction, 4 Lands interaction -- 3 and 4 run no pages). Context 2 also carries
# the call's identity (scope:pantheon_retri_defender / _attacker / _token): the page trigger and every wrath effect
# re-validate it (pantheon_wrath_page_ok_trigger, pantheon_wrath_mutation_ok_trigger), so a replaced or ended call closes
# the page harmlessly. Back leaves the call open; only a wrath that lands (or an answer in 0080) closes it.

# Pass Judgement: navigation only.
pantheon.0070 = {
	type = character_event
	title = pantheon.0070.t
	desc = pantheon.0070.desc
	theme = faith
	left_portrait = {
		character = scope:pantheon_victim
		animation = fear
	}
	trigger = { pantheon_wrath_page_ok_trigger = yes }
	immediate = { pantheon_wrath_open_page_effect = yes }

	option = {
		name = pantheon.0070.person
		trigger_event = pantheon.0071
	}

	# Hidden (not greyed) for a sin-only sinner
	option = {
		name = pantheon.0070.lands
		trigger = {
			NOT = { pantheon_wrath_page_light_trigger = yes }
			NOT = { pantheon_wrath_page_harsh_trigger = yes }
			scope:pantheon_victim = { any_held_county = { always = yes } }
		}
		trigger_event = pantheon.0073
	}

	# Fix round 1 (F6): a sworn oath can be left pending. Shown only on the oath's own page; it changes nothing (the oath
	# stays sworn and the Prayers tab's "Open the Judgement" answers it later). Never greyed, so 0070 has at most 4 options.
	option = {
		name = pantheon.0070.later
		trigger = {
			pantheon_wrath_oath_page_trigger = yes
			NOT = { has_variable = pantheon_great_prayer_unasked }
		}
		custom_tooltip = pantheon.0070.later.tt
	}

	# 2.5 (R7): the same page for an oath sworn UNASKED (the Followers row's answer). Leaving it withdraws the oath, which
	# costs nothing; only on the unasked oath's own page (both conditions), so 0071-0073's Back stays plain navigation.
	option = {
		name = pantheon.0070.later_unasked
		trigger = {
			pantheon_wrath_oath_page_trigger = yes
			has_variable = pantheon_great_prayer_unasked
		}
		custom_tooltip = pantheon.0070.later_unasked.tt
		hidden_effect = { pantheon_unasked_withdraw_effect = yes }
	}

	# Stay: a broken oath (-60) only on the SWORN OATH's own page, about the sworn offender (context 1); on any other route
	# (a Retribution call's Back -> 0070, ...) Stay just closes the page and the oath stays open.
	option = {
		name = pantheon.0070.stay
		# 2.5 (R7): an UNASKED oath, on its own page, is withdrawn for free (no broken-oath cost)
		if = {
			limit = {
				pantheon_wrath_oath_page_trigger = yes
				has_variable = pantheon_great_prayer_unasked
			}
			custom_tooltip = pantheon.0070.stay_unasked.tt
			hidden_effect = { pantheon_unasked_withdraw_effect = yes }
		}
		else_if = {
			limit = { pantheon_wrath_oath_page_trigger = yes }
			custom_tooltip = pantheon.0070.stay_oath.tt
			hidden_effect = { pantheon_broken_oath_effect = yes }
		}
	}
}

# Divine Retribution: someone has attacked the god, a devout follower, or a holy site. The aggressor is open to
# every wrath (no sin-only limit), but the protection rules and the cooldown still hold. 2.3: the call is a RECORD
# (common/scripted_effects/pantheon_retribution_effects.txt); this event carries its defender, attacker and token, and
# every answer re-validates them before it mutates and closes only that call. "Let it wait" leaves it open.
pantheon.0080 = {
	type = character_event
	title = pantheon.0080.t
	desc = pantheon.0080.desc
	theme = war
	left_portrait = {
		character = scope:pantheon_victim
		animation = war_attacker
	}
	right_portrait = {
		character = scope:pantheon_defender
		animation = war_defender
	}
	# The call the event carries (scope:pantheon_retri_defender / _attacker / _token) must still be open and live, or -- a 2.2.2
	# save's pending event, which carries nothing -- the god's two old slots must name a war that is still on (the
	# immediate adopts them into a record, once).
	trigger = {
		OR = {
			pantheon_retri_chain_ok_trigger = yes
			AND = {
				NOT = { exists = scope:pantheon_retri_token }
				exists = var:pantheon_wrath_target
				exists = var:pantheon_retribution_defender
				var:pantheon_wrath_target = { is_alive = yes }
				var:pantheon_retribution_defender = {
					is_alive = yes
					is_at_war_with = root.var:pantheon_wrath_target
				}
			}
		}
	}
	# Frame the chain: the aggressor is the victim, the context is 2, the page restrictions start clear. A Retribution no
	# longer touches a sworn oath (2.3, R8): only a wrath on the sworn offender fulfils it.
	immediate = {
		pantheon_retri_adopt_legacy_effect = yes
		pantheon_retri_frame_effect = yes
	}

	option = {
		name = pantheon.0080.smite
		trigger = {
			scope:pantheon_victim = {
				custom_tooltip = { text = pantheon_wrath_cooldown_tt NOT = { has_variable = pantheon_wrath_cooldown } }
				custom_tooltip = { text = pantheon_wrath_protected_tt NOT = { pantheon_wrath_protected_trigger = { GOD = root } } }
			}
			custom_tooltip = { text = pantheon_needs_worship_curse_tt var:pantheon_worship >= pantheon_cost_curse }
		}
		show_as_unavailable = { always = yes }
		hidden_effect = {
			pantheon_retri_prepare_effect = yes
			pantheon_retri_answer_smite_effect = yes
		}
		custom_tooltip = pantheon_worship_line_curse
	}

	option = {
		name = pantheon.0080.curse_lands
		trigger = {
			scope:pantheon_victim = {
				custom_tooltip = { text = pantheon_wrath_cooldown_tt NOT = { has_variable = pantheon_wrath_cooldown } }
				custom_tooltip = { text = pantheon_wrath_protected_tt NOT = { pantheon_wrath_protected_trigger = { GOD = root } } }
			}
			custom_tooltip = { text = pantheon_needs_worship_disaster_tt var:pantheon_worship >= pantheon_cost_disaster }
		}
		show_as_unavailable = { always = yes }
		hidden_effect = {
			pantheon_retri_prepare_effect = yes
			pantheon_retri_answer_lands_effect = yes
		}
		custom_tooltip = pantheon_worship_line_disaster
	}

	option = {
		name = pantheon.0080.host
		trigger = {
			custom_tooltip = { text = pantheon_needs_worship_host_tt var:pantheon_worship >= pantheon_cost_host }
			scope:pantheon_defender = {
				is_landed = yes
				is_at_war = yes
				any_character_war = { NOT = { has_variable = pantheon_host_called } }
			}
		}
		show_as_unavailable = {
			scope:pantheon_defender = {
				is_landed = yes
				is_at_war = yes
				any_character_war = { NOT = { has_variable = pantheon_host_called } }
			}
		}
		hidden_effect = {
			pantheon_retri_prepare_effect = yes
			pantheon_retri_answer_host_effect = yes
		}
		custom_tooltip = pantheon_worship_line_host
	}

	# 1.6: a free demigod lent to the defender as Commander (600 Worship). Hidden, not greyed: the 3-greyed cap.
	# 2.3 B4: starts SENDING MODE (kind 3, recipient = the attacked defender, token = the call's); the Children tab's Send lends
	# the demigod and then closes the call. The call stays OPEN meanwhile, so Cancel leaves it answerable from the Prayers tab.
	option = {
		name = pantheon.0080.send
		trigger = {
			NOT = { scope:pantheon_defender = root }
			scope:pantheon_defender = { is_landed = yes }
			custom_tooltip = { text = pantheon_has_free_demigod_tt root = { any_child = { pantheon_free_demigod_trigger = { GOD = root } } } }
			custom_tooltip = { text = pantheon_needs_worship_lend_tt var:pantheon_worship >= pantheon_cost_lend }
		}
		custom_tooltip = pantheon.0080.send.tt
		hidden_effect = {
			pantheon_retri_prepare_effect = yes
			pantheon_retri_answer_send_effect = yes
		}
	}

	# 2.3: Let it wait -- decide later from the Prayers tab. The call stays open for as long as the war lasts.
	option = {
		name = pantheon.0080.wait
		custom_tooltip = pantheon.0080.wait.tt
		hidden_effect = { pantheon_retri_prepare_effect = yes }
	}

	option = {
		name = pantheon.0080.mortals
		custom_tooltip = pantheon.0080.mortals.tt
		hidden_effect = {
			pantheon_retri_prepare_effect = yes
			pantheon_retri_answer_mortals_effect = yes
		}
	}
}
