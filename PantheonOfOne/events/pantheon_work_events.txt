namespace = pantheon_work

# Pantheon of One 2.8 (#59): works. Root is the god in every event; scope:scheme (the work), scope:owner (the god) and scope:target (the
# lord) travel in from the scheme's on_monthly. 0001-0003 are the outcome (hidden runner, the success window, the failure window);
# 0100-0150 are the six fear events and 0200-0250 the six kindness events (2-3 options each, none greyed).

# A won work, run as the god. The success effect (pantheon_work_effects.txt) converts the lord and pays out.
pantheon_work.0001 = {
	type = character_event
	hidden = yes
	trigger = {
		has_trait = pantheon_divine
		exists = scope:target
		exists = scope:pantheon_work_lean
	}
	immediate = {
		pantheon_work_success_effect = { LEAN = scope:pantheon_work_lean }
	}
}

# A failed or exposed work: the lord turns hostile and the seed on their capital loses time.
pantheon_work.0003 = {

	right_portrait = {

	}

	trigger = {
		has_trait = pantheon_divine
		exists = scope:target
	}

	option = {
		name = pantheon_work.0003.ok
	}
}

# ---- Fear: disasters, plague, nightmares and burnt holdings (0100-0150) ----
# Every option but "Let it be" adds progress through pantheon_work_progress_effect (a share of the phase, divided by the target's rank). "Let it be" adds none, so a work left alone
# stalls. Choices cost gold, prestige, piety or stress, never Worship; some carry exposure risk (fear's is the higher). A few push the target's
# capital-county meter. 0150 and 0250 offer the switch to the other path.

pantheon_work.0100 = {
	type = character_event
	title = pantheon_work.0100.t
	desc = pantheon_work.0100.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		character = scope:target
		animation = worry
	}

	trigger = {
		has_trait = pantheon_divine
		exists = scope:scheme
		exists = scope:target
		scope:target = { is_alive = yes }
	}

	option = {
		name = pantheon_work.0100.bold
		pantheon_work_progress_effect = { FRACTION = 0.3 }
		add_stress = minor_stress_gain
		pantheon_work_meter_push_effect = { AMOUNT = 8 }
		pantheon_work_risk_effect = { CHANCE = 15 }
	}

	option = {
		name = pantheon_work.0100.soft
		pantheon_work_progress_effect = { FRACTION = 0.15 }
		pantheon_work_risk_effect = { CHANCE = 5 }
	}

	option = {
		name = pantheon_work.0100.let
	}
}

# A won work: what it brought. The lord has already converted; the reward is the Worship lump the success effect kept on the god.

	type = character_event
	title = pantheon_work.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:target = { has_variable = pantheon_fear_convert } }
				desc = pantheon_work.0002.desc_fear
			}
			desc = pantheon_work.0002.desc_kind
		}
	}
	theme = faith
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		character = scope:target
		animation = admiration
	}

	trigger = {
		has_trait = pantheon_divine
		exists = scope:target
	}

	option = {
		name = pantheon_work.0002.ok
	}
}
