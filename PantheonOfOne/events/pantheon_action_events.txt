namespace = pantheon

# Pantheon of One 2.0: the work behind the right-click interactions and the Pantheon window. Hidden events on the
# god, so root is the god as every Pantheon effect expects. Each interaction records its choice in variables on the
# god (never inherited saved scopes: the 1.6 lesson) and fires one of these.

# Bless: var:pantheon_ui_target is the follower, var:pantheon_ui_choice the blessing.
pantheon.0100 = {
	type = character_event
	hidden = yes
	immediate = {
		var:pantheon_act_target ?= { save_scope_as = pantheon_target }
		# No branch runs (and no Devotion is paid) unless a choice was actually sent (I1)
		if = {
			limit = { exists = var:pantheon_ui_choice }
			if = { limit = { var:pantheon_ui_choice = flag:champion } pantheon_grant_trait_effect = { TRAIT = pantheon_champion } }
			else_if = { limit = { var:pantheon_ui_choice = flag:warlord } pantheon_grant_trait_effect = { TRAIT = pantheon_warlord } }
			else_if = { limit = { var:pantheon_ui_choice = flag:sovereign } pantheon_grant_trait_effect = { TRAIT = pantheon_sovereign } }
			else_if = { limit = { var:pantheon_ui_choice = flag:sage } pantheon_grant_trait_effect = { TRAIT = pantheon_sage } }
			else_if = { limit = { var:pantheon_ui_choice = flag:harvest } pantheon_grant_realm_effect = { KEY = harvest MODIFIER = pantheon_harvest } }
			else_if = { limit = { var:pantheon_ui_choice = flag:walls } pantheon_grant_realm_effect = { KEY = walls MODIFIER = pantheon_walls } }
			else_if = { limit = { var:pantheon_ui_choice = flag:warding } pantheon_grant_realm_effect = { KEY = warding MODIFIER = pantheon_warding } }
			else_if = { limit = { var:pantheon_ui_choice = flag:vigor } pantheon_grant_vigor_effect = yes }
			else_if = { limit = { var:pantheon_ui_choice = flag:fertility } pantheon_grant_fertility_effect = yes }
			else_if = { limit = { var:pantheon_ui_choice = flag:lineage } pantheon_grant_lineage_effect = yes }
			else_if = { limit = { var:pantheon_ui_choice = flag:host } pantheon_call_host_effect = yes }
			# 2.4: Bless: Freedom. The Great price when the prisoner is a valuable prisoner or an enemy war leader, the ordinary
			# price otherwise; charged here, BEFORE the grant (which releases them). Skipped if they are already free.
			else_if = {
				limit = {
					var:pantheon_ui_choice = flag:freedom
					exists = scope:pantheon_target
					scope:pantheon_target = { is_imprisoned = yes }
				}
				if = {
					limit = {
						scope:pantheon_target = {
							OR = {
								pantheon_valuable_prisoner_trigger = yes
								pantheon_war_leader_prisoner_trigger = yes
							}
						}
					}
					change_variable = { name = pantheon_worship subtract = pantheon_cost_boon_freedom }
				}
				else = {
					change_variable = { name = pantheon_worship subtract = pantheon_cost_freedom }
				}
				pantheon_grant_freedom_effect = yes
				# 2.4 repair 1 (R3): a pending Freedom prayer of this prisoner is answered by the release; close it silently
				if = {
					limit = { scope:pantheon_target = { has_variable = pantheon_prayer_open } }
					pantheon_prayer_close_core_effect = { WHO = scope:pantheon_target }
					pantheon_ui_refresh_if_open_effect = yes
				}
			}
			# 2.4 (R3): the generic +10 for a Bless is skipped for Freedom: its own Devotion is the whole reward
			if = {
				limit = { NOT = { var:pantheon_ui_choice = flag:freedom } }
				pantheon_change_devotion_effect = { AMOUNT = 10 }
			}
		}
		remove_variable = pantheon_ui_choice
	}
}

# Pass Judgement: var:pantheon_wrath_target is the judged, var:pantheon_ui_choice the wrath. 2.3 (R18): the Judgement
# interaction's route -- it announces context 3 (var:pantheon_wrath_context) and this runs only for that context, so
# a Retribution call is neither read nor closed here (the landed effect closes calls only in context 2).
pantheon.0106 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				has_variable = pantheon_wrath_context
				var:pantheon_wrath_context = 3
			}
			var:pantheon_wrath_target ?= { save_scope_as = pantheon_victim }
			if = { limit = { var:pantheon_ui_choice = flag:strike } pantheon_wrath_strike_effect = yes }
			else_if = { limit = { var:pantheon_ui_choice = flag:maimed } pantheon_wrath_maim_effect = { TRAIT = maimed } }
			else_if = { limit = { var:pantheon_ui_choice = flag:blind } pantheon_wrath_maim_effect = { TRAIT = blind } }
			else_if = { limit = { var:pantheon_ui_choice = flag:wound } pantheon_wrath_maim_effect = { TRAIT = wounded_3 } }
			else_if = { limit = { var:pantheon_ui_choice = flag:curse } pantheon_wrath_curse_effect = yes }
		}
		remove_variable = pantheon_ui_choice
	}
}

# Wrath upon their lands: as above, plus var:pantheon_wrath_county (read by pantheon_wrath_origin_effect).
# 2.3 (R18): the Lands interaction's route, context 4, as 0106 is context 3.
pantheon.0107 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				has_variable = pantheon_wrath_context
				var:pantheon_wrath_context = 4
			}
			var:pantheon_wrath_target ?= { save_scope_as = pantheon_victim }
			if = { limit = { var:pantheon_ui_choice = flag:pestilence } pantheon_wrath_plague_effect = { INTENSITY = minor COST = pantheon_cost_pestilence } }
			else_if = { limit = { var:pantheon_ui_choice = flag:great_dying } pantheon_wrath_plague_effect = { INTENSITY = major COST = pantheon_cost_great_dying } }
			else_if = { limit = { var:pantheon_ui_choice = flag:disaster } pantheon_wrath_disaster_effect = yes }
		}
		remove_variable = pantheon_ui_choice
		# A send with nothing selected leaves this on the god; never leave it to strike a later, different
		# victim (I1 scenario B).
		if = { limit = { has_variable = pantheon_wrath_county } remove_variable = pantheon_wrath_county }
	}
}

# 2.1 Grant their boon (unasked): var:pantheon_act_target is the follower. The request bound to this route is the
# one they hold now; pantheon_grant_great_boon_effect revalidates it, as for the prayer's Grant.
pantheon.0111 = {
	type = character_event
	hidden = yes
	immediate = {
		if = { limit = { exists = scope:pantheon_boon_event_follower } clear_saved_scope = pantheon_boon_event_follower }
		var:pantheon_act_target ?= { save_scope_as = pantheon_boon_event_follower }
		if = {
			limit = { exists = scope:pantheon_boon_event_follower scope:pantheon_boon_event_follower = { has_variable = pantheon_aspiration_id } }
			save_scope_value_as = { name = pantheon_boon_event_id value = scope:pantheon_boon_event_follower.var:pantheon_aspiration_id }
			pantheon_grant_great_boon_effect = { UNASKED = yes }
		}
		# 2.6 (Task 6): the hand-off is used up; nothing reads it later
		if = { limit = { has_variable = pantheon_act_target } remove_variable = pantheon_act_target }
		# 2.6 repair 1 (Codex review 2): a granted boon leaves the Great Prayers list at once
		pantheon_ui_refresh_if_open_effect = yes
	}
}

# 2.4 Free a demigod (pantheon_free_demigod_interaction and the Children row): var:pantheon_act_target is the demigod.
# Hidden, on the god. Re-checks that they are still held by someone else (a stale click does nothing), charges the Great
# price for a valuable prisoner or an enemy war leader and the ordinary price otherwise (only if the god can pay), then
# the shared core effect. NO Devotion.
pantheon.0113 = {
	type = character_event
	hidden = yes
	immediate = {
		if = { limit = { exists = scope:pantheon_target } clear_saved_scope = pantheon_target }
		var:pantheon_act_target ?= { save_scope_as = pantheon_target }
		if = {
			limit = {
				exists = scope:pantheon_target
				scope:pantheon_target = {
					has_trait = pantheon_demigod
					is_imprisoned = yes
					NOT = { is_imprisoned_by = root }
				}
			}
			if = {
				limit = {
					scope:pantheon_target = {
						OR = {
							pantheon_valuable_prisoner_trigger = yes
							pantheon_war_leader_prisoner_trigger = yes
						}
					}
				}
				if = {
					limit = { pantheon_can_afford_trigger = { COST = pantheon_cost_boon_freedom } }
					change_variable = { name = pantheon_worship subtract = pantheon_cost_boon_freedom }
					pantheon_free_prisoner_core_effect = yes
				}
			}
			else_if = {
				limit = { pantheon_can_afford_trigger = { COST = pantheon_cost_freedom } }
				change_variable = { name = pantheon_worship subtract = pantheon_cost_freedom }
				pantheon_free_prisoner_core_effect = yes
			}
		}
	}
}

# 2.4 Dissolve the union (pantheon_dissolve_interaction and the Children row): var:pantheon_act_target is the wed demigod.
# Hidden, on the god. Re-checks that they are still the god's wed, married child (a stale click does nothing), then the
# shared effect, which charges the reason's price and ends the marriage.
pantheon.0114 = {
	type = character_event
	hidden = yes
	immediate = {
		if = { limit = { exists = scope:pantheon_dissolve_demigod } clear_saved_scope = pantheon_dissolve_demigod }
		var:pantheon_act_target ?= { save_scope_as = pantheon_dissolve_demigod }
		if = {
			limit = {
				exists = scope:pantheon_dissolve_demigod
				scope:pantheon_dissolve_demigod = {
					has_trait = pantheon_demigod
					is_child_of = root
					has_variable = pantheon_wed_to_house
					is_married = yes
				}
			}
			pantheon_dissolve_effect = yes
		}
	}
}

# 2.4 Freedom: a follower was just imprisoned (pantheon_on_imprison_freedom). Hidden, on the god, so root is the god as
# delivery and aspiration creation require. First the prune (R15): a newly jailed follower's older, unrelated prayer is
# invalid in a cell and must be closed BEFORE the pass, which skips anyone with an open prayer. Then the pass.
pantheon.0115 = {
	type = character_event
	hidden = yes
	immediate = {
		pantheon_prayer_prune_effect = yes
		pantheon_freedom_pass_effect = yes
	}
}
