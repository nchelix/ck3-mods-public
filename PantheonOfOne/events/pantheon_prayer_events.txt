namespace = pantheon

# The prayer timer, every pantheon_prayer_interval_days (the pantheon_prayer_frequency game rule). Started
# by pantheon_take_throne_effect, like the Worship tick. Stops sending prayers if the rule is Off, but keeps
# quietly rescheduling itself so a later rule change takes effect without restarting.
pantheon.0050 = {
	type = character_event
	hidden = yes
	trigger = { has_trait = pantheon_divine }
	immediate = {
		if = {
			limit = { NOT = { has_game_rule = pantheon_prayers_off } }
			pantheon_send_prayer_effect = yes
		}
		# 2.3 (R4): a saved scope is inherited by this timer chain; never let the next pass see a stale candidate
		pantheon_prayer_clear_delivery_effect = yes
		trigger_event = { id = pantheon.0050 days = pantheon_prayer_interval_days }
	}
}

# A prayer. Root is the god; scope:pantheon_suppliant is whoever prayed (a follower, or about one time in
# eight a bordering stranger of another faith); scope:pantheon_prayer_delivered is the number of their prayer RECORD
# (common/scripted_effects/pantheon_prayer_effects.txt), delivered with the event as persistent saved scopes by
# pantheon_deliver_prayer_effect. pantheon_prayer_need (1-8, set when they were picked) names which need they bring.
# The trigger accepts a prayer whose record is open with the SAME token (an answered or replaced one stays out), or the
# 2.2.2 legacy case: no token delivered and the god still has pantheon_suppliant_var for someone with no record. The
# immediate adopts that legacy prayer once (a record, six months to answer) and forgets the variable.
# Options: Grant, Sign (followers), Send a child, Let it wait, Refuse (followers) or Dismiss (strangers). At most
# three can grey (Grant, Sign, Send), well within the cap (proven in game). Every answer option calls the adopt step
# first, so a 2.2.2 event that was already on screen when the save loaded still works.
pantheon.0060 = {
	type = character_event
	trigger = {
		OR = {
			AND = {
				exists = scope:pantheon_suppliant
				exists = scope:pantheon_prayer_delivered
				pantheon_prayer_is_open_trigger = yes
				pantheon_prayer_token_ok_trigger = yes
			}
			AND = {
				NOT = { exists = scope:pantheon_prayer_delivered }
				exists = var:pantheon_suppliant_var
				var:pantheon_suppliant_var = {
					is_alive = yes
					NOT = { has_variable = pantheon_prayer_open }
				}
			}
		}
	}
	immediate = {
		# Legacy (2.2.2 save): the suppliant was only kept on the god. Pick them up and adopt the prayer, once.
		if = {
			limit = {
				NOT = { exists = scope:pantheon_suppliant }
				exists = var:pantheon_suppliant_var
			}
			var:pantheon_suppliant_var = { save_scope_as = pantheon_suppliant }
		}
		pantheon_prayer_adopt_legacy_effect = yes
		# 1.6: can one of the god's free demigods wed this suppliant? Worked out with the marriage block lifted.
		if = { limit = { has_variable = pantheon_send_wed_ok } remove_variable = pantheon_send_wed_ok }
		pantheon_lift_marriage_block_effect = yes
		pantheon_mark_wed_spouse_effect = { SPOUSE = scope:pantheon_suppliant }
		if = { limit = { any_child = { has_variable = pantheon_can_wed_here } } set_variable = pantheon_send_wed_ok }
		pantheon_restore_marriage_block_effect = yes
		# 2.3 (R1/R14): the prices live on the suppliant's RECORD (snapshotted when the prayer opened), never in a
		# god-wide variable: two prayers open at once cannot overwrite each other's price. Button text cannot read the
		# suppliant (error.log, 1.3 test), so the prices are shown in the desc below, which can.
	}
	title = pantheon.0060.t
	desc = {
		triggered_desc = {
			trigger = { NOT = { scope:pantheon_suppliant = { faith = root.faith } } }
			desc = pantheon.0060.desc_stranger
		}
		first_valid = {
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 1 } }
				desc = pantheon.0060.desc_host
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 2 } }
				desc = pantheon.0060.desc_fertility
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 3 } }
				desc = pantheon.0060.desc_vigor
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 4 } }
				desc = pantheon.0060.desc_warding
			}
			# 2.2.1 (#2): a landless debtor is freed of their debts only (no harvest is on offer)
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 5 NOT = { any_held_county = { always = yes } } } }
				desc = pantheon.0060.desc_debt
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 5 } }
				desc = pantheon.0060.desc_harvest
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 6 var:pantheon_prayer_gift = 1 } }
				desc = pantheon.0060.desc_gift_champion
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 6 var:pantheon_prayer_gift = 2 } }
				desc = pantheon.0060.desc_gift_warlord
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 6 var:pantheon_prayer_gift = 3 } }
				desc = pantheon.0060.desc_gift_sovereign
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 6 var:pantheon_prayer_gift = 4 } }
				desc = pantheon.0060.desc_gift_sage
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 8 } }
				desc = pantheon.0060.desc_spouse
			}
			triggered_desc = {
				trigger = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 7 } }
				desc = pantheon.0060.desc_healing
			}
			desc = pantheon.0060.desc_gift
		}
		triggered_desc = {
			trigger = { scope:pantheon_suppliant = { pantheon_wavering_trigger = yes } }
			desc = pantheon.0060.devotion_wavering
		}
		triggered_desc = {
			trigger = {
				scope:pantheon_suppliant = {
					has_variable = pantheon_devotion
					NOT = { pantheon_wavering_trigger = yes }
					NOT = { pantheon_devout_trigger = yes }
				}
			}
			desc = pantheon.0060.devotion_faithful
		}
		triggered_desc = {
			trigger = { scope:pantheon_suppliant = { pantheon_devout_trigger = yes } }
			desc = pantheon.0060.devotion_devout
		}
		# 2.3 (R14): what answering costs, read from the prayer RECORD (a legacy event adopted later has none yet)
		triggered_desc = {
			trigger = {
				scope:pantheon_suppliant = {
					has_variable = pantheon_prayer_price
					faith = root.faith
				}
			}
			desc = pantheon.0060.prices
		}
		triggered_desc = {
			trigger = {
				scope:pantheon_suppliant = {
					has_variable = pantheon_prayer_price
					NOT = { faith = root.faith }
				}
			}
			desc = pantheon.0060.prices_stranger
		}
	}
	theme = faith
	left_portrait = {
		character = scope:pantheon_suppliant
		animation = personality_zealous
	}

	option = {
		name = pantheon.0060.grant
		trigger = {
			custom_tooltip = {
				text = pantheon_needs_worship_prayer_tt
				OR = {
					scope:pantheon_suppliant = { NOT = { has_variable = pantheon_prayer_price } }
					var:pantheon_worship >= scope:pantheon_suppliant.var:pantheon_prayer_price
				}
			}
		}
		show_as_unavailable = { always = yes }
		custom_tooltip = pantheon.0060.grant.tt
		hidden_effect = {
			pantheon_prayer_adopt_legacy_effect = yes
			pantheon_grant_prayer_effect = yes
		}
		# fix round 1 (F8): the cost line is read from the RECORD's snapshot; a record-less legacy event shows nothing
		if = {
			limit = { scope:pantheon_suppliant = { has_variable = pantheon_prayer_price } }
			custom_tooltip = pantheon_worship_line_record_prayer
		}
	}

	option = {
		name = pantheon.0060.sign
		trigger = {
			scope:pantheon_suppliant = { faith = root.faith }
			custom_tooltip = {
				text = pantheon_needs_worship_sign_tt
				OR = {
					scope:pantheon_suppliant = { NOT = { has_variable = pantheon_prayer_sign_price } }
					var:pantheon_worship >= scope:pantheon_suppliant.var:pantheon_prayer_sign_price
				}
			}
		}
		show_as_unavailable = {
			scope:pantheon_suppliant = { faith = root.faith }
		}
		custom_tooltip = pantheon.0060.sign.tt
		hidden_effect = {
			pantheon_prayer_adopt_legacy_effect = yes
			pantheon_sign_prayer_effect = yes
		}
		if = {
			limit = { scope:pantheon_suppliant = { has_variable = pantheon_prayer_sign_price } }
			custom_tooltip = pantheon_worship_line_record_sign
		}
	}

	# 1.6: Send one of your children. Host -> a demigod lent as Commander; Spouse -> a demigod wed to them.
	# The third greyable option; Refuse never greys.
	# 2.3 B4: it starts SENDING MODE and does not answer the prayer: the pending send is stored on the god (kind 1 Host /
	# 2 Spouse, recipient, the prayer's token), the Pantheon window opens on the Children tab, and a demigod's Send button
	# there lends / weds them at the usual price and then closes the prayer. If the send is cancelled the prayer stays
	# open (this event's other options and the Prayers tab still answer it).
	option = {
		name = pantheon.0060.send
		trigger = {
			scope:pantheon_suppliant = { faith = root.faith }
			custom_tooltip = { text = pantheon_has_free_demigod_tt root = { any_child = { pantheon_free_demigod_trigger = { GOD = root } } } }
			OR = {
				AND = {
					scope:pantheon_suppliant = { var:pantheon_prayer_need = 1 }
					custom_tooltip = { text = pantheon_needs_worship_lend_tt var:pantheon_worship >= pantheon_cost_lend }
				}
				AND = {
					scope:pantheon_suppliant = { var:pantheon_prayer_need = 8 }
					custom_tooltip = { text = pantheon_needs_worship_wed_tt var:pantheon_worship >= pantheon_cost_wed }
					# A Spouse answer always has someone free to marry (safe weddings, 1.6 final review); worked
					# out in this event's immediate, with the marriage block lifted (can_not_marry would always fail here).
					custom_tooltip = { text = pantheon_has_free_demigod_tt has_variable = pantheon_send_wed_ok }
				}
			}
		}
		show_as_unavailable = {
			scope:pantheon_suppliant = { faith = root.faith }
			custom_tooltip = { text = pantheon_has_free_demigod_tt root = { any_child = { pantheon_free_demigod_trigger = { GOD = root } } } }
			OR = {
				AND = {
					scope:pantheon_suppliant = { var:pantheon_prayer_need = 1 }
				}
				AND = {
					scope:pantheon_suppliant = { var:pantheon_prayer_need = 8 }
					custom_tooltip = { text = pantheon_has_free_demigod_tt has_variable = pantheon_send_wed_ok }
				}
			}
		}
		custom_tooltip = pantheon.0060.send.tt
		hidden_effect = {
			pantheon_prayer_adopt_legacy_effect = yes
			pantheon_prayer_send_entry_effect = yes
		}
	}

	# 2.3: close this event and leave the prayer open (the Prayers tab answers it later, until it lapses)
	option = {
		name = pantheon.0060.wait
		custom_tooltip = pantheon.0060.wait.tt
		# 2.3 review fix: an already-displayed 2.2.2 event has no record yet; waiting must create it like every answer does
		hidden_effect = { pantheon_prayer_adopt_legacy_effect = yes }
	}

	option = {
		name = pantheon.0060.refuse
		trigger = {
			scope:pantheon_suppliant = { faith = root.faith }
		}
		custom_tooltip = pantheon.0060.refuse.tt
		hidden_effect = {
			pantheon_prayer_adopt_legacy_effect = yes
			pantheon_refuse_prayer_effect = yes
		}
	}

	# A bordering stranger's plea is turned away: nothing happens to them
	option = {
		name = pantheon.0060.dismiss
		trigger = {
			scope:pantheon_suppliant = { NOT = { faith = root.faith } }
		}
		custom_tooltip = pantheon.0060.dismiss.tt
		hidden_effect = {
			pantheon_prayer_adopt_legacy_effect = yes
			pantheon_dismiss_prayer_effect = yes
		}
	}
}

# 2.3 (R2): the deadline of one prayer, scheduled by pantheon_prayer_create_effect six months after it opened.
# Hidden; root is the god; it carries the suppliant and the token of the prayer it was scheduled for. See
# pantheon_prayer_deadline_effect: invalidation first, then the lapse (a follower -5 Devotion, both a toast).
pantheon.0061 = {
	type = character_event
	hidden = yes
	immediate = {
		pantheon_prayer_deadline_effect = yes
	}
}
