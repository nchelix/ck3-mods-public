# Pantheon of One scripted triggers.

# Does this follower still carry any boon? (Each boon's variable expires with it.)
pantheon_has_any_boon_trigger = {
	OR = {
		has_variable = pantheon_boon_trait
		has_variable = pantheon_boon_harvest
		has_variable = pantheon_boon_walls
		has_variable = pantheon_boon_warding
		has_variable = pantheon_boon_vigor
		has_variable = pantheon_boon_fertility
		has_variable = pantheon_boon_lineage
	}
}

# May the god conceive a child with this follower? Root is the god; scope:pantheon_target the follower.
pantheon_can_conceive_trigger = {
	NOT = { has_character_modifier = pantheon_forsworn }
	# Pregnancy is asked only of women (asked of a man, it made the menu option vanish in game)
	trigger_if = {
		limit = { is_female = yes }
		is_pregnant = no
	}
	scope:pantheon_target = {
		is_adult = yes
		sex_opposite_of = root
		trigger_if = {
			limit = { is_female = yes }
			is_pregnant = no
		}
	}
}

# ---- 1.3: Devotion bands and prayer needs. Called on the character being checked.

# Below 25: gives no Worship, and is tested each year for leaving the faith.
pantheon_wavering_trigger = {
	has_variable = pantheon_devotion
	var:pantheon_devotion < 25
}

# 75 and up: double Worship, and never leaves the faith.
pantheon_devout_trigger = {
	has_variable = pantheon_devotion
	var:pantheon_devotion >= 75
}

# The same disease traits pantheon_divine_health_effect cures on a god (common/scripted_effects/pantheon_effects.txt).
pantheon_is_diseased_trigger = {
	OR = {
		has_trait = ill
		has_trait = pneumonic
		has_trait = great_pox
		has_trait = early_great_pox
		has_trait = lovers_pox
		has_trait = leper
		has_trait = consumption
		has_trait = cancer
		has_trait = typhus
		has_trait = bubonic_plague
		has_trait = smallpox
		has_trait = measles
		has_trait = dysentery
		has_trait = ergotism
		has_trait = gout_ridden
	}
}

# At war, and losing it (the war leader with a negative war score). Skipped if the Host is already out.
pantheon_need_host_trigger = {
	is_at_war = yes
	any_character_war = {
		OR = {
			AND = { primary_attacker = prev attacker_war_score < 0 }
			AND = { primary_defender = prev defender_war_score < 0 }
		}
		NOT = { has_variable = pantheon_host_called }
	}
}

# An adult with no living children.
pantheon_need_fertility_trigger = {
	is_adult = yes
	is_married = yes
	NOT = { any_child = { always = yes } }
	NOT = { has_variable = pantheon_boon_fertility }
	# 1.6: not while a child is on the way (theirs or any wife's), and only a couple who can still conceive:
	# the woman under the engine's childbearing age (fertility alone never reaches 0 with age)
	trigger_if = { limit = { is_female = yes } is_pregnant = no age < define:NChildbirth|MAX_FEMALE_REPRODUCTION_AGE }
	NOT = { any_spouse = { is_female = yes is_pregnant = yes } }
	primary_spouse ?= { trigger_if = { limit = { is_female = yes } age < define:NChildbirth|MAX_FEMALE_REPRODUCTION_AGE } fertility > 0 }
	fertility > 0
}

# Aged 60 or more (the sick pray for Divine Healing).
pantheon_need_vigor_trigger = {
	age >= 60
	NOT = { has_variable = pantheon_boon_vigor }
}

# Their capital county has an epidemic.
pantheon_need_warding_trigger = {
	exists = capital_county
	capital_county = { has_province_with_epidemic = { intensity = any } }
	NOT = { has_variable = pantheon_boon_warding }
}

# Gold below zero, or in debt.
pantheon_need_harvest_trigger = {
	OR = {
		gold < 0
		is_in_debt = yes
	}
	NOT = { has_variable = pantheon_boon_harvest }
}

# Otherwise: only a landed ruler, and even then at most a third as often (the chance is rolled where this
# is used, common/scripted_effects/pantheon_devotion_effects.txt's pantheon_add_prayer_candidate_effect).
pantheon_need_gift_trigger = {
	is_landed = yes
	NOT = { has_variable = pantheon_boon_trait }
}

# Does root have any of the above needs at all? (existence only; the effect that acts on it picks in order)
pantheon_has_a_need_trigger = {
	OR = {
		pantheon_need_host_trigger = yes
		pantheon_need_fertility_trigger = yes
		pantheon_need_vigor_trigger = yes
		pantheon_need_warding_trigger = yes
		pantheon_need_harvest_trigger = yes
		pantheon_need_gift_trigger = yes
	}
}

# Does an independent ruler of another faith, bordering one of the god's faith's rulers, exist? Root is the god.
pantheon_has_a_bordering_stranger_trigger = {
	any_ruler = {
		faith = root.faith
		any_neighboring_top_liege_realm_owner = {
			is_independent_ruler = yes
			NOT = { faith = root.faith }
			NOT = { has_variable = pantheon_prayed_recently }
			pantheon_has_a_need_trigger = yes
		}
	}
}

# Diseased or wounded: they pray for Divine Healing (1.3).
pantheon_need_healing_trigger = {
	OR = {
		pantheon_is_diseased_trigger = yes
		has_trait = wounded_1
		has_trait = wounded_2
		has_trait = wounded_3
	}
}

# Unmarried, childless adults may pray for a spouse and children (1.3, the user's idea).
pantheon_need_spouse_trigger = {
	is_adult = yes
	age < 45
	is_married = no
	is_betrothed = no
	NOT = { any_child = { always = yes } }
	NOT = { has_variable = pantheon_boon_fertility }
	trigger_if = { limit = { is_female = yes } is_pregnant = no }
}

# ---- 1.4 Wrath. In each trigger below, this is the candidate and $GOD$ the god. Pass GOD as a saved scope
# or root, never `this`: inside a nested block `this` changes, and $GOD$.faith would read the wrong faith.

# Never the god, a god's child, or one of the god's devout (spec, Rules for all wraths).
pantheon_wrath_protected_trigger = {
	OR = {
		this = $GOD$
		has_trait = pantheon_divine
		has_trait = pantheon_demigod
		AND = {
			faith = $GOD$.faith
			pantheon_devout_trigger = yes
		}
	}
}

# Another faith, and either at war with a follower of the god's faith or holding a county with one of its
# holy sites. In a war scope, prev is the candidate.
pantheon_enemy_trigger = {
	NOT = { faith = $GOD$.faith }
	OR = {
		any_character_war = {
			OR = {
				AND = {
					is_attacker = prev
					any_war_defender = { faith = $GOD$.faith }
				}
				AND = {
					is_defender = prev
					any_war_attacker = { faith = $GOD$.faith }
				}
			}
		}
		any_held_county = {
			any_county_province = {
				barony = { is_holy_site_of = $GOD$.faith }
			}
		}
	}
}

# A follower carrying a trait the god's rite counts as a crime (proven in the prototype: vanilla's
# trait_is_criminal_in_rite_trigger, one has_trait check per crime trait).
pantheon_criminal_trigger = {
	faith = $GOD$.faith
	OR = {
		AND = {
			has_trait = kinslayer_1
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:kinslayer_1 GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = kinslayer_2
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:kinslayer_2 GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = kinslayer_3
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:kinslayer_3 GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = adulterer
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:adulterer GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = fornicator
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:fornicator GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = sodomite
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:sodomite GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = incestuous
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:incestuous GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = deviant
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:deviant GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = witch
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:witch GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = cannibal
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:cannibal GENDER_CHARACTER = this }
		}
	}
}

# A follower who is Wavering, or carries a trait the god's rite counts as a sin (proven: num_sinful_traits).
pantheon_sinner_trigger = {
	faith = $GOD$.faith
	OR = {
		pantheon_wavering_trigger = yes
		num_sinful_traits = {
			value > 0
			target = $GOD$.rite
		}
	}
}

# A sin weighs less than a crime (user, 2026-10-03): only Curse or Maim, and -25 Devotion rather than 0.
pantheon_sin_only_trigger = {
	pantheon_sinner_trigger = { GOD = $GOD$ }
	NOT = { pantheon_criminal_trigger = { GOD = $GOD$ } }
}

# Who Pass Judgement shows on: enemies, apostates, criminals, sinners; never the protected.
# pantheon_debug_enemy is set only by the local PantheonOfOneDebug mod.
pantheon_judgement_target_trigger = {
	is_alive = yes
	NOT = { pantheon_wrath_protected_trigger = { GOD = $GOD$ } }
	OR = {
		pantheon_enemy_trigger = { GOD = $GOD$ }
		has_variable = pantheon_apostate
		pantheon_criminal_trigger = { GOD = $GOD$ }
		pantheon_sinner_trigger = { GOD = $GOD$ }
		has_variable = pantheon_debug_enemy
	}
}

# Divine Retribution: a war on the god (by anyone), or by someone of another faith on a devout follower or
# over a holy site of the god's faith (user, 2026-10-05: followers' own quarrels stay mortal business).
pantheon_retribution_war_trigger = {
	NOT = { scope:attacker = $GOD$ }
	OR = {
		scope:defender = $GOD$
		AND = {
			NOT = { scope:attacker = { faith = $GOD$.faith } }
			OR = {
				scope:defender = {
					faith = $GOD$.faith
					pantheon_devout_trigger = yes
				}
				# Over a holy site, but only against one of the faithful: two infidels fighting over land with
				# one of your holy sites is not an attack on you (1.5 in-game test, 2026-10-06).
				# casus_belli ?=: it is sometimes an invalid (weak) object (1.5 Task 0 error.log; vanilla
				# guards it the same way).
				AND = {
					scope:defender = { faith = $GOD$.faith }
					scope:war = {
						casus_belli ?= {
							any_target_title = {
								any_in_de_jure_hierarchy = {
									tier = tier_county
									any_county_province = {
										barony = { is_holy_site_of = $GOD$.faith }
									}
								}
							}
						}
					}
				}
			}
		}
	}
}

# ---- 1.5 Great Prayers. In each trigger, this is the character and $GOD$ the god (pass root or a saved
# scope, never `this`).

# A follower who may hold a grievance: of the god's faith, alive, at 90+ Devotion.
pantheon_grievance_eligible_trigger = {
	is_alive = yes
	faith = $GOD$.faith
	has_variable = pantheon_devotion
	var:pantheon_devotion >= 90
}

# A grievance the god can be asked to answer: still eligible, and the offender alive and not protected.
pantheon_grievance_valid_trigger = {
	pantheon_grievance_eligible_trigger = { GOD = $GOD$ }
	has_variable = pantheon_grievance
	var:pantheon_grievance_target ?= {
		is_alive = yes
		NOT = { pantheon_wrath_protected_trigger = { GOD = $GOD$ } }
	}
}

# Root is the god: does any follower hold a valid grievance? The god's faith's rulers and the god's
# courtiers only, as the prayer pick (no world scans beyond it).
pantheon_has_a_great_prayer_trigger = {
	OR = {
		any_ruler = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
		}
		any_courtier = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
		}
	}
}

# ---- 1.6 Demigods. A demigod the god may lend or wed: their own child, grown, of the faith, unlanded, free.
pantheon_free_demigod_trigger = {
	has_trait = pantheon_demigod
	is_alive = yes
	is_adult = yes
	is_child_of = $GOD$
	faith = $GOD$.faith
	is_landed = no
	is_imprisoned = no
	NOT = { has_variable = pantheon_lent_to }
	NOT = { has_variable = pantheon_wed_to_house }
}

# 2.0: could this character (the spouse) wed the god's demigod $DEMIGOD$? Vanilla's can_marry_character_trigger
# minus the demigod's can_not_marry variable: pantheon_wed_demigod_effect lifts that block itself (proven in the
# 2.0 Task 0 prototype). Used by the Wed interaction, which the engine also uses to pre-filter the spouse picker.
pantheon_could_wed_demigod_trigger = {
	save_temporary_scope_as = pantheon_could_wed_spouse
	is_adult = yes
	can_marry_trigger = yes
	$DEMIGOD$ = {
		is_alive = yes
		is_married = no
		NOT = { has_trait_with_flag = can_not_marry }
	}
	trigger_if = {
		limit = { $DEMIGOD$ = { allowed_to_marry_same_sex_trigger = no } }
		sex_opposite_of = $DEMIGOD$
	}
	NOT = { has_opinion_modifier = { modifier = divorced_me_opinion target = $DEMIGOD$ } }
	rite = {
		rite_allows_marriage_consanguinity_trigger = {
			CHARACTER_1 = scope:pantheon_could_wed_spouse
			CHARACTER_2 = $DEMIGOD$
		}
	}
	$DEMIGOD$.rite = {
		rite_allows_marriage_consanguinity_trigger = {
			CHARACTER_1 = scope:pantheon_could_wed_spouse
			CHARACTER_2 = $DEMIGOD$
		}
	}
}

# 2.0.1: can the god afford $COST$ Worship? Interactions evaluate their send_option and is_valid checks even for a
# player who is not a god (so has no pantheon_worship yet): reading the variable then logged "Failed to fetch
# variable" every frame (2.0 error.log). The limit keeps the read away from anyone without Worship.
pantheon_can_afford_trigger = {
	trigger_if = {
		limit = { has_variable = pantheon_worship }
		var:pantheon_worship >= $COST$
	}
	trigger_else = { always = no }
}

# ---- 2.1 Great Boons. this is the character unless noted; $GOD$ is the god (root or a saved scope) and $WHO$ a
# saved scope, never `this` or prev.

# Seriously ill or wounded: the deadly illnesses and grave wounds (user, 2026-10-08: not the common ill, gout,
# lover's pox or early great pox, which would keep replacing other asks; the god's cure still removes them all)
pantheon_seriously_ill_trigger = {
	OR = {
		has_trait = pneumonic
		has_trait = great_pox
		has_trait = leper
		has_trait = consumption
		has_trait = cancer
		has_trait = typhus
		has_trait = bubonic_plague
		has_trait = smallpox
		has_trait = measles
		has_trait = dysentery
		has_trait = ergotism
		has_trait = wounded_3
		has_trait = maimed
	}
}

# Who may hold an aspiration: the Great Prayer pool, but never the god, a demigod or a Godsblooded
pantheon_boon_eligible_trigger = {
	pantheon_grievance_eligible_trigger = { GOD = $GOD$ }
	NOT = { has_trait = pantheon_divine }
	NOT = { has_trait = pantheon_demigod }
	NOT = { has_trait = pantheon_godsblooded }
}

# Kind 2: the primary heir, a spouse or a child is alive and seriously ill or wounded
pantheon_boon_kin_trigger = {
	OR = {
		primary_heir ?= { is_alive = yes pantheon_seriously_ill_trigger = yes }
		any_spouse = { is_alive = yes pantheon_seriously_ill_trigger = yes }
		any_child = { is_alive = yes pantheon_seriously_ill_trigger = yes }
	}
}

# War scope. Kind 3: a holy war $WHO$ leads and is losing, against an unbeliever leading the other side
pantheon_boon_holy_war_trigger = {
	OR = {
		using_cb = religious_war
		using_cb = major_religious_war
		using_cb = excommunication_war
		using_cb = undirected_great_holy_war
		using_cb = directed_great_holy_war
	}
	trigger_if = {
		limit = { primary_attacker = $WHO$ }
		primary_defender = { NOT = { faith = $GOD$.faith } }
		attacker_war_score < 0
	}
	trigger_else_if = {
		limit = { primary_defender = $WHO$ }
		primary_attacker = { NOT = { faith = $GOD$.faith } }
		defender_war_score < 0
	}
	trigger_else = { always = no }
}

# Title scope. Kind 1: a crown $WHO$ may be raised to: unheld, half its counties in their realm (their own
# counties included: Task 0 round 3), and never a vassal raised to their liege's rank
pantheon_boon_crown_title_trigger = {
	save_temporary_scope_as = pantheon_boon_crown_chk
	NOT = { exists = holder }
	any_de_jure_county = {
		percent >= 0.5
		holder ?= { OR = { this = $WHO$ target_is_liege_or_above = $WHO$ } }
	}
	trigger_if = {
		limit = { $WHO$ = { is_independent_ruler = no } }
		$WHO$.liege.highest_held_title_tier > scope:pantheon_boon_crown_chk.tier
	}
}

# Kind 1: a duke or king, and the de jure title one tier above their primary title, containing their capital,
# is such a crown
pantheon_boon_crown_trigger = {
	save_temporary_scope_as = pantheon_boon_crown_who
	is_landed = yes
	OR = {
		AND = {
			primary_title = { tier = tier_duchy }
			capital_county.de_jure_liege.de_jure_liege ?= { pantheon_boon_crown_title_trigger = { WHO = scope:pantheon_boon_crown_who } }
		}
		AND = {
			primary_title = { tier = tier_kingdom }
			capital_county.de_jure_liege.de_jure_liege.de_jure_liege ?= { pantheon_boon_crown_title_trigger = { WHO = scope:pantheon_boon_crown_who } }
		}
	}
}

# Kind 4: a landed ruler, 40 or older, with no living child
pantheon_boon_heir_trigger = {
	is_landed = yes
	age >= 40
	NOT = { any_child = { is_alive = yes } }
}

# The succession laws under which a newborn only child becomes the primary heir (Task 0 rounds 3-4). Seniority
# (single_heir_dynasty_house), elective and every other law offer "a child" instead. A title succession law on
# the primary title (every law in vanilla's 01_title_succession_laws.txt: elective, appointment, noble family,
# temporal head, law of the land) overrides the realm's law for that title, so it offers "a child" too.
pantheon_boon_heir_law_trigger = {
	OR = {
		has_realm_law = confederate_partition_succession_law
		has_realm_law = partition_succession_law
		has_realm_law = high_partition_succession_law
		has_realm_law = single_heir_succession_law
		has_realm_law = single_heir_succession_law_youngest
	}
	primary_title = {
		NOT = {
			OR = {
				has_title_law = feudal_elective_succession_law
				has_title_law = princely_elective_succession_law
				has_title_law = saxon_elective_succession_law
				has_title_law = scandinavian_elective_succession_law
				has_title_law = gaelic_elective_succession_law
				has_title_law = tribal_elective_succession_law
				has_title_law = temporal_head_of_faith_succession_law
				has_title_law = noble_family_succession_law
				has_title_law = law_of_the_land_succession_law
				has_title_law = confederation_elective_succession_law
				has_title_law = celestial_ministry_appointment_succession_law
				has_title_law = celestial_grand_marshal_appointment_succession_law
			}
		}
	}
}

# Kind 5: at least half of their domain counties are of another faith
pantheon_boon_conversion_trigger = {
	is_landed = yes
	any_held_county = { percent >= 0.5 NOT = { faith = $GOD$.faith } }
}

# Kind 6: in debt
pantheon_boon_fortune_trigger = {
	gold < 0
}

# The aspiration they hold is still valid (spec, Validity table), and they're still eligible to hold one
pantheon_aspiration_valid_trigger = {
	pantheon_boon_eligible_trigger = { GOD = $GOD$ }
	has_variable = pantheon_aspiration
	has_variable = pantheon_aspiration_kind
	save_temporary_scope_as = pantheon_boon_valid_who
	trigger_if = {
		limit = { var:pantheon_aspiration_kind = 1 } var:pantheon_aspiration_target ?= { pantheon_boon_crown_title_trigger = { WHO = scope:pantheon_boon_valid_who } }
	}
	trigger_else_if = {
		limit = { var:pantheon_aspiration_kind = 2 } var:pantheon_aspiration_target ?= { is_alive = yes pantheon_seriously_ill_trigger = yes }
	}
	trigger_else_if = {
		limit = { var:pantheon_aspiration_kind = 3 } var:pantheon_aspiration_target ?= { pantheon_boon_holy_war_trigger = { WHO = scope:pantheon_boon_valid_who GOD = $GOD$ } }
	}
	trigger_else_if = {
		limit = { var:pantheon_aspiration_kind = 4 } pantheon_boon_heir_trigger = yes
	}
	trigger_else_if = {
		limit = { var:pantheon_aspiration_kind = 5 } pantheon_boon_conversion_trigger = { GOD = $GOD$ }
	}
	trigger_else_if = {
		limit = { var:pantheon_aspiration_kind = 6 } pantheon_boon_fortune_trigger = yes
	}
	trigger_else = { always = no }
}

# The request this event or route was bound to ($ID$, its captured request ID) is still the one they hold, and
# it hasn't run out its five years. Refuse checks only this; Grant also needs it valid.
pantheon_aspiration_bound_trigger = {
	has_variable = pantheon_aspiration
	has_variable = pantheon_aspiration_live
	has_variable = pantheon_aspiration_id
	var:pantheon_aspiration_id = $ID$
}

pantheon_aspiration_current_trigger = {
	pantheon_aspiration_bound_trigger = { ID = $ID$ }
	pantheon_aspiration_valid_trigger = { GOD = $GOD$ }
}

# A boon the prayer timer may send now: valid, live, and not already offered in an open event
pantheon_boon_sendable_trigger = {
	is_alive = yes
	has_variable = pantheon_aspiration_live
	NOT = { has_variable = pantheon_aspiration_offered }
	pantheon_aspiration_valid_trigger = { GOD = $GOD$ }
}

# God scope: can the god pay for $WHO$'s boon? Always through the guarded check (2.0.1)
pantheon_boon_affordable_trigger = {
	trigger_if = {
		limit = { $WHO$ = { has_variable = pantheon_aspiration_kind var:pantheon_aspiration_kind = 1 } } pantheon_can_afford_trigger = { COST = pantheon_cost_boon_crown }
	}
	trigger_else_if = {
		limit = { $WHO$ = { has_variable = pantheon_aspiration_kind var:pantheon_aspiration_kind = 2 } } pantheon_can_afford_trigger = { COST = pantheon_cost_boon_kin }
	}
	trigger_else_if = {
		limit = { $WHO$ = { has_variable = pantheon_aspiration_kind var:pantheon_aspiration_kind = 3 } } pantheon_can_afford_trigger = { COST = pantheon_cost_boon_war }
	}
	trigger_else_if = {
		limit = { $WHO$ = { has_variable = pantheon_aspiration_kind var:pantheon_aspiration_kind = 4 } } pantheon_can_afford_trigger = { COST = pantheon_cost_boon_heir }
	}
	trigger_else_if = {
		limit = { $WHO$ = { has_variable = pantheon_aspiration_kind var:pantheon_aspiration_kind = 5 } } pantheon_can_afford_trigger = { COST = pantheon_cost_boon_conversion }
	}
	trigger_else_if = {
		limit = { $WHO$ = { has_variable = pantheon_aspiration_kind var:pantheon_aspiration_kind = 6 } } pantheon_can_afford_trigger = { COST = pantheon_cost_boon_fortune }
	}
	trigger_else = { always = no }
}

# God scope: does any holder have a boon the timer may send?
pantheon_has_a_great_boon_trigger = {
	any_in_list = { variable = pantheon_boon_holders pantheon_boon_sendable_trigger = { GOD = root } }
}
