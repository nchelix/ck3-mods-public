# Pantheon of One scripted triggers.

# Does this follower still carry any boon? (Each boon's variable expires with it.)
pantheon_has_any_boon_trigger = {
	OR = {
		has_variable = pantheon_boon_trait
		has_variable = pantheon_boon_harvest
		has_variable = pantheon_boon_walls
		has_variable = pantheon_boon_warding
		has_variable = pantheon_boon_vigor
		has_variable = pantheon_boon_fertility
		has_variable = pantheon_boon_lineage
		# 2.2.1 (#4): timed markers of a Great Boon, a fulfilled Great Prayer and forgiven debts, so the
		# quarterly pruning (pantheon.0010) and the window's Blessed count keep their recipients
		has_variable = pantheon_boon_great
		has_variable = pantheon_boon_prayer
		has_variable = pantheon_boon_debt
		has_variable = pantheon_boon_conception
		# 2.4 repair 1 (R4): Freedom is a blessing
		has_variable = pantheon_boon_freedom
	}
}

# May the god conceive a child with this follower? Root is the god; scope:pantheon_target the follower.
pantheon_can_conceive_trigger = {
	NOT = { has_character_modifier = pantheon_forsworn }
	# Pregnancy is asked only of women (asked of a man, it made the menu option vanish in game)
	trigger_if = {
		limit = { is_female = yes }
		is_pregnant = no
	}
	scope:pantheon_target = {
		is_adult = yes
		sex_opposite_of = root
		trigger_if = {
			limit = { is_female = yes }
			is_pregnant = no
		}
	}
}

# ---- 1.3: Devotion bands and prayer needs. Called on the character being checked.

# Below 25: gives no Worship, and is tested each year for leaving the faith.
pantheon_wavering_trigger = {
	trigger_if = {
		limit = { has_variable = pantheon_devotion }
		var:pantheon_devotion < 25
	}
	trigger_else = { always = no }
}

# 75 and up: double Worship, and never leaves the faith.
pantheon_devout_trigger = {
	trigger_if = {
		limit = { has_variable = pantheon_devotion }
		var:pantheon_devotion >= 75
	}
	trigger_else = { always = no }
}

# At war, and losing it (the war leader with a negative war score). Skipped if the Host is already out.
pantheon_need_host_trigger = {
	is_at_war = yes
	any_character_war = {
		OR = {
			AND = { primary_attacker = prev attacker_war_score < 0 }
			AND = { primary_defender = prev defender_war_score < 0 }
		}
		NOT = { has_variable = pantheon_host_called }
	}
}

# An adult with no living children.
pantheon_need_fertility_trigger = {
	is_adult = yes
	is_married = yes
	NOT = { any_child = { always = yes } }
	NOT = { has_variable = pantheon_boon_fertility }
	# 1.6: not while a child is on the way (theirs or any wife's), and only a couple who can still conceive:
	# the woman under the engine's childbearing age (fertility alone never reaches 0 with age)
	trigger_if = { limit = { is_female = yes } is_pregnant = no age < define:NChildbirth|MAX_FEMALE_REPRODUCTION_AGE }
	NOT = { any_spouse = { is_female = yes is_pregnant = yes } }
	primary_spouse ?= { trigger_if = { limit = { is_female = yes } age < define:NChildbirth|MAX_FEMALE_REPRODUCTION_AGE } fertility > 0 }
	fertility > 0
}

# Aged 60 or more (the sick pray for Divine Healing).
pantheon_need_vigor_trigger = {
	age >= 60
	NOT = { has_variable = pantheon_boon_vigor }
}

# Their capital county has an epidemic.
pantheon_need_warding_trigger = {
	exists = capital_county
	capital_county = { has_province_with_epidemic = { intensity = any } }
	NOT = { has_variable = pantheon_boon_warding }
}

# Gold below zero, or in debt.
pantheon_need_harvest_trigger = {
	OR = {
		gold < 0
		is_in_debt = yes
	}
	NOT = { has_variable = pantheon_boon_harvest }
	# 2.2.1 (#2): a landless debtor whose debts were forgiven does not pray again for the same debt
	NOT = { has_variable = pantheon_boon_debt }
}

# Otherwise: only a landed ruler, and even then at most a third as often (the chance is rolled where this
# is used, common/scripted_effects/pantheon_devotion_effects.txt's pantheon_add_prayer_candidate_effect).
pantheon_need_gift_trigger = {
	is_landed = yes
	NOT = { has_variable = pantheon_boon_trait }
}

# Does root have any of the above needs at all? (existence only; the effect that acts on it picks in order)
pantheon_has_a_need_trigger = {
	OR = {
		pantheon_need_host_trigger = yes
		pantheon_need_fertility_trigger = yes
		pantheon_need_vigor_trigger = yes
		pantheon_need_warding_trigger = yes
		pantheon_need_harvest_trigger = yes
		pantheon_need_gift_trigger = yes
	}
}

# Does an independent ruler of another faith, bordering one of the god's faith's rulers, exist? Root is the god.
pantheon_has_a_bordering_stranger_trigger = {
	any_ruler = {
		faith = root.faith
		any_neighboring_top_liege_realm_owner = {
			is_independent_ruler = yes
			NOT = { faith = root.faith }
			NOT = { has_variable = pantheon_prayed_recently }
			pantheon_has_a_need_trigger = yes
		}
	}
}

# Unmarried, childless adults may pray for a spouse and children (1.3, the user's idea).
pantheon_need_spouse_trigger = {
	is_adult = yes
	age < 45
	is_married = no
	is_betrothed = no
	NOT = { any_child = { always = yes } }
	NOT = { has_variable = pantheon_boon_fertility }
	trigger_if = { limit = { is_female = yes } is_pregnant = no }
	can_marry_common_trigger = yes # 2.4.1 #30: monks, nuns and anyone who may not marry never pray for a spouse (vanilla 00_marriage_triggers.txt)
}

# ---- 2.4 Freedom (need 9). Research (T1 step 1), from the 1.20 logs and vanilla:
#  - place_in_line_of_succession is a LANDED_TITLE trigger, 1-BASED (1 = the first heir), used by vanilla's
#    prisoners_of_war_trigger (00_war_and_peace_triggers.txt) as <war leader>.primary_title = { place_in_line_of_succession =
#    { target = <prisoner> value <= 3 } }: vanilla's "third in line or higher".
#  - ransom_cost_value is a vanilla script value read in the PRISONER's scope; it finds the jailer through `imprisoner`.
#  - `imprisoner` is a character scope link (event_targets.log); in a prisoner's trigger it is the jailer.

# In the prisoner's scope; no parameter. The prisoner leads one side of a war against their jailer's side
# (the war collapses, and peace frees them: they get no Freedom prayer of either kind; only Bless can free them).
pantheon_war_leader_prisoner_trigger = {
	exists = imprisoner
	any_character_war = {
		is_war_leader = prev
		OR = {
			AND = { is_attacker = prev is_defender = prev.imprisoner }
			AND = { is_defender = prev is_attacker = prev.imprisoner }
		}
	}
}

# In the prisoner's scope; no parameter. A valuable prisoner in vanilla's sense: their jailer is fighting a war whose
# OTHER side is led by X, the prisoner is not X, and holds place 1-3 in the line of succession to X's primary title.
pantheon_valuable_prisoner_trigger = {
	exists = imprisoner
	save_temporary_scope_as = pantheon_valuable_prisoner
	imprisoner = {
		any_character_war = {
			OR = {
				AND = {
					is_attacker = prev
					primary_defender = {
						NOT = { this = scope:pantheon_valuable_prisoner }
						primary_title ?= { place_in_line_of_succession = { target = scope:pantheon_valuable_prisoner value <= 3 } }
					}
				}
				AND = {
					is_defender = prev
					primary_attacker = {
						NOT = { this = scope:pantheon_valuable_prisoner }
						primary_title ?= { place_in_line_of_succession = { target = scope:pantheon_valuable_prisoner value <= 3 } }
					}
				}
			}
		}
	}
}

# In the candidate's scope; the god is ROOT (no parameter, like the other need triggers; called `= yes`). A follower of
# the god's faith in someone else's prison: not a hostage, not a demigod, not held by the god, and neither a valuable
# prisoner nor a war leader (kind 7 or nothing for them).
pantheon_need_freedom_trigger = {
	is_imprisoned = yes
	exists = imprisoner
	NOT = { is_imprisoned_by = root }
	is_hostage = no
	faith = root.faith
	# 2.4 repair 1 (R1): a landed ruler, or a devout follower (Devotion 75+). A landless, non-devout prisoner is not prayed for.
	OR = {
		is_landed = yes
		pantheon_devout_trigger = yes
	}
	NOT = { has_trait = pantheon_demigod }
	pantheon_valuable_prisoner_trigger = no
	pantheon_war_leader_prisoner_trigger = no
}

# ---- 2.5 Kin pray for a jailed loved one (need 10; kin kind 7). Engine names: is_sibling_of, is_parent_of, is_child_of,
# is_spouse_of, has_relation_rival, opinion (triggers.log 5907, 5747, 5206, 5913, 4294, 7066); vanilla uses them at
# 00_court_position_effects.txt:454, 00_ai_conqueror_effects.txt:830, 00_accolades_scripted_effects.txt:3809,
# 00_adultery_effects.txt:381 and 00_holy_order_effects.txt:1031.

# In the candidate's scope; $GOD$ is the god (pass root or a saved scope, never `this`). A character who prays in the ordinary
# way: the three pools of pantheon_pick_follower_prayer_effect (rulers, the god's courtiers, Devotion 75+).
pantheon_ordinary_suppliant_trigger = {
	OR = {
		is_ruler = yes
		is_courtier_of = $GOD$
		pantheon_devout_trigger = yes
	}
}

# In the prisoner's scope; the god is ROOT (no parameter, called `= yes`). The ordinary kin route (need 10): someone else's
# prisoner who cannot pray for themselves (R1: never a valuable prisoner, never a war leader; a prisoner who can pray for
# themselves has the need-9 prayer). Any faith: the relative's faith is what counts.
pantheon_kin_jailed_trigger = {
	is_imprisoned = yes
	exists = imprisoner
	NOT = { is_imprisoned_by = root }
	is_hostage = no
	NOT = { this = root }
	NOT = { has_trait = pantheon_demigod }
	pantheon_war_leader_prisoner_trigger = no
	pantheon_valuable_prisoner_trigger = no
	pantheon_need_freedom_trigger = no
}

# In the prisoner's scope; root is the god. The valuable route (kin kind 7, R1): a valuable prisoner, not a war leader, who
# cannot hold their own Great Prayer (not of the faith, or below Devotion 90).
pantheon_kin_valuable_trigger = {
	is_imprisoned = yes
	exists = imprisoner
	NOT = { is_imprisoned_by = root }
	is_hostage = no
	NOT = { this = root }
	NOT = { has_trait = pantheon_demigod }
	pantheon_valuable_prisoner_trigger = yes
	pantheon_war_leader_prisoner_trigger = no
	NOT = { pantheon_boon_eligible_trigger = { GOD = root } }
}

# In the relative's scope. $PRISONER$ is the prisoner and $GOD$ the god (scope references, never `this`). The relative is a
# ADULT follower (repair 1: a child does not pray for a jailed relative) one link from the prisoner (parent, sibling, child
# or spouse: R2, no grandparents), is free, likes the prisoner
# (positive opinion, not a rival) and prays in the ordinary way. GOD is a parameter because the Great Prayer's validity is
# asked with GOD = scope:actor from an interaction, where root is not the god.
pantheon_kin_intercessor_trigger = {
	is_alive = yes
	is_adult = yes
	faith = $GOD$.faith
	NOT = { this = $GOD$ }
	NOT = { has_trait = pantheon_demigod }
	is_imprisoned = no
	OR = {
		is_sibling_of = $PRISONER$
		is_parent_of = $PRISONER$
		is_child_of = $PRISONER$
		is_spouse_of = $PRISONER$
	}
	opinion = { target = $PRISONER$ value > 0 }
	NOT = { has_relation_rival = $PRISONER$ }
	pantheon_ordinary_suppliant_trigger = { GOD = $GOD$ }
}

# In the relative's scope; the god is ROOT. Does the kin prayer (need 10) the relative opened still stand? The prisoner is
# still a kin-prayer case and the relative still passes the intercessor trigger for them. `?=` throughout: an unset
# variable fails the trigger instead of erroring.
pantheon_kin_need_valid_trigger = {
	save_temporary_scope_as = pantheon_kin_valid_relative
	var:pantheon_prayer_kin ?= {
		save_temporary_scope_as = pantheon_kin_valid_prisoner
		pantheon_kin_jailed_trigger = yes
		scope:pantheon_kin_valid_relative = { pantheon_kin_intercessor_trigger = { PRISONER = scope:pantheon_kin_valid_prisoner GOD = root } }
	}
}

# In the prisoner's scope. Is the reservation on them (pantheon_kin_prayer_for = a relative) still alive? It holds while that
# relative either has an open need-10 prayer for THIS prisoner or holds a kind 7 whose target is THIS prisoner. A stale
# marker is not live, so the pass skips live ones and overwrites dead ones (R3).
pantheon_kin_reservation_live_trigger = {
	save_temporary_scope_as = pantheon_kin_live_prisoner
	var:pantheon_kin_prayer_for ?= {
		is_alive = yes
		OR = {
			AND = {
				has_variable = pantheon_prayer_open
				var:pantheon_prayer_need ?= 10
				var:pantheon_prayer_kin ?= { this = scope:pantheon_kin_live_prisoner }
			}
			AND = {
				var:pantheon_aspiration_kind ?= 7
				var:pantheon_aspiration_target ?= { this = scope:pantheon_kin_live_prisoner }
			}
		}
	}
}

# ---- 1.4 Wrath. In each trigger below, this is the candidate and $GOD$ the god. Pass GOD as a saved scope
# or root, never `this`: inside a nested block `this` changes, and $GOD$.faith would read the wrong faith.

# Never the god, a god's child, or one of the god's devout (spec, Rules for all wraths).
pantheon_wrath_protected_trigger = {
	OR = {
		this = $GOD$
		has_trait = pantheon_divine
		has_trait = pantheon_demigod
		AND = {
			faith = $GOD$.faith
			pantheon_devout_trigger = yes
		}
	}
}

# Another faith, and either at war with a follower of the god's faith or holding a county with one of its
# holy sites. In a war scope, prev is the candidate.
pantheon_enemy_trigger = {
	NOT = { faith = $GOD$.faith }
	OR = {
		any_character_war = {
			OR = {
				AND = {
					is_attacker = prev
					any_war_defender = { faith = $GOD$.faith }
				}
				AND = {
					is_defender = prev
					any_war_attacker = { faith = $GOD$.faith }
				}
			}
		}
		any_held_county = {
			any_county_province = {
				barony = { is_holy_site_of = $GOD$.faith }
			}
		}
	}
}

# A follower carrying a trait the god's rite counts as a crime (proven in the prototype: vanilla's
# trait_is_criminal_in_rite_trigger, one has_trait check per crime trait).
pantheon_criminal_trigger = {

	OR = {

		}
		AND = {
			has_trait = kinslayer_2
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:kinslayer_2 GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = kinslayer_3
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:kinslayer_3 GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = adulterer
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:adulterer GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = fornicator
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:fornicator GENDER_CHARACTER = this }
		}
		AND = {
			has_trait = sodomite
			trait_is_criminal_in_rite_trigger = { RITE = $GOD$.rite TRAIT = trait:sodomite GENDER_CHARACTER = this }

# A follower who is Wavering, or carries a trait the god's rite counts as a sin (proven: num_sinful_traits).
pantheon_sinner_trigger = {
	faith = $GOD$.faith
	OR = {
		pantheon_wavering_trigger = yes
		num_sinful_traits = {
			value > 0
			target = $GOD$.rite
		}
	}
}

# A sin weighs less than a crime (user, 2026-10-03): only Curse or Maim, and -25 Devotion rather than 0.
pantheon_sin_only_trigger = {
	pantheon_sinner_trigger = { GOD = $GOD$ }
	NOT = { pantheon_criminal_trigger = { GOD = $GOD$ } }
}

# Who Pass Judgement shows on: enemies, apostates, criminals, sinners; never the protected.
# pantheon_debug_enemy is set only by the local PantheonOfOneDebug mod.
pantheon_judgement_target_trigger = {
	is_alive = yes
	NOT = { pantheon_wrath_protected_trigger = { GOD = $GOD$ } }
	OR = {
		pantheon_enemy_trigger = { GOD = $GOD$ }
		has_variable = pantheon_apostate
		pantheon_criminal_trigger = { GOD = $GOD$ }
		pantheon_sinner_trigger = { GOD = $GOD$ }
		has_variable = pantheon_debug_enemy
		# 2.4: a recent raider of this god's land (pantheon_raided_god, 5 years; set by the raid hook)
		var:pantheon_raided_god ?= $GOD$
	}
}

# Divine Retribution: a war on the god (by anyone), or by someone of another faith on a devout follower or
# over a holy site of the god's faith (user, 2026-10-05: followers' own quarrels stay mortal business).
pantheon_retribution_war_trigger = {
	NOT = { scope:attacker = $GOD$ }
	OR = {
		scope:defender = $GOD$
		AND = {
			NOT = { scope:attacker = { faith = $GOD$.faith } }
			OR = {
				scope:defender = {
					faith = $GOD$.faith
					pantheon_devout_trigger = yes
				}
				# Over a holy site, but only against one of the faithful: two infidels fighting over land with
				# one of your holy sites is not an attack on you (1.5 in-game test, 2026-10-06).
				# casus_belli ?=: it is sometimes an invalid (weak) object (1.5 Task 0 error.log; vanilla
				# guards it the same way).
				AND = {
					scope:defender = { faith = $GOD$.faith }
					scope:war = {
						casus_belli ?= {
							any_target_title = {
								any_in_de_jure_hierarchy = {
									tier = tier_county
									any_county_province = {
										barony = { is_holy_site_of = $GOD$.faith }
									}
								}
							}
						}
					}
				}
			}
		}
	}
}

# ---- 1.5 Great Prayers. In each trigger, this is the character and $GOD$ the god (pass root or a saved
# scope, never `this`).

# A follower who may hold a grievance: of the god's faith, alive, at 90+ Devotion.
pantheon_grievance_eligible_trigger = {
	is_alive = yes
	faith = $GOD$.faith
	trigger_if = {
		limit = { has_variable = pantheon_devotion }
		var:pantheon_devotion >= 90
	}
	trigger_else = { always = no }
}

# A grievance the god can be asked to answer: still eligible, and the offender alive and not protected.
pantheon_grievance_valid_trigger = {
	pantheon_grievance_eligible_trigger = { GOD = $GOD$ }
	has_variable = pantheon_grievance
	# 2.4: freedom before vengeance. Jailed, they ask for their freedom first (the Freedom prayer); only a vengeful
	# follower wants vengeance even from the cell.
	trigger_if = { limit = { is_imprisoned = yes } has_trait = vengeful }
	var:pantheon_grievance_target ?= {
		is_alive = yes
		NOT = { pantheon_wrath_protected_trigger = { GOD = $GOD$ } }
	}
}

# Root is the god: does any follower hold a valid grievance? The god's faith's rulers and the god's
# courtiers only, as the prayer pick (no world scans beyond it).
pantheon_has_a_great_prayer_trigger = {
	OR = {
		any_ruler = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
		}
		any_courtier = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
		}
	}
}

# 2.4 (R7): the same, for a grievance that is NOT a raid grievance (kind 6). The scheduler tries these first.
pantheon_has_a_strong_great_prayer_trigger = {
	OR = {
		any_ruler = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
			NOT = { var:pantheon_grievance_kind ?= 6 }
		}
		any_courtier = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
			NOT = { var:pantheon_grievance_kind ?= 6 }
		}
	}
}

# 2.4 (R7): a raid grievance (kind 6) is owed. The scheduler tries it last, only when nothing else was delivered.
pantheon_has_a_raid_great_prayer_trigger = {
	OR = {
		any_ruler = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
			var:pantheon_grievance_kind ?= 6
		}
		any_courtier = {
			faith = root.faith
			pantheon_grievance_valid_trigger = { GOD = root }
			var:pantheon_grievance_kind ?= 6
		}

	$GOD$ = { is_target_in_variable_list = { name = pantheon_open_retributions target = prev } }

# In the ATTACKER's scope: open AND live (see above). Records that fail this are pruned silently.
pantheon_retri_live_here_trigger = {
	pantheon_retri_open_here_trigger = { GOD = $GOD$ }
	var:pantheon_retri_defender ?= {
		is_alive = yes
		faith = $GOD$.faith
	}
	is_at_war_with = var:pantheon_retri_defender
}

# In the GOD's scope (root = the god): the Retribution chain's identity -- scope:pantheon_retri_defender,
# scope:pantheon_retri_attacker and the value scope:pantheon_retri_token, all PERSISTENT saved scopes carried by 0080,
# 0070-0073 and every effect they run -- still names the attacker's open, live call with the same token and defender.
# A replaced call (new token), an ended war, a dead side or a defender who left the faith all fail it.
pantheon_retri_chain_ok_trigger = {
	exists = scope:pantheon_retri_defender
	exists = scope:pantheon_retri_attacker
	exists = scope:pantheon_retri_token
	scope:pantheon_retri_attacker = {
		pantheon_retri_live_here_trigger = { GOD = root }
		var:pantheon_retri_token = scope:pantheon_retri_token
		var:pantheon_retri_defender = scope:pantheon_retri_defender
	}
}

# ---- 2.3 B5: wrath context (R18). scope:pantheon_wrath_ctx (a persistent saved value, set by the entry route and
# carried through the page chain): 1 sworn oath, 2 Retribution call, 3 Judgement interaction, 4 Lands interaction.

pantheon_wrath_ctx_is_retri_trigger = {
	exists = scope:pantheon_wrath_ctx
	scope:pantheon_wrath_ctx = 2
}

pantheon_wrath_ctx_is_oath_trigger = {
	exists = scope:pantheon_wrath_ctx
	scope:pantheon_wrath_ctx = 1
}

# Root = the god. A wrath page (0070-0073) may open: a Retribution chain must still match its call (context 2 + token);
# any chain needs its victim alive -- the saved victim when the chain has one, else the god's variable (the route that
# sets it before the first page).
pantheon_wrath_page_ok_trigger = {
	trigger_if = {
		limit = { pantheon_wrath_ctx_is_retri_trigger = yes }
		pantheon_retri_chain_ok_trigger = yes
	}
	trigger_if = {
		limit = { exists = scope:pantheon_victim }
		scope:pantheon_victim = { is_alive = yes }
	}
	trigger_else = {
		exists = var:pantheon_wrath_target
		var:pantheon_wrath_target = { is_alive = yes }
	}
}

# Root = the god. A wrath may still be carried out: a Retribution chain re-validates its call right before it charges.
pantheon_wrath_mutation_ok_trigger = {
	trigger_if = {
		limit = { pantheon_wrath_ctx_is_retri_trigger = yes }
		pantheon_retri_chain_ok_trigger = yes
	}
	trigger_else = {
		always = yes
	}
}

# Root = the god. The page belongs to the SWORN OATH (context 1) and its victim is the sworn offender: only then does
# Stay break the oath.
# 2.3 review fix: a 2.2.2 sworn page that was already on screen when the save loaded has no saved context; it counts as
# the sworn oath's page too (narrowly: NO saved context at all, an oath is active and the page's victim is its offender).
pantheon_wrath_oath_page_trigger = {
	OR = {
		pantheon_wrath_ctx_is_oath_trigger = yes
		# a legacy page: no saved context, and not a 2.2.2 Retribution page (those carry scope:pantheon_defender)
		AND = {
			NOT = { exists = scope:pantheon_wrath_ctx }
			NOT = { exists = scope:pantheon_defender }
		}
	}
	has_variable = pantheon_great_prayer_active
	exists = scope:pantheon_victim
	var:pantheon_great_prayer_offender ?= scope:pantheon_victim
}

# 2.3 review fix: the page restrictions are derived from the page's OWN saved context, never from god-wide flags that
# another open page's route could overwrite. Root = the god, scope:pantheon_victim the page's victim.
# Harsh (only Strike or Maim offered): the sworn oath's page (context 1, or a legacy one) about the sworn offender, when
# the oath is harsh. A Retribution page (context 2) is never restricted.
pantheon_wrath_page_harsh_trigger = {
	pantheon_wrath_oath_page_trigger = yes
	pantheon_oath_harsh_trigger = yes
}

# Light (only Curse or Maim): the Judgement route (context 3) on a sin-only recipient -- a sin weighs less than a crime.
# 2.4 (R8): or the sworn oath's own page when the oath is light (a raid grievance, kind 6: pantheon_great_prayer_light).
pantheon_wrath_page_light_trigger = {
	OR = {
		AND = {
			exists = scope:pantheon_wrath_ctx
			scope:pantheon_wrath_ctx = 3
			exists = scope:pantheon_victim
			scope:pantheon_victim = { pantheon_sin_only_trigger = { GOD = root } }
		}
		AND = {
			pantheon_wrath_oath_page_trigger = yes
			has_variable = pantheon_great_prayer_light
		}
	}
}

# Root = the god. Is the sworn oath a harsh one (a torture: only Strike or Maim fulfils it)? The persistent
# pantheon_great_prayer_harsh, or -- for an oath sworn before 2.3 and not yet migrated -- the old page flag.
pantheon_oath_harsh_trigger = {
	OR = {
		has_variable = pantheon_great_prayer_harsh
		AND = {
			NOT = { has_variable = pantheon_great_prayer_kind }
			has_variable = pantheon_wrath_harsh
		}
	}
}

# Faith scope. 2.4.1 (#26): a faith that may be the one a reform just created, when on_faith_created hands over the OLD faith
# as root (pantheon_on_faith_created): same religion (the caller iterates it), not $OLD$, founded by $FOUNDER$, nobody in it yet.
pantheon_reform_new_faith_candidate_trigger = {
	NOT = { this = $OLD$ }
	exists = founder
	founder = $FOUNDER$
	NOT = { any_faith_character = { is_alive = yes } }
}
