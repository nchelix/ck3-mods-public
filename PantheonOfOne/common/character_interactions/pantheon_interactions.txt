# Pantheon of One: right-click a follower (or yourself) to open the Pantheon window (2.0; the 1.0-1.6 event pages
# are no longer reached from here). On yourself it opens The God tab.
pantheon_divine_will_interaction = {
	category = interaction_category_religion
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	desc = pantheon_divine_will_interaction_desc
	is_shown = {
		scope:actor = {
			is_ai = no
			has_trait = pantheon_divine
		}
		# Yourself from the moment you Ascend; your followers once you are enthroned
		OR = {
			scope:recipient = scope:actor
			AND = {
				scope:actor = { has_variable = pantheon_enthroned }
				scope:recipient = { faith = scope:actor.faith }
			}
		}
	}
	on_accept = {
		scope:actor = {
			# The window's own selection; never the god themselves (I3)
			if = {
				limit = { NOT = { scope:recipient = scope:actor } }
				set_variable = { name = pantheon_ui_target value = scope:recipient }
			}
			if = { limit = { scope:recipient = scope:actor } set_variable = { name = pantheon_ui_tab value = 3 } }
			else = { set_variable = { name = pantheon_ui_tab value = 1 } }
			trigger_event = pantheon.0108
		}
	}
	auto_accept = yes
	ai_will_do = { base = 0 }
}

# 2.0: the person wraths as radio options (the land wraths are pantheon_lands_interaction)
pantheon_judgement_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	desc = pantheon_judgement_interaction_desc
	is_shown = {
		scope:actor = {
			is_ai = no
			has_trait = pantheon_divine
			has_variable = pantheon_enthroned
		}
		scope:recipient = { pantheon_judgement_target_trigger = { GOD = scope:actor } }
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_tooltip = { text = pantheon_wrath_cooldown_tt NOT = { has_variable = pantheon_wrath_cooldown } }
		}
	}
	can_send = {
		custom_tooltip = {
			text = pantheon_pick_option_tt
			OR = {
				scope:pantheon_judge_strike = yes
				scope:pantheon_judge_maimed = yes
				scope:pantheon_judge_blind = yes
				scope:pantheon_judge_wound = yes
				scope:pantheon_judge_curse = yes
			}
		}
	}

	send_options_exclusive = yes
	send_option = {
		flag = pantheon_judge_strike
		localization = pantheon_judge_opt_strike
		is_valid = {
			scope:actor = { custom_tooltip = { text = pantheon_needs_worship_strike_tt var:pantheon_worship >= pantheon_cost_strike } }
			scope:recipient = { custom_tooltip = { text = pantheon_judge_sin_only_tt NOT = { pantheon_sin_only_trigger = { GOD = scope:actor } } } }
		}
	}
	send_option = {
		flag = pantheon_judge_maimed
		localization = pantheon_judge_opt_maimed
		is_valid = {
			scope:actor = { custom_tooltip = { text = pantheon_needs_worship_maim_tt var:pantheon_worship >= pantheon_cost_maim } }
			scope:recipient = { custom_tooltip = { text = pantheon_judge_already_tt NOT = { has_trait = maimed } } }
		}
	}
	send_option = {
		flag = pantheon_judge_blind
		localization = pantheon_judge_opt_blind
		is_valid = {
			scope:actor = { custom_tooltip = { text = pantheon_needs_worship_maim_tt var:pantheon_worship >= pantheon_cost_maim } }
			scope:recipient = { custom_tooltip = { text = pantheon_judge_already_tt NOT = { has_trait = blind } } }
		}
	}
	send_option = {
		flag = pantheon_judge_wound
		localization = pantheon_judge_opt_wound
		is_valid = {
			scope:actor = { custom_tooltip = { text = pantheon_needs_worship_maim_tt var:pantheon_worship >= pantheon_cost_maim } }
			scope:recipient = { custom_tooltip = { text = pantheon_judge_already_tt NOT = { has_trait = wounded_3 } } }
		}
	}
	send_option = {
		flag = pantheon_judge_curse
		localization = pantheon_judge_opt_curse
		is_valid = {
			scope:actor = { custom_tooltip = { text = pantheon_needs_worship_curse_tt var:pantheon_worship >= pantheon_cost_curse } }
		}
	}

	on_accept = {
		if = { limit = { scope:pantheon_judge_strike = yes } custom_tooltip = pantheon.0071.strike.tt }
		if = { limit = { scope:pantheon_judge_maimed = yes } custom_tooltip = pantheon_judge_maimed.tt }
		if = { limit = { scope:pantheon_judge_blind = yes } custom_tooltip = pantheon_judge_blind.tt }
		if = { limit = { scope:pantheon_judge_wound = yes } custom_tooltip = pantheon_judge_wound.tt }
		if = { limit = { scope:pantheon_judge_curse = yes } custom_tooltip = pantheon.0071.curse.tt }
		hidden_effect = {
			scope:actor = {
				# A wrath on the sworn offender answers the oath itself: pantheon_wrath_landed_effect must still
				# see pantheon_great_prayer_active (M9).
				if = {
					limit = { NOT = { var:pantheon_great_prayer_offender ?= scope:recipient } }
					pantheon_end_oath_effect = yes
				}
				set_variable = { name = pantheon_wrath_target value = scope:recipient }
				# A sin weighs less than a crime (the Great Prayer pages still read this)
				if = {
					limit = { scope:recipient = { pantheon_sin_only_trigger = { GOD = scope:actor } } }
					set_variable = pantheon_wrath_light
				}
				else_if = {
					limit = { has_variable = pantheon_wrath_light }
					remove_variable = pantheon_wrath_light
				}
				if = { limit = { scope:pantheon_judge_strike = yes } set_variable = { name = pantheon_ui_choice value = flag:strike } }
				else_if = { limit = { scope:pantheon_judge_maimed = yes } set_variable = { name = pantheon_ui_choice value = flag:maimed } }
				else_if = { limit = { scope:pantheon_judge_blind = yes } set_variable = { name = pantheon_ui_choice value = flag:blind } }
				else_if = { limit = { scope:pantheon_judge_wound = yes } set_variable = { name = pantheon_ui_choice value = flag:wound } }
				else_if = { limit = { scope:pantheon_judge_curse = yes } set_variable = { name = pantheon_ui_choice value = flag:curse } }
				trigger_event = pantheon.0106
			}
		}
	}
	auto_accept = yes
	ai_will_do = { base = 0 }
}

# 2.0: Lend a demigod. The demigod comes from the vanilla picker (secondary actor), the role from a radio option.
# Picks are required in can_send, never in is_valid (that greys the interaction out of the menu: Task 0).
pantheon_lend_interaction = {
	category = interaction_category_religion
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = pantheon_lend_interaction
	desc = pantheon_lend_interaction_desc
	is_shown = {
		scope:actor = {
			is_ai = no
			has_trait = pantheon_divine
			has_variable = pantheon_enthroned
		}
		scope:recipient = {
			NOT = { this = scope:actor }
			faith = scope:actor.faith
			is_landed = yes
		}
	}
	populate_actor_list = {
		scope:actor = {
			every_child = {
				limit = { pantheon_free_demigod_trigger = { GOD = scope:actor } }
				add_to_list = characters
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = { text = pantheon_needs_worship_lend_tt var:pantheon_worship >= pantheon_cost_lend }

		}
	}
	can_send = {
		custom_tooltip = { text = pantheon_pick_demigod_tt exists = scope:secondary_actor }
		custom_tooltip = {
			text = pantheon_pick_option_tt
			OR = {
				scope:pantheon_role_champion = yes
				scope:pantheon_role_councillor = yes
				scope:pantheon_role_commander = yes
			}
		}
	}

	send_options_exclusive = yes
	send_option = {
		flag = pantheon_role_champion
		localization = pantheon_menu.0520.champion
	}
	send_option = {
		flag = pantheon_role_councillor
		localization = pantheon_menu.0520.councillor
	}
	send_option = {
		flag = pantheon_role_commander
		localization = pantheon_menu.0520.commander

		hidden_effect = {

				if = { limit = { scope:pantheon_role_champion = yes } set_variable = { name = pantheon_ui_choice value = flag:champion } }
				else_if = { limit = { scope:pantheon_role_councillor = yes } set_variable = { name = pantheon_ui_choice value = flag:councillor } }
				else_if = { limit = { scope:pantheon_role_commander = yes } set_variable = { name = pantheon_ui_choice value = flag:commander } }
				trigger_event = pantheon.0102
			}
		}
	}
	auto_accept = yes
	ai_will_do = { base = 0 }
}

# 2.0: Wed a demigod into the house: the demigod (secondary actor) and the spouse (secondary recipient) both come
# from vanilla pickers. The pair check lives in is_valid_showing_failures_only behind a trigger_if; the engine uses
# it to show only spouses who can wed the chosen demigod (Task 0).
pantheon_wed_interaction = {
	category = interaction_category_religion
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = pantheon_wed_interaction
	desc = pantheon_wed_interaction_desc
	is_shown = {
		scope:actor = {
			is_ai = no
			has_trait = pantheon_divine
			has_variable = pantheon_enthroned
		}
		scope:recipient = {
			NOT = { this = scope:actor }
			faith = scope:actor.faith
			is_ruler = yes
			exists = house
		}
	}
	populate_actor_list = {
		scope:actor = {
			every_child = {
				limit = { pantheon_free_demigod_trigger = { GOD = scope:actor } is_married = no }
				add_to_list = characters
			}
		}
	}
	populate_recipient_list = {
		scope:recipient = {
			if = {
				limit = { is_married = no }
				add_to_list = characters
			}
			every_close_family_member = {
				limit = { is_adult = yes is_married = no OR = { is_child_of = scope:recipient is_sibling_of = scope:recipient } house = scope:recipient.house }
				add_to_list = characters
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = { text = pantheon_needs_worship_wed_tt var:pantheon_worship >= pantheon_cost_wed }
			custom_tooltip = { text = pantheon_has_free_demigod_tt any_child = { pantheon_free_demigod_trigger = { GOD = scope:actor } is_married = no } }
		}
		trigger_if = { limit = { exists = scope:secondary_actor exists = scope:secondary_recipient } scope:secondary_recipient = { custom_tooltip = { text = pantheon_cannot_wed_tt pantheon_could_wed_demigod_trigger = { DEMIGOD = scope:secondary_actor } } } }
	}
	can_send = {
		custom_tooltip = { text = pantheon_pick_demigod_tt exists = scope:secondary_actor }
		custom_tooltip = { text = pantheon_pick_spouse_tt exists = scope:secondary_recipient }
	}
	on_accept = {
		custom_tooltip = pantheon_wed.tt
		hidden_effect = {
			scope:actor = {
				set_variable = { name = pantheon_demigod_host value = scope:recipient }
				set_variable = { name = pantheon_chosen_demigod value = scope:secondary_actor }
				set_variable = { name = pantheon_chosen_spouse value = scope:secondary_recipient }
				trigger_event = pantheon.0103
			}
		}
	}
	auto_accept = yes
	ai_will_do = { base = 0 }
}
