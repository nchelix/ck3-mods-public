# The header's state line: no temple yet / mid-build / dormant (one reason: the god viewing their own tab
# always still has pantheon_divine, so the only dormancy a founder can see here is the county changing
# hands) / active.
pantheon_temple_state = {
	type = character
	text = {
		trigger = {
			NOT = { has_variable = pantheon_temple_tier }
			NOT = { has_variable = pantheon_temple_building }
		}
		localization_key = pantheon_temple_state_none
	}
	text = {
		# Batch u1 repair (rows 16-17): the build must read as PAUSED, not just "Building", while the
		# county has left faithful hands (pantheon_temple_build_paused_trigger, pantheon_temple_triggers.txt).
		trigger = { pantheon_temple_build_paused_trigger = yes }
		localization_key = pantheon_temple_state_build_paused
	}
	text = {
		trigger = { has_variable = pantheon_temple_building }
		localization_key = pantheon_temple_state_building
	}
	text = {
		trigger = { pantheon_temple_active_trigger = yes }
		localization_key = pantheon_temple_state_active
	}
	text = {
		fallback = yes
		localization_key = pantheon_temple_state_dormant
	}
}
