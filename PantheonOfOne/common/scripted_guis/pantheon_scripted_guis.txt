# Pantheon of One 2.0: the Pantheon window's buttons. Called from gui/pantheon_window.gui with
# GuiScope.SetRoot( GetPlayer.MakeScope ), so root and this are the god. Task 5 appends The God tab's buttons.

# The button by the portrait shows only for a god
pantheon_ui_is_god = {
	scope = character
	is_valid = { has_trait = pantheon_divine }
}

# Open (rebuilding the lists first) or close the window
pantheon_ui_toggle = {
	scope = character
	is_valid = { has_trait = pantheon_divine }
	effect = {
		if = {
			limit = { has_variable = pantheon_ui_open }
			remove_variable = pantheon_ui_open
			# A long-open window can't leave a stale pinned reference behind it (M5)
			clear_variable_list = pantheon_ui_followers
			clear_variable_list = pantheon_ui_blessed
			clear_variable_list = pantheon_ui_demigods
			clear_variable_list = pantheon_ui_houses
			# Batch 2r1 (repair 4): closing the window must not leave a miracle armed.
			pantheon_temple_clear_confirm_effect = yes
		}
		else = {
			if = { limit = { NOT = { has_variable = pantheon_ui_tab } } set_variable = { name = pantheon_ui_tab value = 1 } }
			pantheon_ui_rebuild_effect = yes
			# Batch 2r1 (repair 4): re-opening must not resume with a stale armed miracle either.
			pantheon_temple_clear_confirm_effect = yes
			set_variable = pantheon_ui_open
		}
	}
}

# The header's close button (M10): no is_valid, so losing pantheon_divine while the window is open can't trap it.
pantheon_ui_close = {
	scope = character
	effect = {
		remove_variable = pantheon_ui_open
		clear_variable_list = pantheon_ui_followers
		clear_variable_list = pantheon_ui_blessed
		clear_variable_list = pantheon_ui_demigods
		clear_variable_list = pantheon_ui_houses
		# Batch 2r1 (repair 4): same guard as the toggle's close branch.
		pantheon_temple_clear_confirm_effect = yes
	}
}

# A row click selects that follower
pantheon_ui_select = {
	scope = character
	saved_scopes = { target }
	effect = {
		set_variable = { name = pantheon_ui_target value = scope:target }
	}
}

# Followers: everyone, or only those carrying a blessing
pantheon_ui_blessed_only = {
	scope = character
	effect = {
		if = {
			limit = { has_variable = pantheon_ui_blessed_only }
			remove_variable = pantheon_ui_blessed_only
		}
		else = { set_variable = pantheon_ui_blessed_only }
	}
}

# A holy-site row's Found button. scope:landed_title is the county (AddScope('landed_title',
# Title.MakeScope), One Trait's 2.7 list form, OneTraitToRuleThemAll/gui/one_trait_panel.gui). Greyed with
# custom_tooltip reasons (never custom_description -- docs/selftest-kit/GUIDE.md, proven 2026-10-15).
pantheon_ui_temple_found = {
	scope = character
	is_shown = {
		NOT = { has_variable = pantheon_temple_province }
	}
	is_valid = {
		custom_tooltip = {
			text = pantheon_temple_not_faithful_tt
			scope:landed_title = { pantheon_temple_faithful_hands_trigger = { GOD = root } }
		}
		custom_tooltip = {
			text = pantheon_temple_no_site_tt
			scope:landed_title = { pantheon_temple_placeable_trigger = yes }
		}
		custom_tooltip = {
			text = pantheon_temple_cost_found_tt
			pantheon_can_afford_trigger = { COST = pantheon_temple_cost_1 }
		}
	}
	effect = {
		if = { limit = { has_variable = pantheon_temple_confirm } remove_variable = pantheon_temple_confirm }
		pantheon_temple_found_effect = { SITE = scope:landed_title }
		pantheon_ui_rebuild_effect = yes
	}
}

pantheon_ui_temple_upgrade = {
	scope = character
	is_shown = {
		has_variable = pantheon_temple_tier
		var:pantheon_temple_tier < 5
	}
	is_valid = {
		custom_tooltip = {
			text = pantheon_temple_building_tt
			NOT = { has_variable = pantheon_temple_building }
		}
		custom_tooltip = {
			text = pantheon_temple_dormant_tt
			pantheon_temple_active_trigger = yes
		}
		custom_tooltip = {
			text = pantheon_temple_cost_upgrade_tt
			pantheon_can_afford_trigger = { COST = pantheon_temple_next_cost }
		}
	}
	effect = {
		if = { limit = { has_variable = pantheon_temple_confirm } remove_variable = pantheon_temple_confirm }
		pantheon_temple_upgrade_effect = yes
		pantheon_ui_rebuild_effect = yes
	}
}

# The three tier-5 miracle buttons: arm on the first click, apply on the second (One Trait's confirm form,
# one_trait_ui_gov_feudal_click_effect, OneTraitToRuleThemAll/common/scripted_effects/
# one_trait_panel_effects.txt); any other click overwrites pantheon_temple_confirm, disarming it. The
# matching _armed scripted GUI is read-only, for the window's label swap only.
pantheon_ui_miracle_conversion = {
	scope = character
	is_shown = {
		has_variable = pantheon_temple_tier
		var:pantheon_temple_tier = 5

}
