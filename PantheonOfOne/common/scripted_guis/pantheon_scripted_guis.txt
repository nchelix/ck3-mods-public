# Open (rebuilding the lists first) or close the window. 2.3 B4: closing KEEPS a pending send (pantheon_send_kind /
# _recipient / _token); reopening rebuilds (which drops it if its source is gone) and the Children banner shows again.
pantheon_ui_toggle = {
	scope = character
	is_valid = { has_trait = pantheon_divine }
	effect = {
		if = {
			limit = { has_variable = pantheon_ui_open }
			remove_variable = pantheon_ui_open
			# A long-open window can't leave a stale pinned reference behind it (M5)
			clear_variable_list = pantheon_ui_followers
			clear_variable_list = pantheon_ui_blessed
			clear_variable_list = pantheon_ui_demigods
			clear_variable_list = pantheon_ui_houses
			clear_variable_list = pantheon_ui_prayers
			# Batch 2r1 (repair 4): closing the window must not leave a miracle armed.
			pantheon_temple_clear_confirm_effect = yes
		}
		else = {
			if = { limit = { NOT = { has_variable = pantheon_ui_tab } } set_variable = { name = pantheon_ui_tab value = 1 } }
			pantheon_ui_rebuild_effect = yes
			# Fix round 2 (Codex): reopening on the Prayers tab picks the first non-empty sub-tab again
			if = {
				limit = { var:pantheon_ui_tab = 5 }
				pantheon_ui_pick_ptab_effect = yes
			}
			# Batch 2r1 (repair 4): re-opening must not resume with a stale armed miracle either.
			pantheon_temple_clear_confirm_effect = yes
			set_variable = pantheon_ui_open
		}
	}
}

	}
}

# Send a child (Host need 1 / Spouse need 8). B4: starts sending mode -- stores the pending send (kind 1 Host / 2
# Spouse, recipient = the suppliant, token = this prayer's), opens the window if closed and shows the Children tab.
# The prayer stays open; a demigod's Send button there answers it. Valid when a free demigod exists and the Worship is
# there, as before.
pantheon_ui_prayer_send = {
	scope = character
	saved_scopes = { pantheon_suppliant }
	is_valid = {
		custom_tooltip = {
			text = pantheon_prayer_no_longer_tt
			pantheon_prayer_is_eligible_trigger = yes
		}
		custom_tooltip = {
			text = pantheon_has_free_demigod_tt
			any_child = { pantheon_free_demigod_trigger = { GOD = root } }
		}
		custom_tooltip = {
			text = pantheon_needs_worship_lend_tt
			OR = {
				AND = {
					scope:pantheon_suppliant = { var:pantheon_prayer_need = 1 }
					var:pantheon_worship >= pantheon_cost_lend
				}
				AND = {
					scope:pantheon_suppliant = { var:pantheon_prayer_need = 8 }
					var:pantheon_worship >= pantheon_cost_wed
				}
			}
		}
	}
	effect = {
		pantheon_prayer_send_entry_effect = yes
	}
}

# A demigod row's Send, while sending. Called with GuiScope.SetRoot( GetPlayer.MakeScope ) and the row's demigod as the
# saved scope pantheon_send_demigod. is_valid names the reason a row is greyed (never custom_description).
pantheon_ui_send_demigod = {
	scope = character
	saved_scopes = { pantheon_send_demigod }
	is_valid = {
		custom_tooltip = {
			text = pantheon_ui_send_gone_tt
			pantheon_send_source_ok_trigger = yes
		}
		custom_tooltip = {
			text = pantheon_ui_send_not_free_tt
			scope:pantheon_send_demigod = { pantheon_free_demigod_trigger = { GOD = root } }
		}
		trigger_if = {
			limit = {
				has_variable = pantheon_send_kind
				var:pantheon_send_kind = 2
			}
			custom_tooltip = {
				text = pantheon_ui_send_married_tt
				scope:pantheon_send_demigod = { is_married = no }
			}
			custom_tooltip = {
				text = pantheon_cannot_wed_tt
				var:pantheon_send_recipient ?= { pantheon_could_wed_demigod_trigger = { DEMIGOD = scope:pantheon_send_demigod } }
			}
			custom_tooltip = {
				text = pantheon_needs_worship_wed_tt
				var:pantheon_worship >= pantheon_cost_wed
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = pantheon_needs_worship_lend_tt
				var:pantheon_worship >= pantheon_cost_lend
			}
		}
	}
	effect = {
		pantheon_send_demigod_effect = yes
		pantheon_ui_rebuild_effect = yes
	}
}
