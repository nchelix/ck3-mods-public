# Pantheon of One 2.7: follower alerts. Rank 0..5 = Unlanded, Baron, Count, Duke, King, Emperor (anything above empire counts as 5).
# Follower scope.
pantheon_alert_rank = {
	value = 0
	if = {
		limit = { is_landed = no }
		add = 0
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_empire }
		add = 5
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		add = 4
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_duchy }
		add = 3
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_county }
		add = 2
	}
	else = { add = 1 }
}

# 2.7 repair round 3: how many followers qualify for the top-alert diamond, the same conditions as the matching list type in
# common/important_actions. Player scope (root = the god); counted by walking the god's own lists.
pantheon_alert_count_lapsing = {
	value = 0
	if = {
		limit = { has_variable_list = pantheon_glance_marked }
		every_in_list = {
			variable = pantheon_glance_marked
			limit = {
				is_alive = yes
				faith = root.faith
				pantheon_alert_lapsing_trigger = yes
				pantheon_alert_mode_lapse = 2
			}
			add = 1
		}
	}
}

pantheon_alert_count_left = {
	value = 0
	if = {
		limit = { has_variable_list = pantheon_lost }
		every_in_list = {
			variable = pantheon_lost
			limit = {
				is_alive = yes
				NOT = { faith = root.faith }
				has_variable = pantheon_alert_recent_left
				NOT = { has_variable = pantheon_alert_reform }
				pantheon_alert_mode_left = 2
			}
			add = 1
		}
	}
}
