# Pantheon of One: every number in one place. Boon costs were tripled on Earned in 1.3 (too generous in game);
# the Sandbox rule pays 5x, keeps the old Sandbox costs (1/12 of Earned's), and doubles durations.

pantheon_worship_per_county = {
	value = 1
	if = {
		limit = { has_game_rule = pantheon_economy_sandbox }
		multiply = 5
	}
}

pantheon_worship_per_ruler = {
	value = 5
	if = {
		limit = { has_game_rule = pantheon_economy_sandbox }
		multiply = 5
	}
}

# The god's income each quarter, in the god's scope. 1.3: Devotion weighting replaces the old piety-level
# doubling (spec). Wavering rulers (below 25 Devotion) give nothing; a ruler with no Devotion yet counts as
# Faithful; Devout rulers (75+) count twice.
pantheon_quarterly_worship = {
	value = faith.num_county_followers
	multiply = pantheon_worship_per_county
	every_ruler = {
		limit = { faith = root.faith }
		if = {
			limit = { NOT = { pantheon_wavering_trigger = yes } }
			add = pantheon_worship_per_ruler
			if = {
				limit = { pantheon_devout_trigger = yes }
				add = pantheon_worship_per_ruler
			}
		}
	}
}

# ---- 1.3: Prayers

# How often pantheon.0050 sends a prayer, per the pantheon_prayer_frequency game rule.
pantheon_prayer_interval_days = {
	value = 182
	if = {
		limit = { has_game_rule = pantheon_prayers_rare }
		value = 365
	}
	else_if = {
		limit = { has_game_rule = pantheon_prayers_frequent }
		value = 91
	}
}

# The Worship cost of granting whichever need pantheon_prayer_need (1-6, pantheon.0060) names, reusing the
# existing boon costs so the Sandbox rule scales this too.
pantheon_prayer_cost = {
	value = pantheon_cost_trait
	if = {
		limit = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 1 } }
		value = pantheon_cost_host
	}
	else_if = {
		limit = {
			scope:pantheon_suppliant = {
				OR = {
					var:pantheon_prayer_need = 2
					var:pantheon_prayer_need = 3
					var:pantheon_prayer_need = 7
					var:pantheon_prayer_need = 8
				}
			}
		}
		value = pantheon_cost_life
	}
	else_if = {
		limit = {
			scope:pantheon_suppliant = {
				OR = {
					var:pantheon_prayer_need = 4
					var:pantheon_prayer_need = 5
				}
			}
		}
		value = pantheon_cost_realm
	}
}

# Send a sign: a quarter of the prayer's cost, rounded (spec).
pantheon_prayer_sign_cost = {
	value = pantheon_prayer_cost
	divide = 4
	round = yes
}

# 2.0: how many of the god's blessings a character carries (the Pantheon window's "Blessed" mark)

	value = 0
	if = { limit = { has_variable = pantheon_boon_trait } add = 1 }
	if = { limit = { has_variable = pantheon_boon_harvest } add = 1 }
	if = { limit = { has_variable = pantheon_boon_walls } add = 1 }
	if = { limit = { has_variable = pantheon_boon_warding } add = 1 }
	if = { limit = { has_variable = pantheon_boon_vigor } add = 1 }
	if = { limit = { has_variable = pantheon_boon_fertility } add = 1 }
	if = { limit = { has_variable = pantheon_boon_lineage } add = 1 }
}

# 2.0: on a house head, how many living members carry the god's blood (the Children tab)
pantheon_ui_godsblood_count = {
	value = 0
	if = {
		limit = { exists = house }
		house = {
			every_house_member = { limit = { is_alive = yes has_variable = pantheon_godsblood } add = 1 }
		}
	}
}

# 2.0: 1 when this character's house has had its divine blood revealed
pantheon_ui_house_revealed = {
	value = 0
	if = {
		limit = { house ?= { has_variable = pantheon_divine_revealed } }
		add = 1
	}
}
