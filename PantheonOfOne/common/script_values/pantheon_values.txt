# The god's income each quarter, in the god's scope. 1.3: Devotion weighting replaces the old piety-level
# doubling (spec). Wavering rulers (below 25 Devotion) give nothing; a ruler with no Devotion yet counts as
# Faithful; Devout rulers (75+) count twice.
pantheon_quarterly_worship = {
	value = faith.num_county_followers
	multiply = pantheon_worship_per_county
	every_ruler = {
		limit = { faith = root.faith }
		if = {
			limit = { NOT = { pantheon_wavering_trigger = yes } }
			add = pantheon_worship_per_ruler
			if = {
				limit = { pantheon_devout_trigger = yes }
				add = pantheon_worship_per_ruler
			}
		}
	}
}

# 2.9: what answering a quiet prayer costs: half the popup price, rounded. Worked out in the GOD's scope with
# scope:pantheon_suppliant saved (pantheon_prayer_cost reads the need from it, and already carries the Sandbox and temple
# factors), then stored on the suppliant as pantheon_quiet_price so a later temple discount does not change it.

	value = pantheon_prayer_cost
	divide = 2
	round = yes
}

# The Worship cost of granting whichever need pantheon_prayer_need (1-6, pantheon.0060) names, reusing the
# existing boon costs so the Sandbox rule scales this too.
pantheon_prayer_cost = {
	value = pantheon_cost_trait
	if = {
		limit = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 1 } }
		value = pantheon_cost_host
	}
	else_if = {
		limit = {
			scope:pantheon_suppliant = {
				OR = {
					var:pantheon_prayer_need = 2
					var:pantheon_prayer_need = 3
					var:pantheon_prayer_need = 7
					var:pantheon_prayer_need = 8
					# 2.9: Cleanse, Whole, Grief, Feud
					var:pantheon_prayer_need = 11
					var:pantheon_prayer_need = 12
					var:pantheon_prayer_need = 13
					var:pantheon_prayer_need = 14
				}
			}
		}
		value = pantheon_cost_life
	}
	else_if = {
		limit = {
			scope:pantheon_suppliant = {
				OR = {
					var:pantheon_prayer_need = 4
					var:pantheon_prayer_need = 5
				}
			}
		}
		value = pantheon_cost_realm
	}
	# 2.4: Freedom (need 9)
	else_if = {
		limit = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 9 } }
		value = pantheon_cost_freedom
	}
	# 2.5: a relative's prayer for a jailed loved one (need 10) costs the same as Freedom
	else_if = {
		limit = { scope:pantheon_suppliant = { var:pantheon_prayer_need = 10 } }
		value = pantheon_cost_freedom
	}
}

# 2.4 Dissolve the union. In the DEMIGOD's scope (a wed demigod of the god). 3 = the spouse is not of the god's faith;
# 2 = the spouse is landless now and (was landed when wed, or the marriage has no record), or is Wavering; else 1, by
# choice. The god is the demigod's divine parent. Every spouse read is ?=-guarded.
pantheon_dissolve_reason = {
	value = 1
	if = {
		limit = {
			primary_spouse ?= { pantheon_wavering_trigger = yes }
		}
		value = 2
	}
	if = {
		limit = {
			primary_spouse ?= { is_landed = no }
			OR = {
				NOT = { has_variable = pantheon_wed_spouse_landed }
				var:pantheon_wed_spouse_landed ?= 1
			}
		}
		value = 2
	}
	if = {
		limit = {
			primary_spouse ?= { is_alive = yes }
			any_parent = {
				has_trait = pantheon_divine
				has_variable = pantheon_enthroned
				NOT = { faith = prev.primary_spouse.faith }
			}
		}
		value = 3
	}
}

		divide = 12
	}
	floor = yes

}
