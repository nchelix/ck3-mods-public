# Pantheon of One: every number in one place. The Sandbox rule pays 5x, charges 1/4, and doubles durations.

pantheon_worship_per_county = {
	value = 1
	if = {
		limit = { has_game_rule = pantheon_economy_sandbox }
		multiply = 5
	}
}

pantheon_worship_per_ruler = {
	value = 5
	if = {
		limit = { has_game_rule = pantheon_economy_sandbox }
		multiply = 5
	}
}

# The god's income each quarter, in the god's scope. Devout rulers (piety level 3+) count twice.
pantheon_quarterly_worship = {
	value = faith.num_county_followers
	multiply = pantheon_worship_per_county
	every_ruler = {
		limit = { faith = root.faith }
		add = pantheon_worship_per_ruler
		if = {
			limit = { piety_level >= 3 }
			add = pantheon_worship_per_ruler
		}
	}
}
