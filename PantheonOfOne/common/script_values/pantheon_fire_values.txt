# The price of the size picked in the interaction (var:pantheon_fire_size 1 to 4). God scope.
pantheon_fire_cost_selected = {
	value = 0
	if = { limit = { var:pantheon_fire_size ?= 1 } add = pantheon_cost_fire_county }
	else_if = { limit = { var:pantheon_fire_size ?= 2 } add = pantheon_cost_fire_duchy }
	else_if = { limit = { var:pantheon_fire_size ?= 3 } add = pantheon_cost_fire_kingdom }
	else_if = { limit = { var:pantheon_fire_size ?= 4 } add = pantheon_cost_fire_empire }
}

# The calendar month as one number (year x 12 + month): the aftermath ticks once per month, keyed to this (batch B review,
# finding 1). current_year and current_month are date triggers that read as values (docs/reference/ck3-1.20.1.1-triggers.log:556
# and ck3-1.20.1.1-triggers.log:543); vanilla stores both with value = (events/war_events/combat_events.txt:2289-2293) and nests
# current_year in a value block (common/script_values/08_bp3_values.txt:69-73, inspection_levy_reward_value).
pantheon_fire_period_value = {
	value = current_year
	multiply = 12
	add = { value = current_month }
}

# What a restoring lord pays for one stage of healing, in gold. Character scope.
pantheon_fire_restore_cost = {
	value = monthly_character_income
	multiply = 36
	min = 100
}

# Task 8: which struck county the Restore decision works on (ordered_held_title takes the highest). County title scope.
# The lowest stage first; among equal stages, the holder's capital county (the capital bonus stays below one stage's step).
# Capital test form: vanilla common/council_tasks/00_kurultai_tasks.txt (scope:county = scope:county.holder.capital_county).
pantheon_fire_restore_priority = {
	value = 40
	subtract = {
		value = var:pantheon_fire_stage
		multiply = 10
	}
	if = {
		limit = { this = holder.capital_county }
		add = 5
	}
}
