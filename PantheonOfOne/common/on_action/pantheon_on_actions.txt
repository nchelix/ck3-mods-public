# Pantheon of One hooks. Appended to vanilla's on_actions, never replacing them.

# Root = the prisoner. No cell can hold a god (proven in the prototype).
on_imprison = {
	on_actions = { pantheon_on_imprison pantheon_on_imprison_grievance pantheon_on_imprison_freedom }
}

pantheon_on_imprison = {
	trigger = { has_trait = pantheon_divine }
	effect = {
		release_from_prison = yes
		trigger_event = pantheon.0001
	}
}

# 2.4 Freedom. Root = the prisoner; scope:imprisoner = the jailer. A jailed follower prays for their freedom PROMPTLY:
# each enthroned player god gets the hidden event pantheon.0115, so the pass runs with the GOD as root (delivery and
# aspiration creation require it). No imprisoner (some imprisonments fire this without one), no event.
pantheon_on_imprison_freedom = {
	trigger = {
		exists = scope:imprisoner
		is_alive = yes
	}
	effect = {
		every_player = {
			limit = {
				has_trait = pantheon_divine
				has_variable = pantheon_enthroned
			}
			trigger_event = pantheon.0115
		}
	}
}

# 2.4 Raiders on holy ground. Vanilla's on_raid_action_start: root = the raid army, scope:raider = its owner, scope:barony and
# scope:county = the place. Engine-provided scopes are read only after `exists`. For each enthroned player god:
#  - on the god's OWN land (the county's holder is the god):
#      1. a raider at war with the god is skipped (Divine Retribution covers that);
#      2. a raider Marked by this god has the army annihilated at no cost, with a toast and no event;
#      3. otherwise the raider carries pantheon_raid_alerted_by (30 days, refreshed on every barony raid start) -- one
#         continuous raid alerts once (R17) -- and, if it was ABSENT before, the god is asked (pantheon.0112) and the raider
#         becomes a Judgement target (pantheon_raided_god, 5 years);
#  - on a FOLLOWER's barony, raided by anyone the god does not protect (a fellow believer too, 2.4 repair 1): a light grievance (kind 6).
on_raid_action_start = {
	on_actions = { pantheon_on_raid_start }
}

pantheon_on_raid_start = {
	trigger = {
		exists = scope:raider
		exists = scope:barony
		exists = scope:county
	}
	effect = {
		if = {
			limit = { exists = scope:pantheon_raid_army }
			clear_saved_scope = pantheon_raid_army
		}
		save_scope_as = pantheon_raid_army
		if = {
			limit = { exists = scope:pantheon_raider }
			clear_saved_scope = pantheon_raider
		}
		scope:raider = { save_scope_as = pantheon_raider }
		every_player = {
			limit = {
				has_trait = pantheon_divine
				has_variable = pantheon_enthroned
			}
			if = {
				limit = { exists = scope:pantheon_god }
				clear_saved_scope = pantheon_god
			}
			save_scope_as = pantheon_god
			if = {
				limit = {
					scope:county.holder ?= scope:pantheon_god
					NOT = { scope:raider = scope:pantheon_god }
					NOT = { scope:raider = { is_at_war_with = scope:pantheon_god } }
				}
				if = {
					limit = { scope:raider = { var:pantheon_marked_by ?= scope:pantheon_god } }
					scope:pantheon_raid_army ?= { pantheon_annihilate_army_effect = yes }
					send_interface_toast = {
						type = event_toast_effect_good
						title = pantheon_raid_annihilated_t
						left_icon = scope:raider
					}
				}
				else = {
					if = {
						limit = { scope:raider = { NOT = { var:pantheon_raid_alerted_by ?= scope:pantheon_god } } }
						scope:raider = { set_variable = { name = pantheon_raided_god value = scope:pantheon_god years = 5 } }
						trigger_event = pantheon.0112
					}
					scope:raider = { set_variable = { name = pantheon_raid_alerted_by value = scope:pantheon_god days = 30 } }
				}
			}
			else_if = {
				limit = {
					# 2.4 repair 1 (R8): any raider the god does not protect (a fellow believer too), but never the holder's own raid
					scope:raider = { NOT = { pantheon_wrath_protected_trigger = { GOD = scope:pantheon_god } } }
					NOT = { scope:barony.holder ?= { this = scope:raider } }
					scope:barony.holder ?= {
						faith = scope:pantheon_god.faith
						NOT = { this = scope:pantheon_god }
						pantheon_grievance_eligible_trigger = { GOD = scope:pantheon_god }
					}
				}
				pantheon_give_raid_grievance_effect = { WHO = scope:barony.holder OFFENDER = scope:raider }
			}
		}
	}
}

			limit = {
				OR = {
					scope:mother ?= { has_character_flag = pantheon_lineage NOT = { has_character_flag = pantheon_miracle_birth } }
					scope:real_father ?= { has_character_flag = pantheon_lineage NOT = { has_character_flag = pantheon_miracle_birth } }
				}
			}
			add_trait = beauty_good_3
			add_trait = intellect_good_3
			if = {
				limit = { scope:mother ?= { has_character_flag = pantheon_lineage NOT = { has_character_flag = pantheon_miracle_birth } } }

					remove_character_flag = pantheon_lineage
					remove_character_modifier = pantheon_lineage_marker
					remove_variable = pantheon_boon_lineage
				}
			}
			if = {
				limit = { scope:real_father ?= { has_character_flag = pantheon_lineage NOT = { has_character_flag = pantheon_miracle_birth } } }
				scope:real_father = {
					remove_character_flag = pantheon_lineage
					remove_character_modifier = pantheon_lineage_marker
					remove_variable = pantheon_boon_lineage
				}
			}
		}
	}
}

# Root = the mother, when a pregnancy is revealed. A god's child: sanctify the union once vanilla has reacted.
on_pregnancy_mother = {
	on_actions = { pantheon_on_pregnancy_mother }
}

pantheon_on_pregnancy_mother = {
	trigger = {
		OR = {
			scope:mother ?= { has_trait = pantheon_divine }
			scope:real_father ?= { has_trait = pantheon_divine }
		}
	}
	effect = {
		# 1.5: a god who forswore mortal offspring has none. -1 fertility only lowers the odds (the other
		# parent's fertility still counts; children were still born in the 1.5 test), so end the pregnancy.
		# Divine Conception already refuses a forsworn god.
		if = {
			limit = {
				OR = {
					scope:mother ?= { has_trait = pantheon_divine has_character_modifier = pantheon_forsworn }
					scope:real_father ?= { has_trait = pantheon_divine has_character_modifier = pantheon_forsworn }
				}
			}
			scope:mother = { end_pregnancy = yes }
		}
		else = {
			scope:mother = {
				set_variable = {
					name = pantheon_child_father
					value = scope:real_father
				}
				# 2.2.2 (R1/R2): pregnancy.2001/.2002 have just run (events precede on_actions): clear the
				# bastard flags they set at once, and again a day later (0040)
				pantheon_clear_bastard_pregnancy_effect = yes
				# 2.3.2: pregnancy.0001 (the reveal) may have set this flag; birth.3031 would then make the
				# newborn Sickly 10-30 days after birth, after our clean-up. A demigod is never born sickly.
				if = {
					limit = { has_character_flag = birth_child_will_become_sickly }
					remove_character_flag = birth_child_will_become_sickly
				}
				# 2.3.2: the engine does not add the Pregnant trait; vanilla's reveal events do (unmarried_mother_ and
				# married_mother_pregnancy_effect), and for a god's child none of them fits (#14 override). Add it here.
				if = {
					limit = { NOT = { has_trait = pregnant } }
					add_trait = pregnant
				}
				trigger_event = { id = pantheon.0040 days = 1 }
			}
		}
	}
}

# 1.4 Divine Retribution. Scopes proven in the Phase 3 prototype: scope:attacker, scope:defender, scope:war.
# The god is saved, not passed as `this`: inside the trigger's nested blocks `this` would change.
on_war_started = {
	on_actions = { pantheon_on_war_started }
}

pantheon_on_war_started = {
	effect = {
		every_player = {
			limit = {
				has_trait = pantheon_divine
				has_variable = pantheon_enthroned
			}
			if = {
				limit = { exists = scope:pantheon_god }
				clear_saved_scope = pantheon_god
			}
			save_scope_as = pantheon_god
			if = {
				limit = {
					exists = scope:attacker
					exists = scope:defender
					pantheon_retribution_war_trigger = { GOD = scope:pantheon_god }
				}
				# 2.3 (R10): a record on the defender and the god's list, not the two single slots; 0080 carries it
				pantheon_retri_open_effect = yes
			}
		}
	}
}

# 2.3 (R17): a sworn oath whose offender dies by ANY means is released (toast once, no penalty). Deliberately NOT gated on
# scope:killer like the murder hook below: natural deaths, executions and wars all reach it. A wrath (Strike) that
# fulfilled the oath did so BEFORE the death, so the oath is already gone and this finds nothing -- no double notice.
# Root = the dying character.
pantheon_on_death_oath = {
	effect = {
		save_temporary_scope_as = pantheon_dying_one
		every_player = {
			limit = {
				has_trait = pantheon_divine
				has_variable = pantheon_enthroned
				has_variable = pantheon_great_prayer_active
				var:pantheon_great_prayer_offender ?= scope:pantheon_dying_one
			}
			pantheon_oath_release_effect = yes
		}
	}
}

pantheon_on_death = {
	trigger = { exists = scope:killer }
	effect = {
		save_scope_as = pantheon_dying
		every_player = {
			limit = {
				has_trait = pantheon_divine
				has_variable = pantheon_enthroned
			}
			trigger_event = { id = pantheon.0090 days = 1 }
		}
	}
}

# Imprisoned. Root = the prisoner; scope:imprisoner = the jailer. The prisoner, and their devout close kin,
# hold it against the jailer. (1.0 already hooks on_imprison with pantheon_on_imprison, the god's release:
# this joins that list, see the top of this file, under its own name.)
pantheon_on_imprison_grievance = {
	# 2.2.2 (#5): some imprisonments fire on_imprison with no imprisoner (error.log: "Undefined event target
	# 'imprisoner'", then province-scope errors). No imprisoner, no offender: no grievance is stored.
	# (There is no is_character trigger in 1.20.1: exists first, then the character-only is_alive on both.)
	trigger = {
		exists = scope:imprisoner
		is_alive = yes
		scope:imprisoner = { is_alive = yes }
	}
	effect = {
		save_scope_as = pantheon_prisoner
		every_player = {
			limit = {
				has_trait = pantheon_divine
				has_variable = pantheon_enthroned
			}
			if = {
				limit = { exists = scope:pantheon_god }
				clear_saved_scope = pantheon_god
			}
			save_scope_as = pantheon_god
			if = {
				limit = {
					NOT = { scope:imprisoner = { faith = scope:pantheon_god.faith } }
				}
				if = {
					limit = { scope:pantheon_prisoner = { pantheon_grievance_eligible_trigger = { GOD = scope:pantheon_god } } }
					pantheon_give_grievance_effect = { WHO = scope:pantheon_prisoner OFFENDER = scope:imprisoner KIND = 4 }
				}
				scope:pantheon_prisoner = {
					every_close_family_member = {
						limit = { pantheon_grievance_eligible_trigger = { GOD = scope:pantheon_god } }
						pantheon_give_grievance_effect = { WHO = this OFFENDER = scope:imprisoner KIND = 4 }
					}
					every_spouse = {
						limit = { pantheon_grievance_eligible_trigger = { GOD = scope:pantheon_god } }
						pantheon_give_grievance_effect = { WHO = this OFFENDER = scope:imprisoner KIND = 4 }
					}
				}
			}
		}
	}
}

# 2.4.1 (#26): the faith follows its god. Vanilla's on_faith_created: root is documented as the NEW FAITH (not a character),
# scope:founder is the acting character (may be absent), scope:old_faith the faith they came from. For a reform by a sitting Head
# of Faith, 1.20 passes the OLD faith as root (see the else branch below). Fires only for an enthroned Pantheon god, so Proclaim
# Your Divinity before the throne, a faith made by anyone else, and a game without a god are all untouched. Root is not the god
# here, so the hook only saves the two faiths and hands off to pantheon.0120, a hidden event whose root IS the god.
on_faith_created = {
	on_actions = { pantheon_on_faith_created }
}

pantheon_on_faith_created = {
	trigger = {
		exists = scope:old_faith
		scope:founder ?= {
			has_trait = pantheon_divine
			has_variable = pantheon_enthroned
		}
	}
	effect = {
		if = {
			limit = { exists = scope:pantheon_new_faith }
			clear_saved_scope = pantheon_new_faith
		}
		if = {
			limit = { NOT = { this = scope:old_faith } }
			save_scope_as = pantheon_new_faith
		}
		else = {
			# CK3 1.20 reports a reform by a sitting Head of Faith with the OLD faith as root (proven in game 2026-10-23):
			# the engine builds the new faith but cannot move its head into it, and root is the founder's faith. The new
			# faith is then the one in the same religion that the god founded and that nobody follows yet. CK3 offers no
			# way to tell which of two such faiths is newer (an earlier one the god founded may have emptied out), so the
			# hook acts only when exactly one matches; otherwise it does nothing, as vanilla would (Codex review 2026-10-24).
			scope:old_faith.religion = {
				if = {
					limit = {
						# exactly one: "count = N" means at least N
						any_faith = {
							pantheon_reform_new_faith_candidate_trigger = { OLD = scope:old_faith FOUNDER = scope:founder }
						}
						NOT = {
							any_faith = {
								count = 2
								pantheon_reform_new_faith_candidate_trigger = { OLD = scope:old_faith FOUNDER = scope:founder }
							}
						}
					}
					random_faith = {
						limit = { pantheon_reform_new_faith_candidate_trigger = { OLD = scope:old_faith FOUNDER = scope:founder } }
						save_scope_as = pantheon_new_faith
					}
				}
			}
		}
		if = {
			limit = { exists = scope:pantheon_new_faith }
			if = {
				limit = { exists = scope:pantheon_old_faith }
				clear_saved_scope = pantheon_old_faith
			}
			scope:old_faith = { save_scope_as = pantheon_old_faith }
			scope:founder = { trigger_event = pantheon.0120 }
		}
	}
}
