# Pantheon of One 2.9: the steadying act, "A Sign of Steadfastness". A stabiliser for a faith in trouble, sent from the Great Temple
# at one of three reaches (the slipping, one de jure kingdom, the whole faith): everyone it reaches is healed 20 Devotion (capped at
# 100), sheltered from blame's Devotion cost for a year (pantheon_sheltered, read at the top of pantheon_blame_effect), and has any
# open quiet prayer answered as a sign (pantheon_quiet_sign_effect: +10 Devotion, closed; the blessing prayed for is NOT given). It
# is paid in Worship and followed by a five-year rest (pantheon_steady_cooldown).
#
#   On the god (armed, cleared by pantheon_steady_clear_effect):
#       pantheon_steady_reach     1 the slipping, 2 a kingdom, 3 the whole faith (presence = armed)
#       pantheon_steady_kingdom   the picked de jure kingdom (a title; reach 2 only)
#       pantheon_steady_count     followers in reach          pantheon_steady_all   every follower (the share's denominator)
#       pantheon_steady_price     the price in Worship, worked out from the two counts
#   On the god (kept): pantheon_steady_cooldown (years = 5), pantheon_steady_last_count (for the toast after the vars are cleared).
#
# Root/this is the god in every effect here. Arming (the three arm GUIs) only gathers and prices; SEND gathers and prices AGAIN, checks
# the reach is still open, the pool is not empty and Worship covers the recomputed price, and only then charges. Every pool walk runs
# behind pantheon_live_character_trigger and reads nothing of a purged character.

# Add the god's followers to the temporary list $LIST$, each once: the god's faith's rulers, the god's own courtiers of the faith, and
# pantheon_known (the union the Followers tab and the drift walk). Never the god, never a demigod. The live check leads every limit.
pantheon_steady_pool_effect = {
	every_ruler = {
		limit = {
			pantheon_live_character_trigger = yes
			faith = root.faith
			NOT = { this = root }
			NOT = { has_trait = pantheon_demigod }
			NOT = { is_in_list = $LIST$ }
		}
		add_to_temporary_list = $LIST$
	}
	every_courtier = {
		limit = {
			pantheon_live_character_trigger = yes
			faith = root.faith
			NOT = { this = root }
			NOT = { has_trait = pantheon_demigod }
			NOT = { is_in_list = $LIST$ }
		}
		add_to_temporary_list = $LIST$
	}
	every_in_list = {
		variable = pantheon_known
		limit = {
			pantheon_live_character_trigger = yes
			faith = root.faith
			NOT = { this = root }
			NOT = { has_trait = pantheon_demigod }
			NOT = { is_in_list = $LIST$ }
		}
		add_to_temporary_list = $LIST$
	}
}

# Follower scope: save their region (the de jure kingdom of their capital county if they hold land, else of the county they stand in)
# as scope:pantheon_steady_region_one, or leave it unsaved when they have none. The previous follower's region is forgotten first, so
# a follower with no kingdom never inherits it. The same rule as pantheon_steady_in_region_trigger.
pantheon_steady_region_effect = {
	if = { limit = { exists = scope:pantheon_steady_region_one } clear_saved_scope = pantheon_steady_region_one }
	if = {
		limit = { is_landed = yes exists = capital_county.kingdom }
		capital_county.kingdom = { save_scope_as = pantheon_steady_region_one }
	}
	else_if = {
		limit = { is_landed = no exists = location.county.kingdom }
		location.county.kingdom = { save_scope_as = pantheon_steady_region_one }
	}
}

# Gather the reach $REACH$ (1 the slipping, 2 the kingdom in pantheon_steady_kingdom, 3 the whole faith; a literal or var:pantheon_steady_reach).
# Sets the reach, pantheon_steady_count (those reached) and pantheon_steady_all (every follower), and leaves the temporary list
# pantheon_steady_pool with those reached for the caller. Nothing is changed on any follower.
pantheon_steady_gather_effect = {
	set_variable = { name = pantheon_steady_reach value = $REACH$ }
	set_variable = { name = pantheon_steady_count value = 0 }
	set_variable = { name = pantheon_steady_all value = 0 }
	# a kingdom left over from an earlier arming never rides into another reach
	if = {
		limit = { NOT = { var:pantheon_steady_reach = 2 } has_variable = pantheon_steady_kingdom }
		remove_variable = pantheon_steady_kingdom
	}
	pantheon_steady_pool_effect = { LIST = pantheon_steady_everyone }
	every_in_list = {
		list = pantheon_steady_everyone
		limit = { pantheon_live_character_trigger = yes }
		root = { change_variable = { name = pantheon_steady_all add = 1 } }
	}
	if = {
		limit = { var:pantheon_steady_reach = 1 }
		every_in_list = {
			list = pantheon_steady_everyone
			limit = { pantheon_live_character_trigger = yes pantheon_steady_slipping_trigger = yes }
			add_to_temporary_list = pantheon_steady_pool
			root = { change_variable = { name = pantheon_steady_count add = 1 } }
		}
	}
	else_if = {
		limit = { var:pantheon_steady_reach = 2 }
		every_in_list = {
			list = pantheon_steady_everyone
			limit = { pantheon_live_character_trigger = yes pantheon_steady_in_region_trigger = yes }
			add_to_temporary_list = pantheon_steady_pool
			root = { change_variable = { name = pantheon_steady_count add = 1 } }
		}
	}
	else = {
		every_in_list = {
			list = pantheon_steady_everyone
			limit = { pantheon_live_character_trigger = yes }
			add_to_temporary_list = pantheon_steady_pool
			root = { change_variable = { name = pantheon_steady_count add = 1 } }
		}
	}
}

# Arm the reach $REACH$ (1, 2 or 3; the three arm scripted GUIs each pass their own): gather, price, and show the Temple tab, where
# the armed panel is. Charges nothing. An empty reach is never armed: a toast says so. Called by pantheon_ui_steady_arm_*.
pantheon_steady_arm_effect = {
	pantheon_steady_gather_effect = { REACH = $REACH$ }
	if = {
		limit = { has_variable = pantheon_steady_count var:pantheon_steady_count > 0 }
		set_variable = { name = pantheon_steady_price value = pantheon_steady_price_value }
		# a kingdom armed from the Quiet tab opens the Temple tab, where the armed panel lives
		set_variable = { name = pantheon_ui_tab value = 4 }
		pantheon_ui_rebuild_effect = yes
	}
	else = {
		pantheon_steady_clear_effect = yes
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = pantheon_steady_nobody_t
			left_icon = root
			custom_tooltip = pantheon_steady_nobody_tt
		}
	}
}

# Forget everything that is armed: Cancel, a window close, the Temple tab's own button, and the end of Send. Keeps the rest
# (pantheon_steady_cooldown) and the last count the toast reads.
pantheon_steady_clear_effect = {
	if = { limit = { has_variable = pantheon_steady_reach } remove_variable = pantheon_steady_reach }
	if = { limit = { has_variable = pantheon_steady_kingdom } remove_variable = pantheon_steady_kingdom }
	if = { limit = { has_variable = pantheon_steady_count } remove_variable = pantheon_steady_count }
	if = { limit = { has_variable = pantheon_steady_all } remove_variable = pantheon_steady_all }
	if = { limit = { has_variable = pantheon_steady_price } remove_variable = pantheon_steady_price }
}

# Send the armed reach. In this order: gather again and price again (what was armed may be stale); stop with a toast, and NO charge,
# if the reach is no longer open (the tier, a dormant temple, the rest), the pool is empty, or Worship is below the RECOMPUTED price;
# otherwise charge that price, then for each person heal 20 (the clamped change, so capped at 100), shelter for a year and answer
# an open quiet prayer as a sign; start the five-year rest; toast the count; clear what was armed and rebuild.
pantheon_steady_send_effect = {
	if = {
		limit = { has_variable = pantheon_steady_reach }
		pantheon_steady_gather_effect = { REACH = var:pantheon_steady_reach }
		set_variable = { name = pantheon_steady_price value = pantheon_steady_price_value }
		if = {
			limit = {
				pantheon_steady_reach_open_trigger = { TIER = pantheon_steady_tier_needed }
				has_variable = pantheon_steady_count
				var:pantheon_steady_count > 0
				has_variable = pantheon_steady_price
				pantheon_can_afford_trigger = { COST = var:pantheon_steady_price }
			}
			change_variable = { name = pantheon_worship subtract = var:pantheon_steady_price }
			set_variable = { name = pantheon_steady_last_count value = var:pantheon_steady_count }
			if = { limit = { exists = scope:pantheon_target } clear_saved_scope = pantheon_target }
			if = { limit = { exists = scope:pantheon_suppliant } clear_saved_scope = pantheon_suppliant }
			every_in_list = {
				list = pantheon_steady_pool
				limit = { pantheon_live_character_trigger = yes }
				save_scope_as = pantheon_target
				root = { pantheon_change_devotion_effect = { AMOUNT = 20 } }
				set_variable = { name = pantheon_sheltered value = yes years = 1 }
				if = {
					limit = { has_variable = pantheon_quiet_open }
					save_scope_as = pantheon_suppliant
					root = { pantheon_quiet_sign_effect = yes }
				}
			}
			if = { limit = { exists = scope:pantheon_target } clear_saved_scope = pantheon_target }
			if = { limit = { exists = scope:pantheon_suppliant } clear_saved_scope = pantheon_suppliant }
			set_variable = { name = pantheon_steady_cooldown value = yes years = 5 }
			send_interface_toast = {
				type = event_toast_effect_good
				title = pantheon_steady_sent_t
				left_icon = root
				custom_tooltip = pantheon_steady_sent_tt
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = pantheon_steady_stopped_t
				left_icon = root
				custom_tooltip = pantheon_steady_stopped_tt
			}
		}
		pantheon_steady_clear_effect = yes
		pantheon_ui_rebuild_effect = yes
	}
}
