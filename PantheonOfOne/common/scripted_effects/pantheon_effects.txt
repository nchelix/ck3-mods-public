# Pantheon of One scripted effects. Root is always the god.

# Proclaim Your Divinity: a god on a branched rite pushes its divergence until the engine splits it into a
# faith of its own. Proven in game (1.1 spike): one push of 100 reached 95, a second crossed 100 and the rite
# became a new faith. Stops as soon as the rite is its faith's main rite, and after 5 pushes at most.
pantheon_proclaim_effect = {
	set_local_variable = {
		name = pantheon_proclaim_tries
		value = 0
	}
	while = {
		limit = {
			NOT = { rite = faith.main_rite }
			local_var:pantheon_proclaim_tries < 5
		}
		rite = {
			change_rite_divergence = {
				value = 100
				override_tenets_first = yes
			}
		}
		change_local_variable = {
			name = pantheon_proclaim_tries
			add = 1
		}
	}
	# Some rites run out of tenets and doctrines to swap short of 100 (vanilla's Carolingian rite stopped at
	# 99.25 in game). A day later, if it is still a rite, tell the god what to do instead.
	trigger_event = { id = pantheon.0030 days = 1 }
}

# Faith scope: the faith has a spiritual head (2.4.1: shared by the throne and by a reform, pantheon_lead_new_faith_effect).
# Moved verbatim out of pantheon_take_throne_effect.
pantheon_spiritual_head_doctrine_effect = {
	if = {
		limit = { has_doctrine = doctrine_no_head }
		hidden_effect = { remove_doctrine = doctrine_no_head }
	}
	if = {
		limit = { has_doctrine = doctrine_temporal_head }
		hidden_effect = { remove_doctrine = doctrine_temporal_head }
	}
	if = {
		limit = { NOT = { has_doctrine = doctrine_spiritual_head } }
		add_doctrine = doctrine_spiritual_head
	}
}

# 2.6 (#54): God scope. A god with no other ruler of their faith starts with at least the floor in Worship (top up, never add).
pantheon_new_god_worship_floor_effect = {
	if = {
		limit = {
			NOT = { any_ruler = { faith = root.faith NOT = { this = root } } }
			var:pantheon_worship < pantheon_new_god_worship_floor
		}
		set_variable = { name = pantheon_worship value = pantheon_new_god_worship_floor }
		send_interface_toast = {
			type = event_toast_effect_good
			title = pantheon_new_god_worship_t
			left_icon = root
		}
	}
}

# 2.8 (#62): a fresh landless head-of-faith title for $FAITH$ (a faith scope or saved scope), held by the god (root). The throne
# uses it when the faith has no landless head title to take over, and a reform uses it so a god who ceded a church seat (or never
# had a title for the new faith) still sits as head of it. Its seat is a holy site county of the faith the god holds, else the
# god's capital county. Moved verbatim out of pantheon_take_throne_effect's else branch (one implementation, 2.8).
pantheon_create_hof_title_effect = {
	if = {
		limit = { exists = scope:pantheon_hof_faith }
		clear_saved_scope = pantheon_hof_faith
	}
	if = {
		limit = { exists = scope:pantheon_hof_seat }
		clear_saved_scope = pantheon_hof_seat
	}
	if = {
		limit = { exists = scope:new_title }
		clear_saved_scope = new_title
	}
	$FAITH$ = { save_scope_as = pantheon_hof_faith }
	# Vanilla's REL_HEAD_TITLE_NAME reads scope:my_faith (without it the title was named "The")
	if = {
		limit = { exists = scope:my_faith }
		clear_saved_scope = my_faith
	}
	scope:pantheon_hof_faith = { save_scope_as = my_faith }
	scope:pantheon_hof_faith = {
		random_holy_site_barony = {
			limit = { county.holder = root }
			county = { save_scope_as = pantheon_hof_seat }
		}
	}
	if = {
		limit = { NOT = { exists = scope:pantheon_hof_seat } }
		capital_county = { save_scope_as = pantheon_hof_seat }
	}
	create_dynamic_title = {
		tier = duchy
		name = REL_HEAD_TITLE_NAME
		adj = REL_HEAD_TITLE_ADJ
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:new_title = {
		set_capital_county = scope:pantheon_hof_seat
		set_landless_title = yes
		set_no_automatic_claims = yes
		set_definitive_form = yes
		set_always_follows_primary_heir = yes
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	scope:pantheon_hof_faith = { set_religious_head_title = scope:new_title }
	scope:new_title = { generate_coa = religious_title }
}

# Make the god the landless spiritual Head of their faith, holding the Holy Seat. Proven in the prototype.
pantheon_take_throne_effect = {
	# A god is no one's vassal (1.1 in-game test: a liege tried to revoke an enthroned god's county)
	if = {
		limit = { is_independent_ruler = no }
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
			add_claim_on_loss = no
		}
		becomes_independent = { change = scope:change }
		resolve_title_and_vassal_change = scope:change
	}
	faith = {
		save_scope_as = pantheon_faith
		# Vanilla's REL_HEAD_TITLE_NAME reads scope:my_faith (without it the title was named "The")
		save_scope_as = my_faith
		pantheon_spiritual_head_doctrine_effect = yes
	}
	# The Holy Seat: a holy site of the faith that the god holds, else the god's capital
	scope:pantheon_faith = {
		random_holy_site_barony = {
			limit = { county.holder = root }
			county = { save_scope_as = pantheon_seat }
		}
	}
	if = {
		limit = { NOT = { exists = scope:pantheon_seat } }
		capital_county = { save_scope_as = pantheon_seat }
	}
	if = {
		limit = {
			scope:pantheon_faith = {
				exists = religious_head_title
				religious_head_title = { is_landless_type_title = yes }
			}
		}
		# The faith already has a landless head title (a spiritual HoF): take it over.
		# A landed one (the Papal State, a temporal realm) is left as an ordinary realm and replaced below.
		scope:pantheon_faith.religious_head_title = { save_scope_as = new_title }
		if = {
			limit = { NOT = { scope:new_title.holder ?= root } }
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:new_title = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		scope:new_title = {
			# Prototype: the capital must come first, or set_landless_title fails
			set_capital_county = scope:pantheon_seat
			set_landless_title = yes
		}
		scope:pantheon_faith = { set_religious_head_title = scope:new_title }
	}
	else = {
		# 2.8: shared with the reform (pantheon_create_hof_title_effect, above)
		pantheon_create_hof_title_effect = { FAITH = scope:pantheon_faith }
	}
	scope:pantheon_seat = {
		add_county_modifier = { modifier = pantheon_holy_seat }
	}
	# The god's own guardians, to keep raiders from holy ground (1.2)
	# size must be a literal number ("Malformed token" for a script value); matches pantheon_host_stacks
	if = {
		limit = { has_game_rule = pantheon_economy_sandbox }
		create_maa_regiment = {
			type = pantheon_guardian_maa
			check_can_recruit = no
			size = 10
		}
	}
	else = {
		create_maa_regiment = {
			type = pantheon_guardian_maa
			check_can_recruit = no
			size = 5
		}
	}
	set_variable = pantheon_enthroned
	set_variable = pantheon_tick_safety_armed
	if = {
		limit = { NOT = { has_variable = pantheon_worship } }
		set_variable = {
			name = pantheon_worship
			value = 0
		}
	}
	if = {
		limit = { NOT = { has_variable = pantheon_quarter_count } }
		set_variable = {
			name = pantheon_quarter_count
			value = 0
		}
	}
	# 1.3: the faith's rulers start at 50 Devotion the moment the throne is taken (spec, Devotion table, Start).
	every_ruler = {
		limit = { faith = scope:pantheon_faith }
		save_scope_as = pantheon_target
		pantheon_ensure_devotion_effect = { START = 50 }
	}
	set_variable = { name = pantheon_quarterly_alive value = yes days = 100 }
	trigger_event = { id = pantheon.0010 days = 91 }
	set_variable = { name = pantheon_monthly_alive value = yes days = 45 }
	trigger_event = { id = pantheon.0011 days = 1 }
	set_variable = pantheon_prayer_timer
	trigger_event = { id = pantheon.0050 days = pantheon_prayer_interval_days }
	pantheon_new_god_worship_floor_effect = yes
}

# 2.7 (#60): restart a stopped tick chain. God scope; run yearly (pantheon_yearly_tick_safety). Each chain renews a heartbeat
# flag every run (monthly 45 days, quarterly 100 days); a missing flag means the chain has died. A god without pantheon_divine
# is not restarted (the chains' own trigger would stop it anyway). Both pantheon.0010 and pantheon.0011 carry a run marker,
# so a duplicate chain started while the original lives fails its trigger and dies: Worship is never paid twice (spec #60).
# Grace year: a 2.6 save has a live quarterly chain but no pantheon_quarterly_alive yet (the heartbeat arrives within 91
# days), so the quarterly restart also needs pantheon_tick_safety_armed. The first pulse after install arms and schedules a
# hidden recheck (pantheon.0012) 100 days later: that covers a live legacy chain's outstanding 91-day event (it sets
# pantheon_quarterly_alive first, so no double payout), and a dead chain is restarted by the recheck. New gods are armed by
# the take-throne effect. (Monthly: 2.6 already set its flag.)
pantheon_tick_safety_effect = {
	if = {
		limit = {
			has_trait = pantheon_divine
			NOT = { has_variable = pantheon_monthly_alive }
		}
		set_variable = { name = pantheon_monthly_alive value = yes days = 45 }
		trigger_event = { id = pantheon.0011 days = 1 }
	}
	if = {
		limit = {
			has_trait = pantheon_divine
			has_variable = pantheon_tick_safety_armed
			NOT = { has_variable = pantheon_quarterly_alive }
		}
		set_variable = { name = pantheon_quarterly_alive value = yes days = 100 }
		trigger_event = { id = pantheon.0010 days = 1 }
	}
	if = {
		limit = { NOT = { has_variable = pantheon_tick_safety_armed } }
		set_variable = pantheon_tick_safety_armed
		trigger_event = { id = pantheon.0012 days = 100 }
	}
}

			name = pantheon_host_army_name
		}
	}
	pantheon_bless_opinion_effect = yes
}

# Remember a follower who now carries a lasting boon, for the Your Blessed page. Root is the god.
pantheon_remember_blessed_effect = {
	scope:pantheon_target = {
		set_variable = {
			name = pantheon_blessed_by
			value = root
		}
		set_variable = {
			name = pantheon_blessed_year
			value = current_year
		}
	}
	if = {
		limit = {
			NOT = {
				is_target_in_variable_list = {
					name = pantheon_blessed
					target = scope:pantheon_target
				}
			}
		}
		add_to_variable_list = { name = pantheon_blessed target = scope:pantheon_target }
	}
}

# 2.2.1 (#4, R1/R2): a timed blessing that is not one of the five ordinary boons -- a Great Boon (KEY = great,
# VALUE = its kind 1-6), a fulfilled Great Prayer (KEY = prayer) or forgiven debts of a landless debtor
# (KEY = debt). Sets the marker pantheon_boon_$KEY$ with the usual blessing period (it expires by itself, like
# every ordinary marker; pantheon_has_any_boon_trigger and pantheon_ui_boon_count know it, so quarterly pruning
# keeps the recipient), the display year pantheon_ends_$KEY$ (untimed, tidied in pantheon.0010 once the marker is
# gone, and overwritten by a renewal -- no scheduled event can clear it early), and registers them on the god's
# Blessed list. Root is the god; scope:pantheon_target the recipient.
pantheon_mark_timed_blessing_effect = {
	scope:pantheon_target = {
		set_variable = {
			name = pantheon_boon_$KEY$
			value = $VALUE$
			years = pantheon_blessing_years
		}
		set_variable = {
			name = pantheon_ends_$KEY$
			value = {
				value = current_year
				add = pantheon_blessing_years
			}
		}
	}
	pantheon_remember_blessed_effect = yes
}

# A boon has faded: tell the god who granted it, if the Boon Expiry Notices rule allows. Root is the follower.
pantheon_expiry_notice_effect = {
	if = {
		limit = {
			exists = var:pantheon_blessed_by
			OR = {
				has_game_rule = pantheon_notices_all
				AND = {
					has_game_rule = pantheon_notices_gifts
					always = $GIFT$
				}
			}
		}
		save_temporary_scope_as = pantheon_faded
		var:pantheon_blessed_by = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = pantheon_expired_$KIND$_t
				left_icon = scope:pantheon_faded
			}
		}
	}
}

# Divine Conception: a god fathers a child on a follower; a goddess bears a follower's child. Root is the god.
# The child is born a Demigod through pantheon_on_birth_child (the god is a real parent).
pantheon_grant_conception_effect = {
	if = {
		limit = { pantheon_can_conceive_trigger = yes }
		change_variable = { name = pantheon_worship subtract = pantheon_cost_life }
		if = {
			limit = { is_female = no }
			scope:pantheon_target = { make_pregnant = { father = root } }
		}
		else = {
			make_pregnant = { father = scope:pantheon_target }
		}
		# 2.2.2 (#6/#8, R1/R3): never a bastard, and remember what the husbands' relations were BEFORE this
		# union, in the pregnant parent's scope (the mother, or the goddess herself).
		save_temporary_scope_as = pantheon_conception_god
		if = {
			limit = { is_female = no }
			scope:pantheon_target = { pantheon_mark_conception_effect = yes }
		}
		else = {
			pantheon_mark_conception_effect = yes
		}
		# An honour, not a sin: neither parent carries adultery or fornication for a god's child
		root = { add_to_temporary_list = pantheon_parents }
		scope:pantheon_target = { add_to_temporary_list = pantheon_parents }
		every_in_list = {
			list = pantheon_parents
			if = {
				limit = { has_trait = adulterer }
				remove_trait = adulterer
			}
			if = {
				limit = { has_trait = fornicator }
				remove_trait = fornicator
			}
		}
		scope:pantheon_target = {
			add_piety = 200
			if = {
				limit = { NOT = { this = root } }
				add_opinion = {
					target = root
					modifier = pantheon_bore_god_child_opinion
				}
			}
		}
		# 2.3.2: the follower (the mother, or the goddess's partner) counts as Blessed for the usual period
		pantheon_mark_timed_blessing_effect = { KEY = conception VALUE = 1 }
		send_interface_toast = {
			type = event_toast_effect_good
			title = pantheon_conceived_t
			left_icon = scope:pantheon_target
		}
		pantheon_bless_opinion_effect = yes
		pantheon_change_devotion_effect = { AMOUNT = 10 }
	}
	else = {
		# Nothing is paid; say why
		send_interface_toast = {
			type = event_toast_effect_bad
			title = pantheon_conception_refused_t
			left_icon = scope:pantheon_target
			if = {
				limit = { has_character_modifier = pantheon_forsworn }
				custom_tooltip = pantheon_conception_forsworn_tt
			}
			if = {
				limit = { scope:pantheon_target = { NOT = { sex_opposite_of = root } } }
				custom_tooltip = pantheon_conception_opposite_sex_tt
			}
			if = {
				limit = { scope:pantheon_target = { is_adult = no } }
				custom_tooltip = pantheon_conception_adult_tt
			}
			if = {
				limit = {
					OR = {
						scope:pantheon_target = { is_pregnant = yes }
						AND = {
							is_female = yes
							is_pregnant = yes
						}
					}
				}
				custom_tooltip = pantheon_conception_pregnant_tt
			}
		}
	}
}

# 2.2.2 (#6/#8, R1/R3). Root is the pregnant parent (the mother; the goddess when the god is the mother);
# scope:pantheon_conception_god is the god, scope:pantheon_target the mortal parent.
#  - R1 prevention: tell the engine the unborn is a concubine's child (set_child_of_concubine_on_pregnancy) and
#    not a known bastard. Vanilla script never reads either (they are engine-side pregnancy state, read when the
#    engine decides scope:is_bastard at birth); UNPROVEN until the in-game checklist. The script-side bastard
#    flags that pregnancy.2001/.2002 set are cleared separately (pantheon_clear_bastard_pregnancy_effect).
#  - R3 provenance: the union is opened (flag pantheon_union_open, 12 months) and, for every consort of either
#    parent, the relations they already had with the two parents are remembered ON THE MOTHER (per union:
#    pantheon_union_snapshotted / pantheon_union_prior_rel_mother / pantheon_union_prior_rel_father), so cleanup
#    only undoes rivalry that this union caused.
pantheon_mark_conception_effect = {
	if = {
		limit = { is_pregnant = yes }
		set_child_of_concubine_on_pregnancy = yes
		set_known_bastard_on_pregnancy = no
	}
	if = {
		limit = { has_trait = pantheon_divine }
		set_variable = { name = pantheon_child_father value = scope:pantheon_target }
	}
	else = {
		set_variable = { name = pantheon_child_father value = scope:pantheon_conception_god }
	}
	add_character_flag = { flag = pantheon_union_open months = 12 }
	# Provenance lives on the MOTHER (root), per union, so two pregnancies by one god never share it
	save_temporary_scope_as = pantheon_snap_mother
	pantheon_clear_union_provenance_effect = yes
	add_to_temporary_list = pantheon_snap_union
	var:pantheon_child_father ?= { add_to_temporary_list = pantheon_snap_union }
	every_in_list = {
		list = pantheon_snap_union
		every_consort = {
			limit = { NOT = { is_in_list = pantheon_snap_union } }
			save_temporary_scope_as = pantheon_snap_holder
			every_in_list = {
				list = pantheon_snap_union
				save_temporary_scope_as = pantheon_snap_member
				scope:pantheon_snap_holder = { pantheon_snapshot_relation_effect = yes }
			}
		}
	}
}

# Root is the mother. Forgets this union's provenance: the consorts snapshotted, who already had a hostile
# relation with the mother / with the father at conception, and the child. Idempotent.
pantheon_clear_union_provenance_effect = {
	if = { limit = { has_variable_list = pantheon_union_snapshotted } clear_variable_list = pantheon_union_snapshotted }
	if = { limit = { has_variable_list = pantheon_union_prior_rel_mother } clear_variable_list = pantheon_union_prior_rel_mother }
	if = { limit = { has_variable_list = pantheon_union_prior_rel_father } clear_variable_list = pantheon_union_prior_rel_father }
	if = { limit = { has_variable = pantheon_union_child } remove_variable = pantheon_union_child }
}

# Holder scope; scope:pantheon_snap_member is the union member, scope:pantheon_snap_mother the mother who holds
# the lists. Remembers (idempotently) that this holder was snapshotted and whether they already had any
# rivalry-type relation with the member (mother list or father list, by the member's role).
pantheon_snapshot_relation_effect = {
	scope:pantheon_snap_mother = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = pantheon_union_snapshotted target = scope:pantheon_snap_holder } } }
			add_to_variable_list = { name = pantheon_union_snapshotted target = scope:pantheon_snap_holder }
		}
	}
	if = {
		limit = {
			OR = {
				has_relation_rival = scope:pantheon_snap_member
				has_relation_nemesis = scope:pantheon_snap_member
				has_relation_grudge = scope:pantheon_snap_member
				has_relation_potential_rival = scope:pantheon_snap_member
			}
		}
		if = {
			limit = { scope:pantheon_snap_member = scope:pantheon_snap_mother }
			scope:pantheon_snap_mother = {
				if = {
					limit = { NOT = { is_target_in_variable_list = { name = pantheon_union_prior_rel_mother target = scope:pantheon_snap_holder } } }
					add_to_variable_list = { name = pantheon_union_prior_rel_mother target = scope:pantheon_snap_holder }
				}
			}
		}
		else = {
			scope:pantheon_snap_mother = {
				if = {
					limit = { NOT = { is_target_in_variable_list = { name = pantheon_union_prior_rel_father target = scope:pantheon_snap_holder } } }
					add_to_variable_list = { name = pantheon_union_prior_rel_father target = scope:pantheon_snap_holder }
				}
			}
		}
	}
}

# 2.2.2 (R1/R2). Root is the mother. Vanilla's pregnancy.2001/.2002 (an unwed or married mother whose child's
# father is not her consort) set the script flags below when they fire, and birth.0001 / pregnancy
# maintenance read them (secrets, birth.1002/.1005, the known-bastard legitimisation event birth.2001). Take
# them away at every boundary Pantheon can reach; idempotent. Not reachable: an option the player picks
# AFTER the day it was cleared sets its flag again (see the CHANGELOG / checklist).
pantheon_clear_bastard_pregnancy_effect = {
	if = { limit = { has_character_flag = bastard_pregnancy } remove_character_flag = bastard_pregnancy }
	if = { limit = { has_character_flag = unmarried_bastard_pregnancy } remove_character_flag = unmarried_bastard_pregnancy }
	if = { limit = { has_character_flag = pregnancy_real_father_of_bastard_is_known_flag } remove_character_flag = pregnancy_real_father_of_bastard_is_known_flag }
	if = { limit = { has_character_flag = pregnancy_spouse_knows_of_bastardy } remove_character_flag = pregnancy_spouse_knows_of_bastardy }
	if = { limit = { has_character_flag = pregnancy_real_father_knows } remove_character_flag = pregnancy_real_father_knows }
	if = { limit = { has_variable = bastard_pregnancy } remove_variable = bastard_pregnancy }
	if = { limit = { has_variable = unmarried_bastard_pregnancy } remove_variable = unmarried_bastard_pregnancy }
	if = {
		limit = { is_pregnant = yes }
		set_child_of_concubine_on_pregnancy = yes
		set_known_bastard_on_pregnancy = no
	}
}

# Make a god's child an honour, not a scandal. Root is the mother; var:pantheon_child_father is the real father.
# Run a day after the pregnancy is revealed (pantheon.0040) and a day after the birth (pantheon.0042): vanilla's
# reveal and birth events brand adultery, create the affair and parentage secrets, and anger the spouses AFTER
# Pantheon's own hooks have run. 2.2.2: the god's own spouse(s) are neutral (no "Unfaithful", no "Slept with My
# Spouse", no honoured line); only the MORTAL parent's spouse is honoured; relations only undo what this union
# caused (see pantheon_mark_conception_effect); secrets only those about this union's child.
pantheon_sanctify_union_effect = {
	save_temporary_scope_as = pantheon_union_mother
	add_to_temporary_list = pantheon_union
	var:pantheon_child_father ?= { add_to_temporary_list = pantheon_union }
	# The bastard flags pregnancy.2001/.2002 may have just set (R1/R2)
	pantheon_clear_bastard_pregnancy_effect = yes
	every_in_list = {
		list = pantheon_union
		save_temporary_scope_as = pantheon_union_member
		if = {
			limit = { has_trait = adulterer }
			remove_trait = adulterer
		}
		if = {
			limit = { has_trait = fornicator }
			remove_trait = fornicator
		}
		# Nothing left to expose: the affair between the two parents, and the parentage secrets about THIS
		# union's child (recorded on the mother at birth) -- never the owner's other secrets
		every_secret = {
			limit = {
				OR = {
					AND = {
						secret_type = secret_lover
						secret_target ?= { is_in_list = pantheon_union }
					}
					AND = {
						OR = {
							secret_type = secret_disputed_heritage
							secret_type = secret_unmarried_illegitimate_child
						}
						scope:pantheon_union_mother = { has_variable = pantheon_union_child }
						secret_target ?= scope:pantheon_union_mother.var:pantheon_union_child
					}
				}
			}
			remove_secret = yes
		}
		every_consort = {
			limit = { NOT = { is_in_list = pantheon_union } }
			save_temporary_scope_as = pantheon_union_holder
			pantheon_union_consort_effect = yes
		}
		# Freed, if their own spouse imprisoned them for it
		if = {
			limit = {
				is_imprisoned = yes
				imprisoner ?= { is_spouse_of = prev }
			}
			release_from_prison = yes
		}
	}
}

# Consort scope; scope:pantheon_union_holder is this consort, scope:pantheon_union_member the union member they
# are consort of. Removes the adultery opinions toward both parents, undoes this union's rivalry, and honours
# the MORTAL parent's consorts (the god's own are neutral).
pantheon_union_consort_effect = {
	every_in_list = {
		list = pantheon_union
		save_temporary_scope_as = pantheon_union_target
		scope:pantheon_union_holder = {
			# Evidence of this union's exposure is the opinion itself: read it BEFORE removing it
			pantheon_union_undo_relations_effect = yes
			remove_opinion = { target = scope:pantheon_union_target modifier = slept_with_spouse_exposed_opinion }
			remove_opinion = { target = scope:pantheon_union_target modifier = slept_with_spouse_discovered_opinion }
			remove_opinion = { target = scope:pantheon_union_target modifier = unfaithful_spouse_exposed_opinion }
			remove_opinion = { target = scope:pantheon_union_target modifier = unfaithful_spouse_discovered_opinion }
		}
	}
	if = {
		limit = { scope:pantheon_union_member = { NOT = { has_trait = pantheon_divine } } }
		every_in_list = {
			list = pantheon_union
			limit = { has_trait = pantheon_divine }
			save_temporary_scope_as = pantheon_union_god
			scope:pantheon_union_holder = {
				if = {
					limit = { NOT = { has_opinion_modifier = { target = scope:pantheon_union_god modifier = pantheon_honoured_spouse_opinion } } }
					add_opinion = { target = scope:pantheon_union_god modifier = pantheon_honoured_spouse_opinion }
				}
			}
		}
	}
}

# Holder scope; scope:pantheon_union_target a union member, scope:pantheon_union_mother the mother holding the
# provenance. Only while the union is open, only if the holder was snapshotted and had NO such relation with
# that member at conception, AND only with evidence the exposure happened: vanilla's adultery exposure adds
# slept_with_spouse_exposed_opinion toward the lover in the same block that sets the nemesis / progresses the
# rivalry (00_adultery_effects.txt), so a holder without that opinion toward the member keeps the relation.
pantheon_union_undo_relations_effect = {
	if = {
		limit = {
			scope:pantheon_union_mother = {
				has_character_flag = pantheon_union_open
				is_target_in_variable_list = { name = pantheon_union_snapshotted target = scope:pantheon_union_holder }
			}
			OR = {
				has_opinion_modifier = { target = scope:pantheon_union_target modifier = slept_with_spouse_exposed_opinion }
				has_opinion_modifier = { target = scope:pantheon_union_target modifier = slept_with_spouse_discovered_opinion }
			}
			OR = {
				AND = {
					scope:pantheon_union_target = scope:pantheon_union_mother
					NOT = { scope:pantheon_union_mother = { is_target_in_variable_list = { name = pantheon_union_prior_rel_mother target = scope:pantheon_union_holder } } }
				}
				AND = {
					NOT = { scope:pantheon_union_target = scope:pantheon_union_mother }
					NOT = { scope:pantheon_union_mother = { is_target_in_variable_list = { name = pantheon_union_prior_rel_father target = scope:pantheon_union_holder } } }
				}
			}
		}
		if = { limit = { has_relation_rival = scope:pantheon_union_target } remove_relation_rival = scope:pantheon_union_target }
		if = { limit = { has_relation_nemesis = scope:pantheon_union_target } remove_relation_nemesis = scope:pantheon_union_target }
		if = { limit = { has_relation_grudge = scope:pantheon_union_target } remove_relation_grudge = scope:pantheon_union_target }
		if = { limit = { has_relation_potential_rival = scope:pantheon_union_target } remove_relation_potential_rival = scope:pantheon_union_target }
	}
}

# Root is the mother, a day after the birth (pantheon.0042): the last sanctify, then the union is closed so a

pantheon_close_union_effect = {

}
