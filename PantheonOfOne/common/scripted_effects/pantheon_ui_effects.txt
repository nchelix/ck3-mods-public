# Pantheon of One 2.0: the Pantheon window's lists, rebuilt on the god whenever the window opens, a tab changes or
# the sort changes. The window cannot sort a datamodel, so the order lives in the list (pantheon_ui_fill_sorted_effect).
# The dead are dropped first. Saved scopes, not temporary ones, and never pantheon_target (1.4 lesson).
pantheon_ui_rebuild_effect = {
	pantheon_forget_the_dead_effect = yes
	# 2.3 (R2): prayers that stopped being answerable leave the list silently before the window reads it
	pantheon_prayer_prune_effect = yes
	# 2.3 B5: Retribution calls that are no longer live (war over, a side dead, defender left the faith) go silently, and a
	# pre-2.3 oath gets its snapshot before the Prayers tab reads it
	pantheon_retri_prune_effect = yes
	pantheon_oath_migrate_effect = yes
	# 2.3 B4: a pending send whose source is gone (prayer closed, recipient dead) goes with it
	pantheon_send_validate_effect = yes
	save_scope_as = pantheon_ui_god
	# Followers: every living ruler of the god's faith, plus anyone already counted (pantheon_known holds those who
	# prayed or were blessed, landless or not). In-game round 1: pantheon_known alone left the tab empty until a blessing.
	every_ruler = {
		limit = { is_alive = yes faith = scope:pantheon_ui_god.faith NOT = { this = scope:pantheon_ui_god } }
		add_to_list = pantheon_ui_follower_pool
	}
	every_in_list = {
		variable = pantheon_known
		limit = { is_alive = yes NOT = { this = scope:pantheon_ui_god } }
		add_to_list = pantheon_ui_follower_pool
	}
	every_in_list = {
		variable = pantheon_blessed
		limit = { is_alive = yes NOT = { this = scope:pantheon_ui_god } }
		add_to_list = pantheon_ui_blessed_pool
	}
	pantheon_ui_fill_sorted_effect = { LIST = pantheon_ui_follower_pool TARGET = pantheon_ui_followers DEFAULT = pantheon_ui_devotion }
	pantheon_ui_fill_sorted_effect = { LIST = pantheon_ui_blessed_pool TARGET = pantheon_ui_blessed DEFAULT = pantheon_ui_devotion }
	# Children tab: the god's living demigods (best total skill first unless a skill is chosen), and the heads of
	# every house holding a living Godsblood
	every_child = {
		limit = { has_trait = pantheon_demigod is_alive = yes }
		add_to_list = pantheon_ui_demigod_pool
	}
	pantheon_ui_fill_sorted_effect = { LIST = pantheon_ui_demigod_pool TARGET = pantheon_ui_demigods DEFAULT = pantheon_total_skill }
	# 2.3 B3: the Prayers tab's rows -- the god's open prayers (pruned above), OLDEST FIRST: the token only grows, so
	# the lowest token is the prayer with the least time left. ordered_in_list takes the highest first, hence the
	# negated token. Built from a temporary pool like the follower lists (a list is never ordered in place).
	clear_variable_list = pantheon_ui_prayers
	every_in_list = {
		variable = pantheon_open_prayers
		limit = { is_alive = yes has_variable = pantheon_prayer_token }
		add_to_list = pantheon_ui_prayer_pool
	}
	ordered_in_list = {
		list = pantheon_ui_prayer_pool
		limit = { is_alive = yes has_variable = pantheon_prayer_token }
		order_by = {
			value = var:pantheon_prayer_token
			multiply = -1
		}
		max = 300
		check_range_bounds = no
		save_scope_as = pantheon_ui_row
		scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_prayers target = scope:pantheon_ui_row } }
	}
	# 2.3 B5: the Prayers tab's Calls for Retribution -- the open, live calls (pruned above), oldest first (lowest token), by the
	# same negated-token ordering as the prayers. The row is the ATTACKER; its defender is on its record.
	clear_variable_list = pantheon_ui_retris
	every_in_list = {
		variable = pantheon_open_retributions
		limit = { is_alive = yes has_variable = pantheon_retri_token }
		add_to_list = pantheon_ui_retri_pool
	}
	ordered_in_list = {
		list = pantheon_ui_retri_pool
		limit = { is_alive = yes has_variable = pantheon_retri_token }
		order_by = {
			value = var:pantheon_retri_token
			multiply = -1
		}
		max = 100
		check_range_bounds = no
		save_scope_as = pantheon_ui_row
		scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_retris target = scope:pantheon_ui_row } }
	}
	clear_variable_list = pantheon_ui_houses
	every_child = {
		limit = { has_trait = pantheon_demigod }
		every_child = {
			limit = { is_alive = yes has_variable = pantheon_godsblood }
			house ?= {
				house_head ?= {
					save_scope_as = pantheon_ui_row
					# A house with several living Godsbloods would otherwise be added once per one (M1)
					scope:pantheon_ui_god = {
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = pantheon_ui_houses target = scope:pantheon_ui_row } } }
							add_to_variable_list = { name = pantheon_ui_houses target = scope:pantheon_ui_row }
						}
					}
				}
			}
		}
	}
	# 2.2 Task 4: the Temple tab's pre-founding list -- every holy site of the god's own faith (the form
	# pantheon_temple_can_found_trigger already uses, common/scripted_triggers/pantheon_temple_triggers.txt:
	# `any_county_province = { barony = { is_holy_site_of = root.faith } }`, read from county scope).
	# Skipped once a temple exists (nothing to found any more).
	# Batch 1r1 repair (item 4, concern): the review suggested narrowing this world-wide every_county scan
	# to the faith's own holy sites (every_holy_site_barony, faith scope). Left AS-IS: every_holy_site_barony
	# and every_holy_site are documented as iterating only ACTIVE holy sites -- "controlled by the correct
	# faith on the county level and have an enshrined holy relic" (event_targets.log:872/896,
	# barony_active_holy_site/county_active_holy_site) -- a materially narrower condition than the
	# is_holy_site_of check this scan (and pantheon_temple_can_found_trigger itself) already uses, which has
	# no "active"/enshrined-relic requirement. Swapping risks silently dropping a valid founding site (no
	# relic enshrined, or momentarily not held by the faith) from this pick list while can_found_trigger
	# would still accept it elsewhere -- not provable safe from docs/reference alone, so not changed here.
	if = {
		limit = {
			NOT = { has_variable = pantheon_temple_tier }
			NOT = { has_variable = pantheon_temple_building }
			NOT = { has_variable = pantheon_temple_province }
		}
		clear_variable_list = pantheon_ui_temple_sites
		every_county = {
			limit = { any_county_province = { barony = { is_holy_site_of = scope:pantheon_ui_god.faith } } }
			save_scope_as = pantheon_ui_row
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_temple_sites target = scope:pantheon_ui_row } }
		}
		# R13, batch 2b (spec Addendum): the capital county too, unless the loop above already added it as
		# a holy site.
		if = {
			limit = {
				NOT = { is_target_in_variable_list = { name = pantheon_ui_temple_sites target = scope:pantheon_ui_god.capital_county } }
			}
			scope:pantheon_ui_god.capital_county = { save_scope_as = pantheon_ui_row }
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_temple_sites target = scope:pantheon_ui_row } }
		}
	}
}

# Fix round 1 (F5): a record closed from an EVENT (a wrath landing, a deadline, a Send, an oath ending) must also leave an
# open window's lists, or its row stays behind with the record gone. Rebuilds only while the window is open (root = the
# god). NEVER call this from the _core_ closers or the prune effects: the rebuild prunes through them (recursion).
# Fix round 2 (H3): the Prayers tab opens on the first sub-tab that has something: Prayers, then Retribution, then
# Great Prayers; all empty -> Prayers. Run AFTER pantheon_ui_rebuild_effect (it prunes stale records first). God scope.
pantheon_ui_pick_ptab_effect = {
	if = {
		limit = { pantheon_ui_count_prayers > 0 }
		set_variable = { name = pantheon_ui_ptab value = 1 }
	}
	else_if = {
		limit = { pantheon_ui_count_retris > 0 }
		set_variable = { name = pantheon_ui_ptab value = 2 }
	}
	else_if = {
		limit = { pantheon_ui_count_great > 0 }
		set_variable = { name = pantheon_ui_ptab value = 3 }
	}
	else = {
		set_variable = { name = pantheon_ui_ptab value = 1 }
	}
}

pantheon_ui_refresh_if_open_effect = {
	if = {
		# only as the god itself: the oath's death hook reaches here with root = the dying offender (Codex fix-round review)
		limit = {
			this = root
			has_variable = pantheon_ui_open
		}
		pantheon_ui_rebuild_effect = yes
	}
}

# Copy temporary list $LIST$ into the god's variable list $TARGET$, ordered by the chosen sort (var:pantheon_ui_sort:
# 1 diplomacy, 2 martial, 3 stewardship, 4 intrigue, 5 learning, 6 prowess), else by the script value $DEFAULT$.
# Root (the god) is scope:pantheon_ui_god.
pantheon_ui_fill_sorted_effect = {
	clear_variable_list = $TARGET$
	if = {
		limit = { has_variable = pantheon_ui_sort var:pantheon_ui_sort = 1 }
		ordered_in_list = { list = $LIST$ order_by = diplomacy max = 300 check_range_bounds = no save_scope_as = pantheon_ui_row scope:pantheon_ui_god = { add_to_variable_list = { name = $TARGET$ target = scope:pantheon_ui_row } } }
	}
	else_if = {
		limit = { has_variable = pantheon_ui_sort var:pantheon_ui_sort = 2 }
		ordered_in_list = { list = $LIST$ order_by = martial max = 300 check_range_bounds = no save_scope_as = pantheon_ui_row scope:pantheon_ui_god = { add_to_variable_list = { name = $TARGET$ target = scope:pantheon_ui_row } } }
	}
	else_if = {
		limit = { has_variable = pantheon_ui_sort var:pantheon_ui_sort = 3 }
		ordered_in_list = { list = $LIST$ order_by = stewardship max = 300 check_range_bounds = no save_scope_as = pantheon_ui_row scope:pantheon_ui_god = { add_to_variable_list = { name = $TARGET$ target = scope:pantheon_ui_row } } }
	}
	else_if = {
		limit = { has_variable = pantheon_ui_sort var:pantheon_ui_sort = 4 }
		ordered_in_list = { list = $LIST$ order_by = intrigue max = 300 check_range_bounds = no save_scope_as = pantheon_ui_row scope:pantheon_ui_god = { add_to_variable_list = { name = $TARGET$ target = scope:pantheon_ui_row } } }
	}
	else_if = {
		limit = { has_variable = pantheon_ui_sort var:pantheon_ui_sort = 5 }
		ordered_in_list = { list = $LIST$ order_by = learning max = 300 check_range_bounds = no save_scope_as = pantheon_ui_row scope:pantheon_ui_god = { add_to_variable_list = { name = $TARGET$ target = scope:pantheon_ui_row } } }
	}
	else_if = {
		limit = { has_variable = pantheon_ui_sort var:pantheon_ui_sort = 6 }
		ordered_in_list = { list = $LIST$ order_by = prowess max = 300 check_range_bounds = no save_scope_as = pantheon_ui_row scope:pantheon_ui_god = { add_to_variable_list = { name = $TARGET$ target = scope:pantheon_ui_row } } }
	}
	else = {
		ordered_in_list = { list = $LIST$ order_by = $DEFAULT$ max = 300 check_range_bounds = no save_scope_as = pantheon_ui_row scope:pantheon_ui_god = { add_to_variable_list = { name = $TARGET$ target = scope:pantheon_ui_row } } }
	}
}

# Batch 2r1 (repair 4): an armed miracle (pantheon_temple_confirm) must not survive leaving the Temple tab
# or closing the window -- called from every window open/close/toggle path and every tab switch in
# common/scripted_guis/pantheon_scripted_guis.txt. Not called from pantheon_ui_rebuild_effect itself: the
# three miracle click effects below call that too, after arming, so clearing the confirm there would
# disarm a miracle on the very click that armed it. Scope = character (the god).
pantheon_temple_clear_confirm_effect = {
	if = { limit = { has_variable = pantheon_temple_confirm } remove_variable = pantheon_temple_confirm }
}

# 2.2 Task 4: the tier-5 miracle buttons arm on the first click and apply on the second (One Trait's
# confirm form, OneTraitToRuleThemAll/common/scripted_effects/one_trait_panel_effects.txt
# one_trait_ui_gov_feudal_click_effect); clicking a different miracle overwrites pantheon_temple_confirm
# with its own code, disarming whichever was armed before.
pantheon_ui_miracle_conversion_click_effect = {
	if = {
		limit = { has_variable = pantheon_temple_confirm var:pantheon_temple_confirm = 1 }
		remove_variable = pantheon_temple_confirm
		pantheon_miracle_conversion_effect = yes
	}
	else = {
		set_variable = { name = pantheon_temple_confirm value = 1 }
	}
	pantheon_ui_rebuild_effect = yes
}
