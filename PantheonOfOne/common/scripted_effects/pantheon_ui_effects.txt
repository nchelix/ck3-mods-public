# Pantheon of One 2.0: the Pantheon window's lists, rebuilt on the god whenever the window opens, a tab changes or
# the sort changes. The window cannot sort a datamodel, so the order lives in the list (pantheon_ui_fill_sorted_effect).
# The dead are dropped first. Saved scopes, not temporary ones, and never pantheon_target (1.4 lesson).
# 2.9.1 repair 4: a temporary list lives until the enclosing top-level effect ends, not until each scripted-effect call, and one
# effect can run the rebuild twice (pantheon.0011: oath upkeep rebuilds an open window, then the glance sweep rebuilds again).
# The second build would read the first build's pools, so every pool is emptied before anything fills one.
pantheon_ui_clear_pools_effect = {
	every_in_list = { list = pantheon_ui_lost_pool remove_from_list = pantheon_ui_lost_pool }
	every_in_list = { list = pantheon_ui_follower_pool remove_from_list = pantheon_ui_follower_pool }
	every_in_list = { list = pantheon_ui_blessed_pool remove_from_list = pantheon_ui_blessed_pool }
	every_in_list = { list = pantheon_ui_demigod_pool remove_from_list = pantheon_ui_demigod_pool }
	every_in_list = { list = pantheon_ui_prayer_pool remove_from_list = pantheon_ui_prayer_pool }
	every_in_list = { list = pantheon_ui_retri_pool remove_from_list = pantheon_ui_retri_pool }
	every_in_list = { list = pantheon_ui_quiet_pool remove_from_list = pantheon_ui_quiet_pool }
	every_in_list = { list = pantheon_steady_ui_pool remove_from_list = pantheon_steady_ui_pool }
	every_in_list = { list = pantheon_steady_ui_titles remove_from_list = pantheon_steady_ui_titles }
}

pantheon_ui_rebuild_effect = {
	pantheon_ui_clear_pools_effect = yes
	pantheon_forget_the_dead_effect = yes
	# 2.3 (R2): prayers that stopped being answerable leave the list silently before the window reads it
	pantheon_prayer_prune_effect = yes
	# 2.9 (Task 4): quiet prayers that faded, left the faith or lost their need close silently (with the one-year lockout) before the
	# Quiet sub-tab's lists are read. The prune only closes and never calls back into this rebuild.
	pantheon_quiet_prune_effect = yes
	# 2.3 B5: Retribution calls that are no longer live (war over, a side dead, defender left the faith) go silently, and a
	# pre-2.3 oath gets its snapshot before the Prayers tab reads it
	pantheon_retri_prune_effect = yes
	pantheon_oath_migrate_effect = yes
	# 2.3 B4: a pending send whose source is gone (prayer closed, recipient dead) goes with it
	pantheon_send_validate_effect = yes
	save_scope_as = pantheon_ui_god
	# 2.6 repair 1 (UI test LW2): the Great Boons left waiting, for the Great Prayers sub-tab. The same rule as their count
	# (pantheon_ui_count_waiting_boons): still the live, valid request the wait was set for.
	clear_variable_list = pantheon_ui_waiting_boons
	if = {
		limit = { has_variable_list = pantheon_boon_holders }
		every_in_list = {
			variable = pantheon_boon_holders
			limit = {
				pantheon_live_character_trigger = yes
				has_variable = pantheon_aspiration_live
				has_variable = pantheon_aspiration_waiting
				has_variable = pantheon_aspiration_id
				var:pantheon_aspiration_waiting = var:pantheon_aspiration_id
				pantheon_aspiration_valid_trigger = { GOD = scope:pantheon_ui_god }
			}
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_waiting_boons target = prev } }
		}
	}
	# 2.9.1 (#69): every pool below is a TEMPORARY list. pantheon.0011 reschedules itself, so the event chain never breaks and an
	# persistent list would outlive the rebuild and keep the dead and the departed. Repair 4: the pools are also emptied at the start of
	# every rebuild (pantheon_ui_clear_pools_effect), because one effect can rebuild twice and the second would read the first's pools.
	# 2.7: the Followers tab's Lost followers sub-tab -- the living members of pantheon_lost, the latest departure first
	# (year 0 = "before 2.7" last). Copied through a temporary pool; the list is never ordered in place. The monthly sweep
	# prunes the dead; is_alive here covers the days between.
	clear_variable_list = pantheon_ui_lost
	if = {
		limit = { has_variable_list = pantheon_lost }
		every_in_list = {
			variable = pantheon_lost
			limit = { pantheon_live_character_trigger = yes }
			add_to_temporary_list = pantheon_ui_lost_pool
		}
		ordered_in_list = {
			list = pantheon_ui_lost_pool
			order_by = {
				value = 0
				if = {
					limit = { has_variable = pantheon_alert_left_year }
					add = var:pantheon_alert_left_year
				}
			}
			max = 100000
			check_range_bounds = no
			save_scope_as = pantheon_ui_row
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_lost target = scope:pantheon_ui_row } }
		}
	}
	# Followers: every living ruler of the god's faith, plus anyone already counted (pantheon_known holds those who
	# prayed or were blessed, landless or not). In-game round 1: pantheon_known alone left the tab empty until a blessing.
	every_ruler = {
		limit = { pantheon_live_character_trigger = yes faith = scope:pantheon_ui_god.faith NOT = { this = scope:pantheon_ui_god } }
		add_to_temporary_list = pantheon_ui_follower_pool
	}
	# 2.11 P2b: a ruler of the faith who is also in pantheon_known is added once. The docs do not promise that a temporary
	# list refuses a second add, so the walk skips anyone already in the pool (the is_in_list guard of pantheon_kin_one_effect).
	every_in_list = {
		variable = pantheon_known
		limit = { pantheon_live_character_trigger = yes NOT = { this = scope:pantheon_ui_god } NOT = { is_in_list = pantheon_ui_follower_pool } }
		add_to_temporary_list = pantheon_ui_follower_pool
	}
	# 2.11 P2b: the God page's follower statistics, over the whole pool (the sorted list below stops at 300)
	pantheon_ui_follower_stats_effect = yes
	every_in_list = {
		variable = pantheon_blessed
		limit = { pantheon_live_character_trigger = yes NOT = { this = scope:pantheon_ui_god } }
		add_to_temporary_list = pantheon_ui_blessed_pool
	}
	pantheon_ui_fill_sorted_effect = { LIST = pantheon_ui_follower_pool TARGET = pantheon_ui_followers DEFAULT = pantheon_ui_devotion }
	pantheon_ui_fill_sorted_effect = { LIST = pantheon_ui_blessed_pool TARGET = pantheon_ui_blessed DEFAULT = pantheon_ui_devotion }
	# Children tab: the god's living demigods (best total skill first unless a skill is chosen), and the heads of
	# every house holding a living Godsblood
	every_child = {
		limit = { has_trait = pantheon_demigod is_alive = yes }
		add_to_temporary_list = pantheon_ui_demigod_pool
	}
	pantheon_ui_fill_sorted_effect = { LIST = pantheon_ui_demigod_pool TARGET = pantheon_ui_demigods DEFAULT = pantheon_total_skill }
	# 2.3 B3: the Prayers tab's rows -- the god's open prayers (pruned above), OLDEST FIRST: the token only grows, so
	# the lowest token is the prayer with the least time left. ordered_in_list takes the highest first, hence the
	# negated token. Built from a temporary pool like the follower lists (a list is never ordered in place).
	clear_variable_list = pantheon_ui_prayers
	every_in_list = {
		variable = pantheon_open_prayers
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_prayer_token }
		add_to_temporary_list = pantheon_ui_prayer_pool
	}
	ordered_in_list = {
		list = pantheon_ui_prayer_pool
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_prayer_token }
		order_by = {
			value = var:pantheon_prayer_token
			multiply = -1
		}
		max = 300
		check_range_bounds = no
		save_scope_as = pantheon_ui_row
		scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_prayers target = scope:pantheon_ui_row } }
	}
	# 2.3 B5: the Prayers tab's Calls for Retribution -- the open, live calls (pruned above), oldest first (lowest token), by the
	# same negated-token ordering as the prayers. The row is the ATTACKER; its defender is on its record.
	clear_variable_list = pantheon_ui_retris
	every_in_list = {
		variable = pantheon_open_retributions
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_retri_token }
		add_to_temporary_list = pantheon_ui_retri_pool
	}
	ordered_in_list = {
		list = pantheon_ui_retri_pool
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_retri_token }
		order_by = {
			value = var:pantheon_retri_token
			multiply = -1
		}
		max = 100
		check_range_bounds = no
		save_scope_as = pantheon_ui_row
		scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_retris target = scope:pantheon_ui_row } }
	}
	# 2.9 (Task 4): the Quiet sub-tab. Rows = the god's quiet prayers (pruned above) that are live and still marked, WEAKEST DEVOTION
	# FIRST: ordered_in_list takes the highest first, hence the negated Devotion (pantheon_ui_devotion reads 50 when they have none).
	# Copied through a pool like the other lists. The limit is repeated on the ordering, as the prayers' is, because a pool built
	# earlier in the same chain may hold someone no longer valid.
	clear_variable_list = pantheon_ui_quiet
	clear_variable_list = pantheon_ui_quiet_kingdoms
	clear_variable_list = pantheon_ui_quiet_kingdom_all
	every_in_list = {
		variable = pantheon_quiet_prayers
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_quiet_open }
		add_to_temporary_list = pantheon_ui_quiet_pool
	}
	ordered_in_list = {
		list = pantheon_ui_quiet_pool
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_quiet_open }
		order_by = {
			value = pantheon_ui_devotion
			multiply = -1
		}
		max = 100
		check_range_bounds = no
		save_scope_as = pantheon_ui_row
		scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_quiet target = scope:pantheon_ui_row } }
	}
	# The kingdom summary: how many quiet prayers each de jure kingdom holds. The count lives on the kingdom TITLE
	# (pantheon_ui_quiet_n), so first every kingdom about to be counted is reset to 0 -- a stale count from an earlier build (or one
	# the window close left behind) never adds to a new one. A follower with no de jure kingdom has no pantheon_quiet_kingdom and is
	# skipped here (their row says "No kingdom"); the title is only entered through var:, behind exists. One entry per kingdom
	# goes to a hidden list, and the top FIVE by count go to the window's list; the hidden list is dropped again at once.
	every_in_list = {
		variable = pantheon_ui_quiet
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_quiet_kingdom exists = var:pantheon_quiet_kingdom }
		var:pantheon_quiet_kingdom = {
			set_variable = { name = pantheon_ui_quiet_n value = 0 }
		}
	}
	every_in_list = {
		variable = pantheon_ui_quiet
		limit = { pantheon_live_character_trigger = yes has_variable = pantheon_quiet_kingdom exists = var:pantheon_quiet_kingdom }
		var:pantheon_quiet_kingdom = {
			change_variable = { name = pantheon_ui_quiet_n add = 1 }
			save_scope_as = pantheon_ui_row
			scope:pantheon_ui_god = {
				if = {
					limit = { NOT = { is_target_in_variable_list = { name = pantheon_ui_quiet_kingdom_all target = scope:pantheon_ui_row } } }
					add_to_variable_list = { name = pantheon_ui_quiet_kingdom_all target = scope:pantheon_ui_row }
				}
			}
		}
	}
	if = {
		limit = { has_variable_list = pantheon_ui_quiet_kingdom_all }
		ordered_in_list = {
			variable = pantheon_ui_quiet_kingdom_all
			order_by = { value = var:pantheon_ui_quiet_n }
			max = 5
			check_range_bounds = no
			save_scope_as = pantheon_ui_row
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_quiet_kingdoms target = scope:pantheon_ui_row } }
		}
	}
	clear_variable_list = pantheon_ui_quiet_kingdom_all
	clear_variable_list = pantheon_ui_houses
	every_child = {
		limit = { has_trait = pantheon_demigod }
		every_child = {
			limit = { is_alive = yes has_variable = pantheon_godsblood }
			house ?= {
				house_head ?= {
					save_scope_as = pantheon_ui_row
					# A house with several living Godsbloods would otherwise be added once per one (M1)
					scope:pantheon_ui_god = {
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = pantheon_ui_houses target = scope:pantheon_ui_row } } }
							add_to_variable_list = { name = pantheon_ui_houses target = scope:pantheon_ui_row }
						}
					}
				}
			}
		}
	}
	# 2.8 (#59): the God tab's Seeded lands, copied from the god's own list. Only live seeds of this god are shown (an expired one
	# is pruned by the monthly pass; this keeps the window right in between). Counties never leave the world, so no dead entries.
	clear_variable_list = pantheon_ui_seeds
	if = {
		limit = { has_variable_list = pantheon_seeded_counties }
		every_in_list = {
			variable = pantheon_seeded_counties
			limit = { pantheon_county_seeded_trigger = { GOD = scope:pantheon_ui_god } }
			save_scope_as = pantheon_ui_row
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_seeds target = scope:pantheon_ui_row } }
		}
	}
	# 2.2 Task 4: the Temple tab's pre-founding list -- every holy site of the god's own faith (the form
	# pantheon_temple_can_found_trigger already uses, common/scripted_triggers/pantheon_temple_triggers.txt:
	# `any_county_province = { barony = { is_holy_site_of = root.faith } }`, read from county scope).
	# Skipped once a temple exists (nothing to found any more).
	# Batch 1r1 repair (item 4, concern): the review suggested narrowing this world-wide every_county scan
	# to the faith's own holy sites (every_holy_site_barony, faith scope). Left AS-IS: every_holy_site_barony
	# and every_holy_site are documented as iterating only ACTIVE holy sites -- "controlled by the correct
	# faith on the county level and have an enshrined holy relic" (event_targets.log:872/896,
	# barony_active_holy_site/county_active_holy_site) -- a materially narrower condition than the
	# is_holy_site_of check this scan (and pantheon_temple_can_found_trigger itself) already uses, which has
	# no "active"/enshrined-relic requirement. Swapping risks silently dropping a valid founding site (no
	# relic enshrined, or momentarily not held by the faith) from this pick list while can_found_trigger
	# would still accept it elsewhere -- not provable safe from docs/reference alone, so not changed here.
	if = {
		limit = {
			NOT = { has_variable = pantheon_temple_tier }
			NOT = { has_variable = pantheon_temple_building }
			NOT = { has_variable = pantheon_temple_province }
		}
		clear_variable_list = pantheon_ui_temple_sites
		every_county = {
			limit = { any_county_province = { barony = { is_holy_site_of = scope:pantheon_ui_god.faith } } }
			save_scope_as = pantheon_ui_row
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_temple_sites target = scope:pantheon_ui_row } }
		}
		# R13, batch 2b (spec Addendum): the capital county too, unless the loop above already added it as
		# a holy site.
		if = {
			limit = {
				NOT = { is_target_in_variable_list = { name = pantheon_ui_temple_sites target = scope:pantheon_ui_god.capital_county } }
			}
			scope:pantheon_ui_god.capital_county = { save_scope_as = pantheon_ui_row }
			scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_temple_sites target = scope:pantheon_ui_row } }
		}
	}
	# 2.9 (Task 5): the Temple tab's kingdom list for the steadying act (the kingdoms that hold followers, most first, each carrying
	# its count). Built ONLY while the Temple tab is shown and a temple stands: it walks the whole follower pool. Anywhere else it is
	# emptied, so no row of it lingers.
	clear_variable_list = pantheon_ui_steady_kingdoms
	if = {
		limit = {
			has_variable = pantheon_ui_tab
			var:pantheon_ui_tab = 4
			has_variable = pantheon_temple_province
		}
		pantheon_steady_ui_build_effect = yes
	}
}

# 2.9 (Task 5): the Temple tab's kingdom list (called from pantheon_ui_rebuild_effect ONLY while the Temple tab shows and a temple exists; god scope,
# scope:pantheon_ui_god saved). Counts the followers in each de jure kingdom on the TITLE (pantheon_ui_steady_n), after resetting every
# kingdom about to be counted, so a stale count never adds to a new one; a follower with no kingdom counts toward none. The titles go
# to a deduplicated list and the god's list pantheon_ui_steady_kingdoms is filled most followers first. It also leaves the two counts
# the arm buttons read to grey an empty reach (pantheon_ui_steady_all_n, pantheon_ui_steady_slipping_n).
pantheon_steady_ui_build_effect = {
	set_variable = { name = pantheon_ui_steady_all_n value = 0 }
	set_variable = { name = pantheon_ui_steady_slipping_n value = 0 }
	pantheon_steady_pool_effect = { LIST = pantheon_steady_ui_pool }
	every_in_list = {
		list = pantheon_steady_ui_pool
		limit = { pantheon_live_character_trigger = yes }
		scope:pantheon_ui_god = { change_variable = { name = pantheon_ui_steady_all_n add = 1 } }
		if = {
			limit = { pantheon_steady_slipping_trigger = yes }
			scope:pantheon_ui_god = { change_variable = { name = pantheon_ui_steady_slipping_n add = 1 } }
		}
		pantheon_steady_region_effect = yes
		if = {
			limit = { exists = scope:pantheon_steady_region_one }
			scope:pantheon_steady_region_one = { set_variable = { name = pantheon_ui_steady_n value = 0 } }
		}
	}
	every_in_list = {
		list = pantheon_steady_ui_pool
		limit = { pantheon_live_character_trigger = yes }
		pantheon_steady_region_effect = yes
		if = {
			limit = { exists = scope:pantheon_steady_region_one }
			scope:pantheon_steady_region_one = {
				change_variable = { name = pantheon_ui_steady_n add = 1 }
				if = {
					limit = { NOT = { is_in_list = pantheon_steady_ui_titles } }
					add_to_temporary_list = pantheon_steady_ui_titles
				}
			}
		}
	}
	ordered_in_list = {
		list = pantheon_steady_ui_titles
		order_by = { value = var:pantheon_ui_steady_n }
		max = 300
		check_range_bounds = no
		save_scope_as = pantheon_ui_row
		scope:pantheon_ui_god = { add_to_variable_list = { name = pantheon_ui_steady_kingdoms target = scope:pantheon_ui_row } }
	}
	if = { limit = { exists = scope:pantheon_steady_region_one } clear_saved_scope = pantheon_steady_region_one }
}

# Fix round 1 (F5): a record closed from an EVENT (a wrath landing, a deadline, a Send, an oath ending) must also leave an
# open window's lists, or its row stays behind with the record gone. Rebuilds only while the window is open (root = the
# god). NEVER call this from the _core_ closers or the prune effects: the rebuild prunes through them (recursion).
# Fix round 2 (H3): the Prayers tab opens on the first sub-tab that has something: Prayers, then Retribution, then
# Great Prayers, then (2.9) Quiet; all empty -> Prayers. Run AFTER pantheon_ui_rebuild_effect (it prunes stale records first). God scope.
pantheon_ui_pick_ptab_effect = {
	if = {
		limit = { pantheon_ui_count_prayers > 0 }
		set_variable = { name = pantheon_ui_ptab value = 1 }
	}
	else_if = {
		limit = { pantheon_ui_count_retris > 0 }
		set_variable = { name = pantheon_ui_ptab value = 2 }
	}
	else_if = {
		limit = { pantheon_ui_count_great > 0 }
		set_variable = { name = pantheon_ui_ptab value = 3 }
	}
	# 2.9 (Task 4): the Quiet sub-tab comes last, so a popup, a call or a Great Prayer always shows first
	else_if = {
		limit = { pantheon_ui_count_quiet > 0 }
		set_variable = { name = pantheon_ui_ptab value = 4 }
	}
	else = {
		set_variable = { name = pantheon_ui_ptab value = 1 }
	}
}

pantheon_ui_refresh_if_open_effect = {
	if = {
		# only as the god itself: the oath's death hook reaches here with root = the dying offender (Codex fix-round review)
		limit = {
			this = root
			has_variable = pantheon_ui_open
		}
		pantheon_ui_rebuild_effect = yes
	}
}

# 2.11 P2b: the God page's follower statistics, stored on the god for the script values pantheon_ui_follower_count,
# _measured, pantheon_ui_devotion_avg and pantheon_ui_band_* (pantheon_values.txt). Counted over the whole follower set,
# pantheon_ui_follower_pool (every living ruler of the god's faith plus pantheon_known, the god left out, each once), not
# over pantheon_ui_followers, which pantheon_ui_fill_sorted_effect caps at 300: a faith with 1,000 followers must not read
# "300 followers", and its average and bands must not lean to the 300 most devout. Every stat is reset first, so a value
# from an earlier rebuild never survives. "Measured" = has pantheon_devotion; the Devotion sum and the five bands count only
# the measured. The bands are the Follower trait's (common/traits/pantheon_follower_trait.txt): Zealous >= 90, Devout 75-89,
# Faithful 25-74, Wavering 10-24, Lapsing < 10. God scope, called by pantheon_ui_rebuild_effect once the pool is complete
# (scope:pantheon_ui_god is saved there). UI only: nothing here changes play.
pantheon_ui_follower_stats_effect = {
	set_variable = { name = pantheon_ui_stat_count value = 0 }
	set_variable = { name = pantheon_ui_stat_measured value = 0 }
	set_variable = { name = pantheon_ui_stat_devotion_sum value = 0 }
	set_variable = { name = pantheon_ui_stat_zealous value = 0 }
	set_variable = { name = pantheon_ui_stat_devout value = 0 }
	set_variable = { name = pantheon_ui_stat_faithful value = 0 }
	set_variable = { name = pantheon_ui_stat_wavering value = 0 }
	set_variable = { name = pantheon_ui_stat_lapsing value = 0 }
	every_in_list = {
		list = pantheon_ui_follower_pool
		limit = { pantheon_live_character_trigger = yes }
		scope:pantheon_ui_god = { change_variable = { name = pantheon_ui_stat_count add = 1 } }
		if = {
			limit = { has_variable = pantheon_devotion }
			save_scope_as = pantheon_ui_stat_row
			scope:pantheon_ui_god = {
				change_variable = { name = pantheon_ui_stat_measured add = 1 }
