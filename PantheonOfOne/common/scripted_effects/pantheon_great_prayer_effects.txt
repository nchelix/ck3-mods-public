# Pantheon of One 1.5: Great Prayers. Root is the god unless noted.

# A follower suffers a wrong: $WHO$ gains a grievance against $OFFENDER$ of $KIND$ (1 murdered kin, 2 lost a
# war, 3 holy site seized, 4 imprisoned). A newer wrong replaces an older one, except that nothing replaces a
# torture (kind 5, user 2026-10-05); all three variables fade after 5 years.
pantheon_give_grievance_effect = {
	$WHO$ = {
		if = {
			limit = { NOT = { AND = { has_variable = pantheon_grievance var:pantheon_grievance_kind = 5 } } }
			set_variable = { name = pantheon_grievance value = yes years = 5 }
			set_variable = { name = pantheon_grievance_target value = $OFFENDER$ years = 5 }
			set_variable = { name = pantheon_grievance_kind value = $KIND$ years = 5 }
			pantheon_wronged_register_effect = yes
		}
	}
}

# Tortured (kind 5): always replaces, even an older torture.
pantheon_give_torture_grievance_effect = {
	$WHO$ = {
		set_variable = { name = pantheon_grievance value = yes years = 5 }
		set_variable = { name = pantheon_grievance_target value = $OFFENDER$ years = 5 }
		set_variable = { name = pantheon_grievance_kind value = 5 years = 5 }
		pantheon_wronged_register_effect = yes
	}
}

# 2.6 (R11, R13): the offender keeps a LIST of the followers they wronged (pantheon_wronged_by_list), so the Pass Judgement menu
# item can be highlighted without a world scan. Called in follower scope, ONLY in the branches that actually write a grievance
# (the ordinary one, the torture one, the raid one) and by the monthly upkeep's backfill of old saves. pantheon_wronged_latest
# names the newest, for the highlight's reason (the reason checks it is still valid).
pantheon_wronged_register_effect = {
	save_temporary_scope_as = pantheon_wronged_one
	var:pantheon_grievance_target ?= {
		if = {
			limit = { is_alive = yes }
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = pantheon_wronged_by_list target = scope:pantheon_wronged_one } } }
				add_to_variable_list = { name = pantheon_wronged_by_list target = scope:pantheon_wronged_one }
			}
			set_variable = { name = pantheon_wronged_latest value = scope:pantheon_wronged_one }
		}
	}
}

# Offender scope: drop the entries that are no longer a live grievance against them (dead, grievance gone, or aimed at someone
# else). The stale ones are gathered first and removed afterwards: a list is never edited while it is walked. Every removal
# re-checks staleness, so a temporary list that outlives one call can never remove a valid entry.
pantheon_wronged_prune_effect = {
	save_temporary_scope_as = pantheon_wronged_offender
	every_in_list = {
		variable = pantheon_wronged_by_list
		limit = { pantheon_wronged_stale_trigger = yes }
		add_to_temporary_list = pantheon_wronged_stale
	}
	every_in_list = {
		list = pantheon_wronged_stale
		limit = { pantheon_wronged_stale_trigger = yes }
		save_temporary_scope_as = pantheon_wronged_stale_one
		scope:pantheon_wronged_offender = {
			if = {
				limit = { is_target_in_variable_list = { name = pantheon_wronged_by_list target = scope:pantheon_wronged_stale_one } }
				remove_list_variable = { name = pantheon_wronged_by_list target = scope:pantheon_wronged_stale_one }
			}
		}
	}
}

# In the follower's scope: the grievance is answered, refused or broken.
pantheon_clear_grievance_effect = {
	if = { limit = { has_variable = pantheon_grievance } remove_variable = pantheon_grievance }
	if = { limit = { has_variable = pantheon_grievance_target } remove_variable = pantheon_grievance_target }
	if = { limit = { has_variable = pantheon_grievance_kind } remove_variable = pantheon_grievance_kind }
}

# Tortured (kind 5). There is no torture hook, so each quarter (pantheon.0010) look at followers newly tortured:
# vanilla's torture_memory_effect gives the victim recently_tortured and a tortured_memory naming the torturer.
# The victim and their devout kin hold it against the torturer, whatever the torturer's faith (user,
# 2026-10-05). Root is the god.
pantheon_torture_grievance_pass_effect = {
	every_ruler = {
		limit = {
			faith = root.faith
			has_character_modifier = recently_tortured
			NOT = { has_variable = pantheon_torture_noted }
		}
		pantheon_note_torture_effect = yes
	}
	every_courtier = {
		limit = {
			faith = root.faith
			has_character_modifier = recently_tortured
			NOT = { has_variable = pantheon_torture_noted }
		}
		pantheon_note_torture_effect = yes
	}
}

# In the victim's scope. Root is the god. Everything happens inside the newest torture memory's torturer: a
# victim tortured without a memory (some vanilla events add recently_tortured only) gets nothing, rather than
# the previous victim's torturer from a scope the quarterly chain inherited (1.5 review).
pantheon_note_torture_effect = {
	set_variable = { name = pantheon_torture_noted value = yes years = 5 }
	save_scope_as = pantheon_tortured
	ordered_memory = {
		memory_type = tortured_memory
		order_by = memory_creation_date
		memory_participant:torturer ?= {
			save_scope_as = pantheon_torturer
			scope:pantheon_tortured = {
				if = {
					limit = { pantheon_grievance_eligible_trigger = { GOD = root } }
					pantheon_give_torture_grievance_effect = { WHO = this OFFENDER = scope:pantheon_torturer }
				}
				every_close_family_member = {
					limit = { pantheon_grievance_eligible_trigger = { GOD = root } }
					pantheon_give_torture_grievance_effect = { WHO = this OFFENDER = scope:pantheon_torturer }
				}
				every_spouse = {
					limit = { pantheon_grievance_eligible_trigger = { GOD = root } }
					pantheon_give_torture_grievance_effect = { WHO = this OFFENDER = scope:pantheon_torturer }
				}
			}
		}
	}
}

# Pick one follower owed vengeance (rulers first, then the god's courtiers) and send their Great Prayer.
# The suppliant is kept in a variable: the event reads it back.
# (1.5 review: never tests exists on the suppliant scope, which the prayer timer chain inherits.)
pantheon_pick_great_prayer_effect = {
	if = {
		limit = {
			any_ruler = {
				faith = root.faith
				pantheon_grievance_valid_trigger = { GOD = root }
				NOT = { var:pantheon_grievance_kind ?= 6 }
			}
		}
		random_ruler = {
			limit = {
				faith = root.faith
				pantheon_grievance_valid_trigger = { GOD = root }
				NOT = { var:pantheon_grievance_kind ?= 6 }
			}
			save_scope_as = pantheon_great_suppliant
		}
	}
	else = {
		random_courtier = {
			limit = {
				faith = root.faith
				pantheon_grievance_valid_trigger = { GOD = root }
				NOT = { var:pantheon_grievance_kind ?= 6 }
			}
			save_scope_as = pantheon_great_suppliant
		}
	}
	# Only send it if a fresh suppliant was saved this time (an inherited one would be stale): the caller
	# already checked pantheon_has_a_strong_great_prayer_trigger (2.4: a raid grievance, kind 6, is picked by
	# pantheon_pick_raid_great_prayer_effect below, never here), so one of the two branches above saved someone.
	set_variable = { name = pantheon_great_suppliant_var value = scope:pantheon_great_suppliant }
	trigger_event = pantheon.0065
}

# 2.4 (R7): the same pick for a raid grievance (kind 6) ONLY. Called by the scheduler last, when nothing was delivered.
pantheon_pick_raid_great_prayer_effect = {
	if = {
		limit = {
			any_ruler = {
				faith = root.faith
				pantheon_grievance_valid_trigger = { GOD = root }
				var:pantheon_grievance_kind ?= 6
			}
		}
		random_ruler = {
			limit = {
				faith = root.faith
				pantheon_grievance_valid_trigger = { GOD = root }
				var:pantheon_grievance_kind ?= 6
			}
			save_scope_as = pantheon_great_suppliant
		}
	}
	else = {
		random_courtier = {
			limit = {
				faith = root.faith
				pantheon_grievance_valid_trigger = { GOD = root }
				var:pantheon_grievance_kind ?= 6
			}
			save_scope_as = pantheon_great_suppliant
		}
	}
	set_variable = { name = pantheon_great_suppliant_var value = scope:pantheon_great_suppliant }
	trigger_event = pantheon.0065
}

# Swear vengeance: the sworn state lives on the god (it must survive the Judgement page chain), and Judgement
# opens at the offender with every wrath on offer (no sin-only limit; the cooldown is not re-checked there).
# 2.3 (R8, R17, R18): the wrong is SNAPSHOT at swearing (pantheon_great_prayer_kind 1-5, 0 = unknown) and so is whether it
# is harsh (pantheon_great_prayer_harsh, a torture: answered only by Strike down or Maim, user 2026-10-05) -- the page
# restrictions are derived by the page triggers from the page's saved context (pantheon_wrath_page_harsh_trigger), not stored.
pantheon_swear_vengeance_effect = {
	# 2.3 review fix: a live oath is never replaced. A Great Prayer event that was already open when another oath was sworn
	# (the option is greyed, but an event on screen is not re-evaluated) does nothing here; the prayer stays unanswered
	# (Refuse it, or swear once the first oath is over).
	if = {
		limit = { NOT = { has_variable = pantheon_great_prayer_active } }
		pantheon_end_oath_effect = yes
		set_variable = { name = pantheon_great_prayer_active value = yes }
		set_variable = { name = pantheon_great_prayer_suppliant value = scope:pantheon_great_suppliant }
		set_variable = { name = pantheon_great_prayer_offender value = scope:pantheon_great_offender }
		set_variable = { name = pantheon_great_prayer_kind value = 0 }
		if = {
			limit = { scope:pantheon_great_suppliant = { has_variable = pantheon_grievance_kind } }
			set_variable = { name = pantheon_great_prayer_kind value = scope:pantheon_great_suppliant.var:pantheon_grievance_kind }
		}
		if = {
			limit = { var:pantheon_great_prayer_kind ?= 5 }
			set_variable = { name = pantheon_great_prayer_harsh value = yes }
		}
		# 2.4 (R8): a raid grievance (kind 6) is a LIGHT oath: only Curse or Maim fulfils it
		if = {
			limit = { var:pantheon_great_prayer_kind ?= 6 }
			set_variable = { name = pantheon_great_prayer_light value = yes }
		}
		set_variable = { name = pantheon_wrath_target value = scope:pantheon_great_offender }
		pantheon_wrath_set_context_effect = { CTX = 1 }
		pantheon_wrath_save_ctx_effect = { CTX = 1 }
		trigger_event = pantheon.0070
		if = {
			limit = { exists = scope:pantheon_wrath_ctx }
			clear_saved_scope = pantheon_wrath_ctx
		}
	}
}

# 2.5 (Task 5): answer a grievance UNASKED -- the god swears vengeance for a follower who has not prayed for it (the Followers
# row button, pantheon_ui_row_answer). Root is the god, $WHO$ the follower. R8: a stale click does nothing: the grievance must
# still be valid with its offender alive, and no oath may be outstanding; checked BEFORE any scope is saved or the oath is sworn.
# The oath is flagged unasked only when it is this follower's (swear itself refuses a live oath); no timer, no cooldown.
pantheon_answer_unasked_effect = {
	if = {
		limit = {
			NOT = { has_variable = pantheon_great_prayer_active }
			$WHO$ = {
				pantheon_grievance_valid_trigger = { GOD = root }
				var:pantheon_grievance_target ?= { is_alive = yes }
			}
		}
		if = { limit = { exists = scope:pantheon_great_suppliant } clear_saved_scope = pantheon_great_suppliant }
		if = { limit = { exists = scope:pantheon_great_offender } clear_saved_scope = pantheon_great_offender }
		$WHO$ = { save_scope_as = pantheon_great_suppliant }
		$WHO$.var:pantheon_grievance_target ?= { save_scope_as = pantheon_great_offender }
		pantheon_swear_vengeance_effect = yes
		if = {
			limit = {
				has_variable = pantheon_great_prayer_active
				var:pantheon_great_prayer_suppliant ?= $WHO$
			}
			set_variable = { name = pantheon_great_prayer_unasked value = yes }
		}
		clear_saved_scope = pantheon_great_suppliant
		clear_saved_scope = pantheon_great_offender
	}
}

# 2.5 (Task 5, R7): backing out of an UNASKED oath (0070's Leave it for now / Stay your hand on its own page) costs nothing:
# no Devotion, no opinion, no cooldown, and the grievance stays. Root is the god.
pantheon_unasked_withdraw_effect = {
	pantheon_end_oath_effect = yes
}

# 2.3: the Prayers tab's "Open the Judgement" on the sworn oath: re-open the sworn page (pantheon.0070) on the offender in
# context 1. The page's option triggers derive the restrictions from its context and the oath, so another route cannot block it.
# Does nothing if the oath is gone or its offender is dead. Root is the god.
pantheon_oath_open_judgement_effect = {
	if = {
		limit = {
			has_variable = pantheon_great_prayer_active
			var:pantheon_great_prayer_offender ?= { is_alive = yes }
		}
		if = {
			limit = { exists = scope:pantheon_victim }
			clear_saved_scope = pantheon_victim
		}
		var:pantheon_great_prayer_offender ?= { save_scope_as = pantheon_victim }
		set_variable = { name = pantheon_wrath_target value = scope:pantheon_victim }
		pantheon_wrath_set_context_effect = { CTX = 1 }
		pantheon_wrath_save_ctx_effect = { CTX = 1 }
		trigger_event = pantheon.0070
		clear_saved_scope = pantheon_victim
		clear_saved_scope = pantheon_wrath_ctx
	}
}

# Refuse a Great Prayer: the Devotion loss scaled by the wrong (2.5: pantheon_great_prayer_refusal, -15 to -50, -30 when the
# kind is unknown) and the refused-prayer opinion. The value is read in the SUPPLIANT's scope (pantheon_change_devotion_effect
# evaluates $AMOUNT$ inside scope:pantheon_target) and BEFORE the grievance that names the kind is cleared. The applied
# amount is kept on the god for the toast (R5).
pantheon_refuse_great_prayer_effect = {
	if = {
		limit = { exists = scope:pantheon_target }
		clear_saved_scope = pantheon_target
	}
	scope:pantheon_great_suppliant = { save_scope_as = pantheon_target }
	pantheon_change_devotion_effect = { AMOUNT = pantheon_great_prayer_refusal }
	scope:pantheon_target = {
		set_variable = { name = pantheon_refusal_loss value = pantheon_great_prayer_refusal days = 1 }
	}
	if = {
		limit = { scope:pantheon_target = { has_variable = pantheon_refusal_loss } }
		set_variable = { name = pantheon_last_devotion_loss value = scope:pantheon_target.var:pantheon_refusal_loss days = 1 }
	}
	scope:pantheon_target = {
		add_opinion = {
			target = root
			modifier = pantheon_refused_prayer_opinion
			years = 5
		}
		pantheon_clear_grievance_effect = yes
	}
	set_variable = { name = pantheon_great_prayer_cooldown value = yes years = 2 }
	send_interface_toast = {
		type = event_toast_effect_bad
		title = pantheon_great_prayer_refused_t
		left_icon = scope:pantheon_target
	}
}

# Swore vengeance, then stayed the hand: a broken oath is worse than a refusal (user, 2026-10-05). 2.5: scaled by the wrong
# snapshotted at swearing (pantheon_oath_broken_loss, -30 to -90, -60 for an unknown kind or an oath sworn before 2.5).
# R4: the value is GOD-scoped, so it is captured here (before the oath ends) and handed on as a scope value, because
# pantheon_change_devotion_effect evaluates $AMOUNT$ inside the follower.
pantheon_broken_oath_effect = {
	if = {
		limit = { exists = scope:pantheon_target }
		clear_saved_scope = pantheon_target
	}
	var:pantheon_great_prayer_suppliant ?= { if = { limit = { is_alive = yes } save_scope_as = pantheon_target } }
	if = {
		limit = { var:pantheon_great_prayer_suppliant ?= { is_alive = yes } }
		save_temporary_scope_value_as = { name = pantheon_oath_loss value = pantheon_oath_broken_loss }
		set_variable = { name = pantheon_last_devotion_loss value = scope:pantheon_oath_loss days = 1 }
		pantheon_change_devotion_effect = { AMOUNT = scope:pantheon_oath_loss }
		scope:pantheon_target = {
			add_opinion = {
				target = root
				modifier = pantheon_refused_prayer_opinion
				years = 5
			}
			pantheon_clear_grievance_effect = yes
		}
		send_interface_toast = {
			type = event_toast_effect_bad
			title = pantheon_oath_broken_t
			left_icon = scope:pantheon_target
		}
	}
	set_variable = { name = pantheon_great_prayer_cooldown value = yes years = 2 }
	pantheon_end_oath_effect = yes
	pantheon_ui_refresh_if_open_effect = yes
}

# The oath is over: fulfilled, broken (Stay), released (the offender died) or replaced by a newly sworn one. 2.3 (R8): no
# Judgement, Lands or Retribution route ends it any more -- a wrath on someone else leaves it open.
pantheon_end_oath_effect = {
	if = { limit = { has_variable = pantheon_great_prayer_active } remove_variable = pantheon_great_prayer_active }
	if = { limit = { has_variable = pantheon_great_prayer_suppliant } remove_variable = pantheon_great_prayer_suppliant }
	if = { limit = { has_variable = pantheon_great_prayer_offender } remove_variable = pantheon_great_prayer_offender }
	if = { limit = { has_variable = pantheon_great_prayer_kind } remove_variable = pantheon_great_prayer_kind }
	if = { limit = { has_variable = pantheon_great_prayer_harsh } remove_variable = pantheon_great_prayer_harsh }
	if = { limit = { has_variable = pantheon_great_prayer_light } remove_variable = pantheon_great_prayer_light }
	if = { limit = { has_variable = pantheon_great_prayer_unasked } remove_variable = pantheon_great_prayer_unasked }
	if = { limit = { has_variable = pantheon_wrath_harsh } remove_variable = pantheon_wrath_harsh }
}

# 2.3 (R17): the sworn offender died by any other means than a wrath that already fulfilled the oath: it is released, with no
# penalty and no cooldown, and the god is told once. The suppliant's grievance, which named the dead, is forgotten.
# Root is the god. Called by the ungated on_death hook (pantheon_on_death_oath) and by the monthly safety net.
pantheon_oath_release_effect = {
	if = {
		limit = { has_variable = pantheon_great_prayer_active }
		save_temporary_scope_as = pantheon_oath_god
		if = {
			limit = { exists = scope:pantheon_target }
			clear_saved_scope = pantheon_target
		}
		var:pantheon_great_prayer_suppliant ?= { if = { limit = { is_alive = yes } save_scope_as = pantheon_target } }
		if = {
			limit = { exists = scope:pantheon_target }
			scope:pantheon_target = { pantheon_clear_grievance_effect = yes }
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = pantheon_oath_released_t
				left_icon = scope:pantheon_target
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = pantheon_oath_released_t
				left_icon = scope:pantheon_oath_god
			}
		}
		pantheon_end_oath_effect = yes
		pantheon_ui_refresh_if_open_effect = yes
	}
}

# 2.3: an oath sworn before 2.3 has no snapshot. Save it once: the wrong from the suppliant's grievance if it is still there
# (else 0 = unknown, shown as "the wrong done to them"), and harshness from the old page flag. Root is the god. Run by
# the monthly upkeep, the window rebuild, and every route that is about to clear the page flags.
pantheon_oath_migrate_effect = {
	if = {
		limit = {
			has_variable = pantheon_great_prayer_active
			NOT = { has_variable = pantheon_great_prayer_kind }
		}
		set_variable = { name = pantheon_great_prayer_kind value = 0 }
		if = {
			limit = { var:pantheon_great_prayer_suppliant ?= { has_variable = pantheon_grievance_kind } }
			set_variable = { name = pantheon_great_prayer_kind value = var:pantheon_great_prayer_suppliant.var:pantheon_grievance_kind }
		}
		if = {
			limit = {
				has_variable = pantheon_wrath_harsh
				NOT = { has_variable = pantheon_great_prayer_harsh }
			}
			set_variable = { name = pantheon_great_prayer_harsh value = yes }
		}
	}
}

# The god's monthly upkeep (pantheon.0011): migrate an old oath, and release one whose offender no longer exists (the
# safety net behind the on_death hook).
pantheon_oath_upkeep_effect = {
	pantheon_oath_migrate_effect = yes
	if = {
		limit = {
			has_variable = pantheon_great_prayer_active
			NOT = { var:pantheon_great_prayer_offender ?= { is_alive = yes } }
		}
		pantheon_oath_release_effect = yes
	}
}

# A sworn Great Prayer, answered (called from pantheon_wrath_landed_effect, after the 1.4 Devotion sweep and
# before the wrath itself). The suppliant to 100 and Avenged; a thank-offering of half the wrath's cost; the
# awe sweep a second time (awe x2) when it applied; then the 2-year cooldown and the oath ends.
pantheon_great_prayer_answered_effect = {
	if = {
		limit = { exists = scope:pantheon_target }
		clear_saved_scope = pantheon_target
	}
	var:pantheon_great_prayer_suppliant ?= { if = { limit = { is_alive = yes } save_scope_as = pantheon_target } }
	if = {
		limit = { var:pantheon_great_prayer_suppliant ?= { is_alive = yes } }
		pantheon_change_devotion_effect = { AMOUNT = 100 }
		# 2.5 (Task 5): an UNASKED oath is Foreseen, not Avenged (1.5x the values, 20 years), the thank-offering is the
		# wrath's full cost, the awe sweeps the faith as for an unasked Great Boon (10), and there is no cooldown.
		if = {
			limit = { has_variable = pantheon_great_prayer_unasked }
			scope:pantheon_target = {
				add_character_modifier = { modifier = pantheon_foreseen years = 20 }
				add_opinion = { target = root modifier = pantheon_foreseen_opinion years = 20 }
				pantheon_clear_grievance_effect = yes
			}
			# the awe effect reads scope:pantheon_boon_event_follower (the one it spares): saved for the call, cleared after
			if = { limit = { exists = scope:pantheon_boon_event_follower } clear_saved_scope = pantheon_boon_event_follower }
			scope:pantheon_target = { save_scope_as = pantheon_boon_event_follower }
			pantheon_miracle_awe_effect = { AMOUNT = 10 }
			clear_saved_scope = pantheon_boon_event_follower
		}
		else = {
			scope:pantheon_target = {
				add_character_modifier = { modifier = pantheon_avenged years = 10 }
				add_opinion = { target = root modifier = pantheon_avenged_opinion years = 10 }
				pantheon_clear_grievance_effect = yes
			}
		}
		# 2.2.1 (#4): the avenged suppliant shows as Blessed for the usual blessing period
		pantheon_mark_timed_blessing_effect = { KEY = prayer VALUE = yes }
		set_variable = { name = pantheon_offering value = var:pantheon_last_wrath_cost }
		if = {
			limit = { has_variable = pantheon_great_prayer_unasked }
			change_variable = { name = pantheon_offering multiply = 1 }
		}
		else = {
			change_variable = { name = pantheon_offering multiply = 0.5 }
		}
		change_variable = { name = pantheon_worship add = var:pantheon_offering }
		send_interface_toast = {
			type = event_toast_effect_good
			title = pantheon_great_prayer_answered_t
			left_icon = scope:pantheon_target
		}
	}
	if = {
		limit = {
			scope:pantheon_victim = {
				NOT = { faith = root.faith }
				NOT = { has_variable = pantheon_apostate }
			}
		}
		pantheon_wrath_awe_sweep_effect = yes
	}
	# the 2-year cooldown follows an ASKED prayer only (2.5: an unasked answer starts none)
	if = {
		limit = { NOT = { has_variable = pantheon_great_prayer_unasked } }
		set_variable = { name = pantheon_great_prayer_cooldown value = yes years = 2 }
	}
	pantheon_end_oath_effect = yes
	pantheon_ui_refresh_if_open_effect = yes
}

# One character of that walk (root is the god). 2.6 (R13): the same visit also backfills an old save's offender list from the
# follower's own grievance, and prunes the list a character keeps as an offender. No second world walk.
pantheon_grievance_upkeep_one_effect = {
	# 2.6 (R14, user 2026-10-26): a follower whose Devotion was never measured starts at 50 (the normal start), joins the god's
	# known list and gets the Follower trait. Same visit, no second world walk. Not the god, not a demigod (their own trait).
	if = {
		limit = {
			is_alive = yes
			faith = root.faith
			NOT = { this = root }
			NOT = { has_variable = pantheon_devotion }
			NOT = { has_trait = pantheon_demigod }
		}
		set_variable = { name = pantheon_devotion value = 50 }
		save_temporary_scope_as = pantheon_glance_new
		root = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = pantheon_known target = scope:pantheon_glance_new } } }
				add_to_variable_list = { name = pantheon_known target = scope:pantheon_glance_new }
			}
		}
		pantheon_glance_refresh_effect = { GOD = root }
	}
	# 2.6 repair 1 (UI test FT7): a follower with Devotion but no trait (left the faith and came back, and in neither the
	# marked nor the known list) gets it back here, the only walk that reaches them.
	else_if = {
		limit = {
			is_alive = yes
			faith = root.faith
			NOT = { this = root }
			has_variable = pantheon_devotion
			NOT = { has_trait = pantheon_follower }
			NOT = { has_trait = pantheon_demigod }
		}
		pantheon_glance_refresh_effect = { GOD = root }
	}
	if = {
		limit = {
			has_variable = pantheon_grievance
			NOT = { var:pantheon_grievance_target ?= { is_alive = yes } }
			NOT = { root = { var:pantheon_great_prayer_suppliant ?= prev } }
		}
		pantheon_clear_grievance_effect = yes
	}
	if = {
		limit = { has_variable = pantheon_grievance }
		pantheon_wronged_register_effect = yes
	}
	if = {
		limit = { has_variable_list = pantheon_wronged_by_list }
		pantheon_wronged_prune_effect = yes
	}
}
