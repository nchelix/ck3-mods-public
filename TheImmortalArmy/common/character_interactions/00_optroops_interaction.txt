# Immortal Army interactions. Everything but the menu toggle is hidden until
# the menu is opened, so a right-click stays short.
#
# Strength levels: the player picks Godlike (default, no flag), Elite or Regular.
# The level is stored as a character flag on the actor and only affects new
# spawns; armies already raised keep the regiment type they were made with.

open_optroop_menu = {
	category = interaction_category_diplomacy
	interface_priority = 70
	icon = icon_combat
	desc = open_optroop_menu_desc
	common_interaction = yes

	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			NOT = {
				has_character_flag = open_optroop_menu
			}
		}
	}

	on_send = {
		scope:actor = {
			add_character_flag = open_optroop_menu
		}
	}
}

close_optroop_menu = {
	category = interaction_category_diplomacy
	interface_priority = 69
	icon = icon_combat
	desc = close_optroop_menu_desc
	common_interaction = yes

	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			has_character_flag = open_optroop_menu
		}
	}

	on_send = {
		scope:actor = {
			remove_character_flag = open_optroop_menu
		}
	}
}

optroop_strength_godlike_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 68
	icon = icon_combat
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	desc = optroop_strength_godlike_interaction_desc

	is_shown = {
		scope:actor = {
			has_character_flag = open_optroop_menu
		}
	}

	is_valid = {
		custom_tooltip = {
			text = optroop_strength_already_set
			scope:actor = {
				OR = {
					has_character_flag = optroops_level_elite
					has_character_flag = optroops_level_regular
				}
			}
		}
	}

	on_accept = {
		scope:actor = {
			remove_character_flag = optroops_level_elite
			remove_character_flag = optroops_level_regular
		}
	}

	auto_accept = yes
}

inheritable_optroop_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 65
	icon = request_maa_interaction
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	desc = inheritable_optroop_interaction_desc

	is_shown = {
		scope:actor = {
			has_character_flag = open_optroop_menu
		}
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { scope:actor = { has_character_flag = optroops_level_elite } }
				spawn_army = {
					levies = 0
					men_at_arms = {
						type = op_troops_elite
						stacks = 1
					}
					location = scope:recipient.capital_province
					inheritable = yes
					uses_supply = no
					name = op_troops_elite
				}
			}
			else_if = {
				limit = { scope:actor = { has_character_flag = optroops_level_regular } }
				spawn_army = {
					levies = 0
					men_at_arms = {
						type = op_troops_regular
						stacks = 1
					}
					location = scope:recipient.capital_province
					inheritable = yes
					uses_supply = no
					name = op_troops_regular
				}
			}
			else = {
				spawn_army = {
					levies = 0
					men_at_arms = {
						type = op_troops
						stacks = 1
					}
					location = scope:recipient.capital_province
					inheritable = yes
					uses_supply = no
					name = op_troops
				}
			}
		}
	}

	auto_accept = yes
}
