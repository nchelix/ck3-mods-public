# Immortal Army interactions.
#
# 1.4: two radio popups replace the eleven entries of 1.3 (the menu toggle, three
# strength setters and six spawners). CK3 makes a whole popup either radios or
# checkboxes, so strength has its own popup rather than sharing one with the
# troop choice.
#
# Strength levels: Godlike (default, no flag), Elite or Regular. The level is
# stored as a character flag on the actor and only affects new spawns; armies
# already raised keep the regiment type they were made with.
#
# 1.3 kept its entries behind a menu flag. Nothing reads it any more, so each
# on_accept clears it from old saves. The add is deliberate: the game logs an
# error at load for a flag that is used but never set anywhere in script.

optroop_raise_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 70
	icon = optroop_raise_interaction
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	desc = optroop_raise_interaction_desc

	# The army gathers at the recipient's capital, so a landless recipient would
	# get nothing (and an Invalid location error).
	is_shown = {
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			exists = capital_province
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = optroop_raise_choose_tt
			OR = {
				scope:optroop_raise_infantry_inheritable ?= yes
				scope:optroop_raise_infantry_personal ?= yes
				scope:optroop_raise_cavalry_inheritable ?= yes
				scope:optroop_raise_cavalry_personal ?= yes
				scope:optroop_raise_archers_inheritable ?= yes
				scope:optroop_raise_archers_personal ?= yes
			}
		}
	}

	options_heading = optroop_raise_heading
	send_options_exclusive = yes

	send_option = {
		flag = optroop_raise_infantry_inheritable
		localization = optroop_raise_infantry_inheritable
		current_description = optroop_raise_infantry_inheritable_desc
		starts_enabled = {
			always = yes
		}
	}

	send_option = {
		flag = optroop_raise_infantry_personal
		localization = optroop_raise_infantry_personal
		current_description = optroop_raise_infantry_personal_desc
	}

	send_option = {
		flag = optroop_raise_cavalry_inheritable
		localization = optroop_raise_cavalry_inheritable
		current_description = optroop_raise_cavalry_inheritable_desc
	}

	send_option = {
		flag = optroop_raise_cavalry_personal
		localization = optroop_raise_cavalry_personal
		current_description = optroop_raise_cavalry_personal_desc
	}

	send_option = {
		flag = optroop_raise_archers_inheritable
		localization = optroop_raise_archers_inheritable
		current_description = optroop_raise_archers_inheritable_desc
	}

	send_option = {
		flag = optroop_raise_archers_personal
		localization = optroop_raise_archers_personal
		current_description = optroop_raise_archers_personal_desc
	}

	on_accept = {
		scope:actor = {
			hidden_effect = {
				add_character_flag = open_optroop_menu
				remove_character_flag = open_optroop_menu
			}
			if = {
				limit = { has_character_flag = optroops_level_elite }
				custom_tooltip = optroop_raise_at_elite_tt
			}
			else_if = {
				limit = { has_character_flag = optroops_level_regular }
				custom_tooltip = optroop_raise_at_regular_tt
			}
			else = {
				custom_tooltip = optroop_raise_at_godlike_tt
			}
		}
		if = {
			limit = { scope:optroop_raise_infantry_inheritable ?= yes }
			scope:recipient = {
				optroops_spawn_effect = { GODLIKE = op_troops ELITE = op_troops_elite REGULAR = op_troops_regular INHERITABLE = yes }
			}
		}
		else_if = {
			limit = { scope:optroop_raise_infantry_personal ?= yes }
			scope:recipient = {
				optroops_spawn_effect = { GODLIKE = op_troops ELITE = op_troops_elite REGULAR = op_troops_regular INHERITABLE = no }
			}
		}
		else_if = {
			limit = { scope:optroop_raise_cavalry_inheritable ?= yes }
			scope:recipient = {
				optroops_spawn_effect = { GODLIKE = op_troops_cavalry ELITE = op_troops_cavalry_elite REGULAR = op_troops_cavalry_regular INHERITABLE = yes }
			}
		}
		else_if = {
			limit = { scope:optroop_raise_cavalry_personal ?= yes }
			scope:recipient = {
				optroops_spawn_effect = { GODLIKE = op_troops_cavalry ELITE = op_troops_cavalry_elite REGULAR = op_troops_cavalry_regular INHERITABLE = no }
			}
		}
		else_if = {
			limit = { scope:optroop_raise_archers_inheritable ?= yes }
			scope:recipient = {
				optroops_spawn_effect = { GODLIKE = op_troops_archers ELITE = op_troops_archers_elite REGULAR = op_troops_archers_regular INHERITABLE = yes }
			}
		}
		else_if = {
			limit = { scope:optroop_raise_archers_personal ?= yes }
			scope:recipient = {
				optroops_spawn_effect = { GODLIKE = op_troops_archers ELITE = op_troops_archers_elite REGULAR = op_troops_archers_regular INHERITABLE = no }
			}
		}
	}

	auto_accept = yes
}

optroop_strength_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 69
	icon = optroop_strength_interaction
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	desc = optroop_strength_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
		}
	}

	# No validity check on the chosen row: the right-click menu runs it against the
	# preselected row (the current level), so rejecting the current level greyed
	# out the whole entry. Sending the current level just keeps it.

	options_heading = optroop_strength_heading
	send_options_exclusive = yes

	send_option = {
		flag = optroop_strength_godlike
		localization = optroop_strength_godlike
		current_description = optroop_strength_godlike_desc
		starts_enabled = {
			scope:actor = {
				NOR = {
					has_character_flag = optroops_level_elite
					has_character_flag = optroops_level_regular
				}
			}
		}
	}

	send_option = {
		flag = optroop_strength_elite
		localization = optroop_strength_elite
		current_description = optroop_strength_elite_desc
		starts_enabled = {
			scope:actor = {
				has_character_flag = optroops_level_elite
			}
		}
	}

	send_option = {
		flag = optroop_strength_regular
		localization = optroop_strength_regular
		current_description = optroop_strength_regular_desc
		starts_enabled = {
			scope:actor = {
				has_character_flag = optroops_level_regular
				NOT = { has_character_flag = optroops_level_elite }
			}
		}
	}

	on_accept = {
		scope:actor = {
			hidden_effect = {
				add_character_flag = open_optroop_menu
				remove_character_flag = open_optroop_menu
			}
			if = {
				limit = { scope:optroop_strength_godlike ?= yes }
				custom_tooltip = optroop_strength_set_godlike_tt
				remove_character_flag = optroops_level_elite
				remove_character_flag = optroops_level_regular
			}
			else_if = {
				limit = { scope:optroop_strength_elite ?= yes }
				custom_tooltip = optroop_strength_set_elite_tt
				remove_character_flag = optroops_level_regular
				add_character_flag = optroops_level_elite
			}
			else_if = {
				limit = { scope:optroop_strength_regular ?= yes }
				custom_tooltip = optroop_strength_set_regular_tt
				remove_character_flag = optroops_level_elite
				add_character_flag = optroops_level_regular
			}
		}
	}

	auto_accept = yes
}
