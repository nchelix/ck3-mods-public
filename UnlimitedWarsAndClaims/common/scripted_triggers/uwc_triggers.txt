# Unlimited Wars and Claims 1.0: all the rule logic. Character scope = the attacker.
uwc_shared_titles_trigger = {
	OR = {
		has_game_rule = uwc_shared_titles_everyone
		AND = {
			has_game_rule = uwc_shared_titles_players
			is_ai = no
		}
	}
}
