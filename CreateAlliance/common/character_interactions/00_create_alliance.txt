# Create Alliance
#
# Instant, unconditional alliances. The base game has had its own negotiated
# alliance interaction since this mod was written, so this one leans into what
# vanilla will not do: no reason required, no acceptance roll, no distance, and
# no requirement that either party hold land.
#
# The alliance is a real one. It is held together by perk_negotiated_alliance_opinion,
# which the game's own is_alliance_valid rule counts as a qualifying bond -- and
# that rule has no landed or ruler requirement, which is why allying an unlanded
# adventurer works and persists.
#
# 1.3: Isolate and Remove only act on alliances that a pact alone holds together
# (create_alliance_rests_on_pact_only_trigger). Anything else is held by a bond
# the game re-checks, so the click would do nothing or damage that bond.

create_alliance_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 60
	icon = create_alliance_interaction
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = create_alliance_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
			NOR = {
				is_allied_to = scope:recipient
				this = scope:recipient
			}
		}
	}

	# is_alliance_valid opens with NOT has_title = title:h_china on the first
	# character of the alliance, and this interaction makes the player that
	# character, so the game would dissolve the alliance straight away.
	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = create_alliance_hegemon_tt
			scope:actor = {
				NOT = { has_title = title:h_china }
			}
		}
	}

	on_accept = {
		scope:recipient = {
			# This opinion modifier is what keeps the alliance valid; removing it
			# is how the alliance is later dissolved.
			add_opinion = {
				modifier = perk_negotiated_alliance_opinion
				target = scope:actor
			}
		}
		scope:actor = {
			allow_alliance = scope:recipient
			create_alliance = {
				target = scope:recipient
				allied_through_owner = scope:actor
				allied_through_target = scope:recipient
			}
		}
	}

	auto_accept = yes
	ai_will_do = 0
}

inherit_alliances_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 59
	icon = inherit_alliances_interaction
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = inherit_alliances_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
			NOT = { this = scope:recipient }
		}
		scope:recipient = {
			any_ally = {
				NOR = {
					this = scope:actor
					is_allied_to = scope:actor
				}
			}
		}
	}

	# The same h_china rule as Create Alliance: the player is the alliance's first character.
	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = create_alliance_hegemon_tt
			scope:actor = {
				NOT = { has_title = title:h_china }
			}
		}
	}

	on_accept = {
		scope:recipient = {
			every_ally = {
				limit = {
					NOR = {
						this = scope:actor
						is_allied_to = scope:actor
					}
				}
				save_scope_as = inherited_ally
				add_opinion = {
					modifier = perk_negotiated_alliance_opinion
					target = scope:actor
				}
				scope:actor = {
					allow_alliance = scope:inherited_ally
					create_alliance = {
						target = scope:inherited_ally
						allied_through_owner = scope:actor
						allied_through_target = scope:inherited_ally
					}
				}
			}
		}
	}

	auto_accept = yes
	ai_will_do = 0
}

isolate_character_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 58
	icon = isolate_character_interaction
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = isolate_character_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			any_ally = { always = yes }
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = isolate_no_pact_ally_tt
			scope:recipient = {
				any_ally = {
					create_alliance_rests_on_pact_only_trigger = { OTHER = scope:recipient }
				}
			}
		}
	}

	on_accept = {
		scope:recipient = {
			every_ally = {
				# break_alliance runs vanilla's on_alliance_broken, which also clears the
				# pact opinion both ways, ends a blood brotherhood and cancels a betrothal
				# behind the alliance. The limit keeps it away from those.
				limit = {
					create_alliance_rests_on_pact_only_trigger = { OTHER = scope:recipient }
				}
				break_alliance = scope:recipient
			}
		}
	}

	auto_accept = yes
	ai_will_do = 0
}

break_alliance_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	interface_priority = 57
	icon = break_alliance_interaction
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = break_alliance_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			is_allied_to = scope:actor
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = break_alliance_bound_tt
			scope:recipient = {
				create_alliance_rests_on_pact_only_trigger = { OTHER = scope:actor }
			}
		}
	}

	on_accept = {
		# Either side may have made the pact, so clear it both ways. With no bond
		# left, is_alliance_valid fails and the game ends the alliance the next day.
		scope:recipient = {
			try_remove_perk_alliance_opinion_effect = { TARGET = scope:actor }
		}
		scope:actor = {
			try_remove_perk_alliance_opinion_effect = { TARGET = scope:recipient }
		}
	}

	auto_accept = yes
	ai_will_do = 0
}
