# Become the Snake interactions.
# The three looks are mutually exclusive; No More Snake clears whichever is present.

become_scaly_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	interface_priority = 64
	icon = icon_personal

	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = become_scaly_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			NOR = {
				has_trait = fake_scaly
				has_trait = fake_scaly_serpent
				has_trait = fake_scaly_bloodline
			}
		}
	}

	on_accept = {
		scope:recipient = {
			add_trait = fake_scaly
		}
	}

	auto_accept = yes
}

scaly_children_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	interface_priority = 61
	icon = icon_personal

	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = scaly_children_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			any_child = {
				NOR = {
					has_trait = fake_scaly
					has_trait = fake_scaly_serpent
					has_trait = fake_scaly_bloodline
				}
			}
		}
	}

	on_accept = {
		scope:recipient = {
			every_child = {
				limit = {
					NOR = {
						has_trait = fake_scaly
						has_trait = fake_scaly_serpent
						has_trait = fake_scaly_bloodline
					}
				}
				add_trait = fake_scaly_bloodline
			}
		}
	}

	auto_accept = yes
}

remove_scaly_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	interface_priority = 60
	icon = icon_personal

	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = remove_scaly_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			OR = {
				has_trait = fake_scaly
				has_trait = fake_scaly_serpent
				has_trait = fake_scaly_bloodline
			}
		}
	}

	on_accept = {
		scope:recipient = {
			remove_trait = fake_scaly
			remove_trait = fake_scaly_serpent
			remove_trait = fake_scaly_bloodline
		}
	}

	auto_accept = yes
}
