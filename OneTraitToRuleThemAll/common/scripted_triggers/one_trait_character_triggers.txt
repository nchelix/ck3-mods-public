# One Trait 2.8: the Character rows' hand-written triggers. For an action key K (tools/build_one_trait_panel.py,
# CHARACTER_ACTIONS): one_trait_character_K_shown_trigger and one_trait_character_K_valid_trigger. Spec:
# docs/superpowers/specs/2026-10-17-one-trait-2.8-character-pages-design.md section 2.
#
# Scope rule (plan "Names used across tasks" + R5): every trigger here runs IN the panel character's scope --
# the generated scripted GUI calls it as "var:one_trait_ui_target = { one_trait_character_K_... = yes }" --
# so the retired menu option's own "scope:one_trait_target" becomes "this" (left bare), and "root" is still
# the player. Shown = a row that can never apply is really hidden (the action row's outer visible); valid =
# greyed with CK3's own reason on hover, when the row could apply but not now.
#
# Each row carries the retired option's conditions, split per the spec: shown holds what can never apply
# (yourself, a missing relative), valid holds what could apply but not now (greyed with CK3's own reason).
# A missing relative is genuinely false in shown by explicit existence, never a bare ?= chain (R6).

one_trait_character_family_heir_shown_trigger = {
	NOT = { this = root }
}
# Playtest 2.8 fix 1: set_designated_heir only accepts a direct child (docs/reference/ck3-1.20.1.1-effects.log:10217),
# so a non-child must grey with a readable reason. Both clauses are custom_tooltip: CK3 prints nothing for a
# bare scope comparison.
one_trait_character_family_heir_valid_trigger = {
	custom_tooltip = {
		text = one_trait_character_own_child_tt
		is_child_of = root
	}
	custom_tooltip = {
		text = one_trait_character_not_heir_yet_tt
		NOT = { root.designated_heir ?= this }
	}
	# C-fix3 (2.13 phase C2 error.log: set_designated_heir "Trying to assign '<daughter>' as designated heir to
	# '<king>', but they cannot inherit from them"): the child must be able to inherit. Vanilla 1.21's own
	# designate_heir_interaction (game/common/character_interactions/00_heir.txt:136-149) refuses a recipient who is a
	# eunuch, disinherited, adopted_no_inheritance, devoted, an order_member or a bastard, or outside the actor's
	# house; vanilla's own scripted set_designated_heir (hold_court.4001, game/events/activities/hold_court_activity/
	# hold_court_events_general.txt:3631-3639 and :3757) also requires character_gender_can_inherit_from_trigger =
	# { CHARACTER = root } (game/common/scripted_triggers/00_law_triggers.txt:32: the primary title's gender law).
	custom_tooltip = {
		text = one_trait_character_heir_can_inherit_tt
		is_eunuch_trigger = no
		NOR = {
			has_trait = disinherited
			has_trait = adopted_no_inheritance
			has_trait = devoted
			has_trait = order_member
			has_trait = bastard
		}
		house ?= root.house
	}
	custom_tooltip = {
		text = one_trait_character_heir_gender_law_tt
		character_gender_can_inherit_from_trigger = { CHARACTER = root }
	}
	# C-fix5 (2.13 phase C2 re-run 2, self-test s13c1: the row was valid for Boyan, an adult unlanded child of your
	# house with no blocking trait, yet set_designated_heir was a silent no-op, with no error.log line): ask the engine
	# itself. is_valid_designated_heir, docs/reference/ck3-1.21.0-triggers.log:8906-8910 ("can they inherit one or more
	# of the titles the scoped character holds?", character scope, character target). Vanilla 1.21 uses it in the
	# same form, scope:puppet_or_actor = { is_valid_designated_heir = scope:recipient }
	# (game/common/character_interactions/pam_interactions.txt:105-107, the line itself at pam_interactions.txt:106).
	# Here the scoped character is you (root) and the target is this child (prev: the scope outside root = { }).
	custom_tooltip = {
		text = one_trait_character_heir_engine_tt
		root = { is_valid_designated_heir = prev }
	}
}
one_trait_character_family_adopt_shown_trigger = {
	NOT = { this = root }
}
# C-fix3 (2.13 phase C2 error.log: set_father "[ Parent can't be younger than their child ]"): you become their
# parent, so they must be younger than you. age: docs/reference/ck3-1.21.0-triggers.log:3647; comparing one
# character's age with another's is vanilla's own form (age > scope:recipient.age,
# game/common/character_interactions/06_ep3_interactions.txt:4557). Vanilla's adopt_interaction keeps the adoptee
# a child and the adopter an adult (00_adoption.txt:12, 137), so it never meets the case; whole years are compared,
# so a strictly smaller age is always a later birth.
one_trait_character_family_adopt_valid_trigger = {
	NOT = { is_child_of = root }
	custom_tooltip = {
		text = one_trait_character_younger_than_you_tt
		age < root.age
	}
}
