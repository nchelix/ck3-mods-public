# One Trait 2.9: the Diplomacy, Court and puppet rows' hand-written triggers. For an action key K of area A
# (tools/build_one_trait_panel.py): one_trait_A_K_shown_trigger and one_trait_A_K_valid_trigger (Court rows are
# area "realm": one_trait_realm_court_*; their effects live in one_trait_summon_effects.txt). Spec:
# docs/superpowers/specs/2026-10-19-one-trait-2.9-diplomacy-summon-design.md section 2.
#
# Scope rule: every trigger runs IN the panel character's scope (the generated scripted GUI calls it as
# "var:one_trait_ui_target = { one_trait_A_K_... = yes }"), root = the player. A condition on the player is
# written "root = { ... }" with the panel's character as prev. A trigger with no entry in the game's
# common/trigger_localization/ gets its own custom_tooltip reason (the reasons table, DIPLOMACY_REASONS).
# Batch 1 made relations_friend, court_chancellor and puppets_secular real; batch 2 made every other key real.
# Every Diplomacy row is SHOWN always (the category's lock covers yourself and dead targets) and VALID by the
# conditions below, each failing condition with a readable reason: a trigger with its own vanilla
# trigger_localization entry stands alone (is_landed, is_independent_ruler, is_at_war, is_imprisoned, has_trait,
# is_clergy: game/common/trigger_localization/00_character_triggers.txt:27, 362, 463, 351, 293, 1717); anything
# relational to the player (root), a flag, or a scope comparison sits in a custom_tooltip with its own reason key.

# ---- Real (batch 1)
one_trait_diplomacy_relations_friend_shown_trigger = { always = yes }
# Vanilla's own guard for set_relation_friend: game/common/scripted_triggers/00_relation_triggers.txt:14-22
# (not themselves, and no friend, best friend, rival or nemesis relation yet). It mixes triggers with and
# without trigger_localization, so one reason text covers it.
one_trait_diplomacy_relations_friend_valid_trigger = {
	custom_tooltip = {
		text = one_trait_diplomacy_relations_friend_tt
		root = { can_set_relation_friend_trigger = { CHARACTER = prev } }
	}
}

# Court rows. The five councillor rows seat the summoned character on the court owner's council (this = the panel's
# character), so they are VALID only when that seat can exist (Codex review #1 finding 1; the engine refuses the
# assignment otherwise and the row would promise a seat that never comes):
# 1. The owner's government has a council. The governments with council = no, in
#    game/common/governments/00_government_types.txt: mercenary_government (:330, flag government_is_mercenary :349),
#    holy_order_government (:367, government_is_holy_order :389), monastic_holy_order_government (:403, the same flag
#    :432), landless_adventurer_government (:568, government_is_landless_adventurer :616) and herder_government (:743,
#    government_is_herder :778).
# 2. The position's own valid_position (game/common/council_positions/00_council_positions.txt): chancellor :53-58,
#    steward :203-208, marshal :339-344 and court chaplain :823-828 exclude government_is_landless_adventurer and
#    government_is_nomadic; spymaster :481-483 excludes only government_is_landless_adventurer.
# 0. The owner is a playable character (count tier or above). A baron has_council = yes but is not playable and has no
#    seat to fill: assign_councillor_type then logs "doesn't have an open council position" (batch 6 F1, self-test
#    s21c4). It is the FIRST condition of each councillor row.
# government_has_flag has trigger_localization of its own, but it would read as a bare flag name, so each guard sits in
# a custom_tooltip with a plain reason. Champion, spouse and ward take no seat and stay always valid, as does every
# shown trigger: Realm's lock (the target is a landed ruler) hides the rest.

		text = one_trait_diplomacy_court_has_council_tt

			government_has_flag = government_is_mercenary
			government_has_flag = government_is_holy_order

			government_has_flag = government_is_herder
		}
	}
	custom_tooltip = {
		text = one_trait_diplomacy_court_seat_open_tt
		NOR = {
			government_has_flag = government_is_landless_adventurer

		NOT = { puppeteer ?= root }
	}

one_trait_diplomacy_alliances_isolate_shown_trigger = { always = yes }
# Changed (batch 7 U2): they need an ally whose alliance rests on a pact alone; marriage and family alliances are
# kept by vanilla's is_alliance_valid, so an isolate that could only reach those would do nothing. In any_ally the
# scope is the ally and prev is the panel's character.
one_trait_diplomacy_alliances_isolate_valid_trigger = {
	custom_tooltip = {
		text = one_trait_diplomacy_isolate_pact_ally_tt
		any_ally = { one_trait_alliance_rests_on_pact_only_trigger = { OTHER = prev } }
	}
}

# Added (batch 7 U2, checklist #40): does the alliance between THIS character and $OTHER$ rest on a pact alone?
# True when none of the reasons vanilla's is_alliance_valid keeps an alliance for holds
# (game/common/scripted_rules/00_rules.txt:376-432: marriage, betrothal, close family, blood brotherhood, the
# event-negotiated opinion either way, the childhood hostage oath of friendship, :411-428, and the FP3 loyalist
# flag on either character, :430-431). Evaluated in one participant's scope with the other passed as OTHER, and
# symmetric over the two directions. The perk-negotiated opinion is NOT here: break_alliance removes it both ways
# (game/common/on_action/alliance_on_actions.txt:296-305, on_alliance_broken), so it never stops an isolate; the
# end row's effect removes only their side, so that row adds the other side itself.
# Used by alliances_end (OTHER = root) and alliances_isolate (OTHER = prev inside any_ally).
# Scopes (batch 8, Codex review #2): inside any_memory the scope is the memory and prev is its OWNER, never the
# other participant, and $OTHER$ may itself expand to prev; so the oath branch first saves both characters
# (one_trait_pact_self / one_trait_pact_other) and the memory checks name those saved scopes. The only other prev
# is in the event-negotiated opinion branch: it sits in a plain $OTHER$ = { ... } block, not in any_memory, so
# prev is the scope before the block, this character, whether OTHER is root or prev.
one_trait_alliance_rests_on_pact_only_trigger = {
	NOR = {
		is_spouse_of = $OTHER$
		betrothed = $OTHER$
		is_parent_of = $OTHER$
		is_child_of = $OTHER$
		is_sibling_of = $OTHER$
		is_grandparent_of = $OTHER$
		is_great_grandparent_of = $OTHER$
		is_grandchild_of = $OTHER$
		is_great_grandchild_of = $OTHER$
		is_uncle_or_aunt_of = $OTHER$
		is_nibling_of = $OTHER$
		has_relation_blood_brother = $OTHER$
		has_opinion_modifier = {
			modifier = event_negotiated_alliance_opinion
			target = $OTHER$
		}
		$OTHER$ = {
			has_opinion_modifier = {
				modifier = event_negotiated_alliance_opinion
				target = prev
			}
		}
		AND = {
			save_temporary_scope_as = one_trait_pact_self
			$OTHER$ = { save_temporary_scope_as = one_trait_pact_other }
			OR = {
				has_relation_friend = $OTHER$
				has_relation_best_friend = $OTHER$
			}
			OR = {
				any_memory = {
					memory_type = hostage_oath_of_friendship
					has_memory_participant = scope:one_trait_pact_other
				}
				scope:one_trait_pact_other = {
					any_memory = {
						memory_type = hostage_oath_of_friendship
						has_memory_participant = scope:one_trait_pact_self
					}
				}
			}
		}
		has_character_flag = fp3_is_a_loyalist_puppet
		$OTHER$ = { has_character_flag = fp3_is_a_loyalist_puppet }
	}
}

# Old: allied to them, and the one-day "ending" flag not raised. Added: the alliance ends by losing the
# negotiated-alliance opinion, so it must not rest on anything else vanilla's is_alliance_valid accepts
# (game/common/scripted_rules/00_rules.txt:379-432: marriage, betrothal, close family, blood brotherhood, the
# event-negotiated opinion either way, or the same perk opinion held by you on them; the childhood hostage oath of
# friendship, :411-428: friend or best friend plus a hostage_oath_of_friendship memory in either direction; and the
# FP3 loyalist flag on either character, :430-431): otherwise the click would do nothing and the alliance would
# stay. Scopes: this = the panel's character, root = the player; vanilla's root / scope:second are checked in both
# directions by the same symmetric expression below.
one_trait_diplomacy_alliances_end_shown_trigger = { always = yes }
one_trait_diplomacy_alliances_end_valid_trigger = {
	custom_tooltip = {
		text = one_trait_diplomacy_allied_with_you_tt
		root = { is_allied_to = prev }
	}
	custom_tooltip = {
		text = one_trait_diplomacy_alliance_pact_only_tt
		one_trait_alliance_rests_on_pact_only_trigger = { OTHER = root }
		NOT = {
			root = {
				has_opinion_modifier = {
					modifier = perk_negotiated_alliance_opinion
					target = prev
				}
			}
		}
	}
	custom_tooltip = {
		text = one_trait_diplomacy_alliance_not_ending_tt
		NOT = { has_character_flag = one_trait_ending_alliance }
	}
}

# Changed (batch 7 U1, checklist #29): is_foreign_court_guest_of matches only guests that have a liege elsewhere
# (docs/reference/ck3-1.20.1.1-triggers.log:5427); an ordinary guest is a pool guest (is_pool_guest_of, :5836).
# is_foreign_court_or_pool_guest_of (:5439) covers both, and remove_courtier_or_guest (effects.log:9532) removes either.

	is_landed = no
	custom_tooltip = {
		text = one_trait_diplomacy_at_your_court_tt

			is_foreign_court_or_pool_guest_of = root
		}
	}
}
