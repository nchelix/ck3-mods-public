# One Trait 2.7 Task 2 (batch-2 handoff): the Realm tab's hand-written triggers. For an action key K:
# one_trait_realm_K_shown_trigger and one_trait_realm_K_valid_trigger. For a list key L: ONE
# one_trait_realm_L_shown_trigger (shared by every button of that list -- "is the list usable at all"),
# plus one one_trait_realm_L_<button>_valid_trigger per button. Spec:
# docs/superpowers/specs/2026-10-14-one-trait-2.7-realm-pages-design.md section 2.
#
# Scope rule (plan "Names used across tasks" + R1): every trigger here runs IN the panel character's
# scope -- the generated scripted GUI calls it as "var:one_trait_ui_target = { one_trait_realm_K_... =
# yes }" (action rows) or, for a list button's valid trigger, the same call with scope:landed_title also
# set (from the row, via AddScope). So a moved option's own "scope:one_trait_target" becomes "this" (left
# bare) and "root" still means the player.
#
# Succession's six (batch-2, Task 2): shown = "is_ruler = yes" (R1: "when in doubt, the condition goes into
# valid, not shown" -- none of the six original conditions are categorically never-true for a ruler, so
# nothing else joins shown); valid = the retired option's own `trigger` (one_trait_menu_events.txt:1221-
# 1314), re-scoped (bare already, since none referenced scope:one_trait_target or root).
#
# Titles, Culture, Buildings, Realm events and the three lists (Batch 3, Task 3): every remaining
# shown/valid trigger below is real; each row's split between the two is explained where it appears.

# ---- Culture (menu 0430), real this batch ----
# innovations_era/innovations_all carried no trigger at all in the menu: always shown and valid.
# acceptance's original trigger (one_trait_menu_events.txt:1902-1905) required
# "scope:one_trait_target != root AND scope:one_trait_target.culture != root.culture"; since a character's
# own culture always equals itself, the second half alone (re-scoped: this.culture != root.culture) already
# implies the first, so the whole condition collapses into the culture-mismatch shown check with nothing
# left for valid.

one_trait_realm_culture_innovations_era_shown_trigger = { always = yes }
one_trait_realm_culture_innovations_era_valid_trigger = { always = yes }

one_trait_realm_culture_innovations_all_shown_trigger = { always = yes }
one_trait_realm_culture_innovations_all_valid_trigger = { always = yes }

one_trait_realm_culture_acceptance_shown_trigger = {
	culture != root.culture
}
one_trait_realm_culture_acceptance_valid_trigger = { always = yes }

one_trait_realm_buildings_great_projects_shown_trigger = { always = yes }
# UI checklist 2026-10-15 row 22: greyed unless they own a great project. No phase test: "completed" is the
# funding phase finished, not the building (a project "Completed in 4 years" tested completed; vanilla
# reads in_progress OR completed as still standing: common/scripted_triggers/tgp_mandala_triggers.txt:252-255).
# An unstarted project has no owner, so owning one means one is under way. The plain
# reason on hover is a custom_tooltip, not custom_description: the latter names a common/trigger_localization
# entry, so with a plain loc key its tooltip came out empty (checklist row 28). Form:
# common/scripted_guis/ep2_activities.txt:27-30.
one_trait_realm_buildings_great_projects_valid_trigger = {
	is_landed = yes
	custom_tooltip = {
		text = one_trait_realm_great_project_under_way_tt
		any_great_project = { great_project_owner = prev }
	}
}

one_trait_realm_buildings_rebuild_shown_trigger = { always = yes }
one_trait_realm_buildings_rebuild_valid_trigger = { is_landed = yes }

# ---- List rows (R11/R12), real this batch ----
# One shown trigger per list (shared by every button of that list): the Titles list follows the same

# One valid trigger per button: the panel character must still hold scope:landed_title, plus the old
# option's own condition generalised from the capital/capital_province it used to run on (R12's example:
# "slot: the province exists").

# batch-4r4 repair (item 1): also exclude a landless-type title (CK3 will not transfer one, same cite as
# the panel's filter and the other Titles effects above).
one_trait_realm_titles_take_one_take_valid_trigger = {
	scope:landed_title.holder = this
	scope:landed_title = { is_landless_type_title = no }
}
# UI checklist 2026-10-15 (user): Claim and Destroy per title, same holder/landless-type gate as Take.
one_trait_realm_titles_take_one_claim_valid_trigger = {
	scope:landed_title.holder = this
	scope:landed_title = { is_landless_type_title = no }
	# UI checklist 2026-10-15 9b: greys once you hold a claim on it.
	custom_tooltip = {
		text = one_trait_realm_no_claim_yet_tt
		NOT = { root = { has_claim_on = scope:landed_title } }
	}
}
one_trait_realm_titles_take_one_destroy_valid_trigger = {
	scope:landed_title.holder = this
	scope:landed_title = { is_landless_type_title = no }
}
one_trait_realm_buildings_county_max_valid_trigger = {
	scope:landed_title.holder = this
}
one_trait_realm_buildings_county_slot_valid_trigger = {
	scope:landed_title.holder = this
	exists = scope:landed_title.title_province
}
# UI checklist 2026-10-15 row 28: greyed once the county already has an outbreak (province form:
# game/common/scripted_triggers/06_ce1_epidemic_triggers.txt:27-29).
one_trait_realm_realm_events_plague_plague_valid_trigger = {
	scope:landed_title.holder = this
	exists = scope:landed_title.title_province
	custom_tooltip = {
		text = one_trait_realm_no_outbreak_yet_tt
		scope:landed_title.title_province = {
			NOT = { any_province_epidemic = { always = yes } }
		}
	}
}
