# One Trait 2.1: does the realm have an authority law group that is not
# already at its LIMIT (maximum or minimum)?
one_trait_authority_can_change_trigger = {
	OR = {
		AND = {
			OR = {
				has_realm_law = crown_authority_0
				has_realm_law = crown_authority_1
				has_realm_law = crown_authority_2
				has_realm_law = crown_authority_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = crown_authority }
		}
		AND = {
			OR = {
				has_realm_law = tribal_authority_0
				has_realm_law = tribal_authority_1
				has_realm_law = tribal_authority_2
				has_realm_law = tribal_authority_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = tribal_authority }
		}
		AND = {
			OR = {
				has_realm_law = church_authority_0
				has_realm_law = church_authority_1
				has_realm_law = church_authority_2
				has_realm_law = church_authority_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = church_authority }
		}
		AND = {
			OR = {
				has_realm_law = imperial_bureaucracy_0
				has_realm_law = imperial_bureaucracy_1
				has_realm_law = imperial_bureaucracy_2
				has_realm_law = imperial_bureaucracy_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = imperial_bureaucracy }
		}
		AND = {
			OR = {
				has_realm_law = nomadic_authority_1
				has_realm_law = nomadic_authority_2
				has_realm_law = nomadic_authority_3
				has_realm_law = nomadic_authority_4
				has_realm_law = nomadic_authority_5
			}
			NOT = { realm_law_group_at_$LIMIT$_level = nomadic_authority }
		}
		AND = {
			OR = {
				has_realm_law = celestial_bureaucracy_0
				has_realm_law = celestial_bureaucracy_1
				has_realm_law = celestial_bureaucracy_2
				has_realm_law = celestial_bureaucracy_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = celestial_bureaucracy }
		}
		AND = {
			OR = {
				has_realm_law = meritocratic_bureaucracy_0
				has_realm_law = meritocratic_bureaucracy_1
				has_realm_law = meritocratic_bureaucracy_2
				has_realm_law = meritocratic_bureaucracy_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = meritocratic_bureaucracy }
		}
		AND = {
			OR = {
				has_realm_law = japanese_bureaucracy_0
				has_realm_law = japanese_bureaucracy_1
				has_realm_law = japanese_bureaucracy_2
				has_realm_law = japanese_bureaucracy_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = japanese_bureaucracy }
		}
	}
}
