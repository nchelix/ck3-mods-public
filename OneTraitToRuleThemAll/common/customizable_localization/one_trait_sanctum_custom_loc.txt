# One Trait 1.7 -- On/Off for each Sanctum power, shown in the story's
# Situations info line (ONE_TRAIT_SANCTUM_STATUS) as [Story.Custom('<name>')].

	type = all
	text = {
		trigger = { has_variable = power_builder }

		trigger = { always = no }
		fallback = yes
		localization_key = one_trait_sanctum_status_off
