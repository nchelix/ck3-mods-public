# One Trait 2.9: Summon, as the Court rows of the panel. Each effect creates one lowborn character.

# ---- 2.9: the Court rows (Realm > Court). They replace the 2.4 summon effects, which created the character in the
# player's own court; the old one_trait_summon_<k>_effect names and the menu that called them are gone (2.9 Task 4).
# Run with root = the player and scope:one_trait_target = the panel's character (the court owner): employer,
# culture, faith and rite are the owner's, the gender chance is the owner-based script value
# (common/script_values/one_trait_court_values.txt), and a master councillor takes the seat on the OWNER's
# council (assign_councillor_type runs in the owner's scope: effects.log:3503, "Supported Scopes: character").
# Batch 1 made court_chancellor real, batch 2 the other seven. Champion, spouse and ward get no seat and no
# relation, exactly as the old effects did (plan R1).

one_trait_realm_court_chancellor_effect = {
	create_character = {
		age = 30
		employer = scope:one_trait_target
		culture = scope:one_trait_target.culture
		faith = scope:one_trait_target.faith
		rite = scope:one_trait_target.rite
		dynasty = none
		gender_female_chance = one_trait_court_owner_faith_dominant_gender_female_chance
		trait = education_diplomacy_5
		trait = loyal
		trait = diligent
		diplomacy = 20
		martial = 10
		stewardship = 10
		intrigue = 10
		learning = 10
		prowess = 10
		save_temporary_scope_as = one_trait_summoned
	}
	# Vanilla's own override: eligible for any seat, whatever the gender laws
	scope:one_trait_summoned = {
		set_variable = { name = can_be_councillor_override value = yes }
	}
	scope:one_trait_target = {
		assign_councillor_type = {
			type = councillor_chancellor
			target = scope:one_trait_summoned
			remove_existing_councillor = yes
		}
	}
}

one_trait_realm_court_chaplain_effect = {
	create_character = {
		age = 30
		employer = scope:one_trait_target
		culture = scope:one_trait_target.culture
		faith = scope:one_trait_target.faith
		rite = scope:one_trait_target.rite
		dynasty = none
		gender_female_chance = one_trait_court_owner_clergy_gender_female_chance
		trait = education_learning_5
		trait = loyal
		trait = diligent
		diplomacy = 10
		martial = 10
		stewardship = 10
		intrigue = 10
		learning = 20
		prowess = 10
		save_temporary_scope_as = one_trait_summoned
	}
	# Vanilla's own override: eligible for any seat, whatever the gender laws
	scope:one_trait_summoned = {
		set_variable = { name = can_be_councillor_override value = yes }
	}
	# 2.13 C-fix3 (self-test s21c5 in phase C2: on yourself, nobody joined the court). Vanilla's court chaplain
	# on_get_position (game/common/council_positions/00_council_positions.txt:878-905, the same in 1.20.1.2) treats a
	# non-ruler seated by script as the AUTOMATIC fill when the owner's rite has clerical regions and cannot appoint
	# its own chaplain (Catholic and the like): it seats a landed realm priest instead
	# (assign_landed_realm_priest_effect) and move_to_pool's our character. It is skipped for a player owner flagged
	# manual_council_appointment_in_progress, the flag vanilla's own council-position offer sets around its
	# assignment (game/common/character_interactions/00_vassal_interactions.txt:1729-1737), and for any owner flagged
	# pam_church_replacing_realm_priest, the flag vanilla's church sets around installing its own pick
	# (pam_interactions.txt:2628-2635); an AI owner is only exempted by the second. Both live for this one assignment.
	scope:one_trait_target = {
		add_character_flag = { flag = manual_council_appointment_in_progress days = 1 }
		if = {
			limit = { is_ai = yes }
			add_character_flag = { flag = pam_church_replacing_realm_priest days = 1 }
		}
		assign_councillor_type = {
			type = councillor_court_chaplain
			target = scope:one_trait_summoned
			remove_existing_councillor = yes
		}
		remove_character_flag = manual_council_appointment_in_progress
		if = {
			limit = { has_character_flag = pam_church_replacing_realm_priest }
			remove_character_flag = pam_church_replacing_realm_priest
		}
	}
}
