# One Trait 2.4: Summon. Each effect creates one lowborn character in root's
# court, with root's culture and faith. Run in root (the player) scope.

one_trait_summon_chancellor_effect = {
	create_character = {
		age = 30
		employer = root
		culture = root.culture
		faith = root.faith
		rite = root.rite
		dynasty = none
		gender_female_chance = root_faith_dominant_gender_adjusted_female_chance
		trait = education_diplomacy_5
		trait = loyal
		trait = diligent
		diplomacy = 20
		martial = 10
		stewardship = 10
		intrigue = 10
		learning = 10
		prowess = 10
		save_temporary_scope_as = one_trait_summoned
	}
	# Vanilla's own override: eligible for any seat, whatever the gender laws
	scope:one_trait_summoned = {
		set_variable = { name = can_be_councillor_override value = yes }
	}
	assign_councillor_type = {
		type = councillor_chancellor
		target = scope:one_trait_summoned
		remove_existing_councillor = yes
	}
}
