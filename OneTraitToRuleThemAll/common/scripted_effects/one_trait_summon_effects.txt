# One Trait 2.9: Summon, as the Court rows of the panel. Each effect creates one lowborn character.

# ---- 2.9: the Court rows (Realm > Court). They replace the 2.4 summon effects, which created the character in the
# player's own court; the old one_trait_summon_<k>_effect names and the menu that called them are gone (2.9 Task 4).
# Run with root = the player and scope:one_trait_target = the panel's character (the court owner): employer,
# culture, faith and rite are the owner's, the gender chance is the owner-based script value
# (common/script_values/one_trait_court_values.txt), and a master councillor takes the seat on the OWNER's
# council (assign_councillor_type runs in the owner's scope: effects.log:3503, "Supported Scopes: character").
# Batch 1 made court_chancellor real, batch 2 the other seven. Champion, spouse and ward get no seat and no
# relation, exactly as the old effects did (plan R1).

one_trait_realm_court_chancellor_effect = {
	create_character = {
		age = 30
		employer = scope:one_trait_target
		culture = scope:one_trait_target.culture
		faith = scope:one_trait_target.faith
		rite = scope:one_trait_target.rite
		dynasty = none
		gender_female_chance = one_trait_court_owner_faith_dominant_gender_female_chance
		trait = education_diplomacy_5
		trait = loyal
		trait = diligent
		diplomacy = 20
		martial = 10
		stewardship = 10
		intrigue = 10
		learning = 10
		prowess = 10
		save_temporary_scope_as = one_trait_summoned
	}
	# Vanilla's own override: eligible for any seat, whatever the gender laws
	scope:one_trait_summoned = {
		set_variable = { name = can_be_councillor_override value = yes }
	}
	scope:one_trait_target = {
		assign_councillor_type = {
			type = councillor_chancellor
			target = scope:one_trait_summoned
			remove_existing_councillor = yes
		}
	}
}
