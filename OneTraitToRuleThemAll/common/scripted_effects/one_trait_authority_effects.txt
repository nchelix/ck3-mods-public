# One Trait 2.1: step whichever authority law group the realm has (root
# scope: the player). LIMIT is maximum or minimum.
# The limit triggers take the law GROUP, not one of its laws.
# UI checklist 2026-10-15: change_realm_law_level charged the law's own pass_cost (prestige,
# game/common/laws/00_realm_laws.txt:94-103) and set its cooldown in on_pass (:105-120). A cheat should
# do neither, so each step adds the next law directly with add_realm_law_skip_effects (the free form the
# Crowned row already uses). One else_if chain per law group, so each group the realm has moves exactly one level.
# LIMIT = maximum raises, LIMIT = minimum lowers. CK3 refuses to compile a call passing an argument the
# effect never uses ("failed for unknown arguments: CHANGE"), so the callers pass LIMIT alone.
one_trait_authority_change_effect = {
	one_trait_authority_step_$LIMIT$_effect = yes
}

one_trait_authority_step_maximum_effect = {
	if = { limit = { has_realm_law = crown_authority_0 } add_realm_law_skip_effects = crown_authority_1 }
	else_if = { limit = { has_realm_law = crown_authority_1 } add_realm_law_skip_effects = crown_authority_2 }
	else_if = { limit = { has_realm_law = crown_authority_2 } add_realm_law_skip_effects = crown_authority_3 }
	if = { limit = { has_realm_law = tribal_authority_0 } add_realm_law_skip_effects = tribal_authority_1 }
	else_if = { limit = { has_realm_law = tribal_authority_1 } add_realm_law_skip_effects = tribal_authority_2 }

	if = { limit = { has_realm_law = church_authority_0 } add_realm_law_skip_effects = church_authority_1 }
	else_if = { limit = { has_realm_law = church_authority_1 } add_realm_law_skip_effects = church_authority_2 }
	else_if = { limit = { has_realm_law = church_authority_2 } add_realm_law_skip_effects = church_authority_3 }
	if = { limit = { has_realm_law = imperial_bureaucracy_0 } add_realm_law_skip_effects = imperial_bureaucracy_1 }
	else_if = { limit = { has_realm_law = imperial_bureaucracy_1 } add_realm_law_skip_effects = imperial_bureaucracy_2 }
	else_if = { limit = { has_realm_law = imperial_bureaucracy_2 } add_realm_law_skip_effects = imperial_bureaucracy_3 }
	if = { limit = { has_realm_law = nomadic_authority_1 } add_realm_law_skip_effects = nomadic_authority_2 }
	else_if = { limit = { has_realm_law = nomadic_authority_2 } add_realm_law_skip_effects = nomadic_authority_3 }
	else_if = { limit = { has_realm_law = nomadic_authority_3 } add_realm_law_skip_effects = nomadic_authority_4 }

	else_if = { limit = { has_realm_law = celestial_bureaucracy_1 } add_realm_law_skip_effects = celestial_bureaucracy_2 }
	else_if = { limit = { has_realm_law = celestial_bureaucracy_2 } add_realm_law_skip_effects = celestial_bureaucracy_3 }

	else_if = { limit = { has_realm_law = meritocratic_bureaucracy_2 } add_realm_law_skip_effects = meritocratic_bureaucracy_3 }
