# One Trait 2.1: step whichever authority law group the realm has (root
# scope: the player). CHANGE is 1 or -1; LIMIT is maximum or minimum.
# The limit triggers take the law GROUP, not one of its laws.
one_trait_authority_change_effect = {
	if = {
		limit = {
			OR = {
				has_realm_law = crown_authority_0
				has_realm_law = crown_authority_1
				has_realm_law = crown_authority_2
				has_realm_law = crown_authority_3
			}
			NOT = { realm_law_group_at_$LIMIT$_level = crown_authority }
