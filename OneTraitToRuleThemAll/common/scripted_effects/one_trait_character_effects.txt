# One Trait character editor (1.4). Since 2.8 the bundles below, and the one_trait_character_<key>_effect
# rows at the end of this file, are called from the panel's Character sub-tabs (the generated
# one_trait_ui_act_<key>_click_effect, tools/build_one_trait_panel.py), not from the retired menu pages.
#
# Every trait key below is checked by tools/validate_mod.py against the
# installed game, so a patch that renames a trait fails validation rather than
# silently doing nothing. When that happens, this file is the one place to fix.

# Guards. Vanilla never add/removes a named trait unguarded
# (see remove_all_bastard_traits); neither do we.
one_trait_remove_if_present = {
	if = {
		limit = { has_trait = $TRAIT$ }
		remove_trait = $TRAIT$
	}
}

one_trait_add_if_absent = {
	if = {
		limit = { NOT = { has_trait = $TRAIT$ } }
		add_trait = $TRAIT$
	}
}

# Bundle 3. Enumerates by category rather than by list, so it survives patches
# that add personality traits -- the pattern is vanilla's, from
# events/dlc/bp2/bp2_yearly_2.txt. Clearing the receiver first means opposite
# traits cannot collide; the giver's own set is already consistent.
# The list is filled before anything is removed: never mutate the collection
# being iterated.
one_trait_copy_personality_effect = {
	$RECEIVER$ = {
		every_character_trait = {
			limit = { has_trait_category = personality }
			add_to_temporary_list = one_trait_old_personality
		}
	}
	every_in_list = {
		list = one_trait_old_personality
		save_temporary_scope_as = one_trait_trait_to_remove
		$RECEIVER$ = { remove_trait = scope:one_trait_trait_to_remove }
	}
	$GIVER$ = {
		every_character_trait = {
			limit = { has_trait_category = personality }
			save_temporary_scope_as = one_trait_trait_to_copy
			$RECEIVER$ = { add_trait = scope:one_trait_trait_to_copy }
		}
	}
}

# There is no random-name effect in 1.19. A donor of the same culture, faith
# and sex gets a culture-appropriate name from the engine; we copy it and the
# donor disappears silently (vanilla's own effect). The donor's location is the
# player's, which always exists. One dead, unremarkable character is left in
# history; accepted in the spec.
one_trait_reroll_name_effect = {
	save_temporary_scope_as = one_trait_rename_target
	create_character = {
		gender_female_chance = {
			if = {
				limit = { scope:one_trait_rename_target = { is_female = yes } }
				add = 100
			}
		}
		culture = scope:one_trait_rename_target.culture
		faith = scope:one_trait_rename_target.faith
		age = 20
		dynasty = none
		location = root.location
		save_scope_as = one_trait_name_donor # vanilla never uses the temporary form here
	}
	scope:one_trait_rename_target = {
		change_first_name = { template_character = scope:one_trait_name_donor }
	}
	scope:one_trait_name_donor = {
		silent_disappearance_effect = yes
	}
}

# Real (2.8 Task 1): the retired father option of the canonize page (menu page 0140). Saved scopes persist
# along an event chain, so clear before saving again (as the page's immediate did). The father's scope and
# the consecration place are saved here from scope:one_trait_target: its capital province, else the
# player's. pam_canonize_saint_effect (By God Alone): the Saint trait, sainthood in the faith, and relics;
# the shown/valid triggers keep it to a dead father who is not yet a saint.
one_trait_character_legacy_canonize_father_effect = {
	if = {
		limit = { exists = scope:one_trait_canon_father }
		clear_saved_scope = one_trait_canon_father
	}
	scope:one_trait_target.father ?= { save_scope_as = one_trait_canon_father }
	if = {
		limit = { exists = scope:one_trait_canon_place }
		clear_saved_scope = one_trait_canon_place
	}
	if = {
		limit = { exists = scope:one_trait_target.capital_province }
		scope:one_trait_target.capital_province = { save_scope_as = one_trait_canon_place }
	}
	else = {
		root.capital_province = { save_scope_as = one_trait_canon_place }
	}
	pam_canonize_saint_effect = {
		CANONIZER = scope:one_trait_canon_father
		CANDIDATE = scope:one_trait_canon_father
		CONSECRATION_PROVINCE = scope:one_trait_canon_place
		PRESET = no
	}
}
