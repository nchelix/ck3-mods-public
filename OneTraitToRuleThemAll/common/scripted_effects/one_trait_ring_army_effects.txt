# One Trait 2.9.1: the ring army's strength level, Count, grants, conversion and Disband. The Army tab's scripted
# GUIs (tools/build_one_trait_panel.py) call these. Plan: docs/superpowers/plans/2026-10-19-one-trait-2.9.1-ring-army-and-independence.md
# (Task 1 and Revisions R2-R4). Engine names: effects.log create_maa_regiment :3996, change_maa_regiment_size :13326,
# destroy_maa_regiment :13347, every_/random_maa_regiment :4757 / :8814, clamp_variable :223, set_variable :2797,
# save_temporary_scope_value_as :2699; triggers.log maa_size :9400, is_maa_type :9351, is_raised :9364;
# event_targets.log regiment_owner :1079.
#
# Scope rules: the level effects, grant, convert and disband run in the CHARACTER whose army it is (the panel's
# target); root is the PLAYER (it holds the Count and the target variable, R3). Variables are read with has_variable.

# ---- the Count (the player's own variable, default 5000, range 100-50000) ---------------------------------------
# The player scope (the Army tab's scripted GUI). A step below 100 stays 100, above 50000 stays 50000.
one_trait_ring_count_step_effect = {
	if = {
		limit = { NOT = { has_variable = one_trait_ring_count } }
		set_variable = { name = one_trait_ring_count value = 5000 }
	}
	change_variable = { name = one_trait_ring_count add = $AMOUNT$ }
	clamp_variable = { name = one_trait_ring_count min = 100 max = 50000 }
}

# ---- the level (character scope) ----------------------------------------------------------------------------
# Godlike = no variable (a 2.9 save has none); Elite / Regular store flag:elite / flag:regular.
one_trait_ring_apply_level_godlike_effect = {
	if = {
		limit = { has_variable = one_trait_ring_strength }
		remove_variable = one_trait_ring_strength
	}
}

one_trait_ring_apply_level_elite_effect = {
	set_variable = { name = one_trait_ring_strength value = flag:elite }
}

one_trait_ring_apply_level_regular_effect = {
	set_variable = { name = one_trait_ring_strength value = flag:regular }
}

# LEVEL = godlike | elite | regular. Sets the level, then converts every ring regiment to it.
one_trait_ring_set_level_effect = {
	one_trait_ring_apply_level_$LEVEL$_effect = yes
	one_trait_ring_convert_effect = yes
}

# ---- creating one regiment of N companies (R2, reworked after Codex review 1) --------------------------------------
# create_maa_regiment's size must be a literal number (Pantheon proved "Malformed token" for a script value,
# PantheonOfOne/common/scripted_effects/pantheon_effects.txt:137), and a regiment cannot hold a variable, so a
# new one cannot be told apart from an old one afterwards. one_trait_ring_create_sized_effect (GENERATED by
# tools/build_one_trait_ring_sizes.py, one_trait_ring_sizes_generated.txt) therefore creates the regiment directly at
# its literal size by chaining on scope:ring_n = 1 to 500 (ring_n above 500, only an unusual conversion, makes 500).
# Character scope; scope:ring_n = the whole companies wanted (set by the caller).
# The refill then fills every unraised regiment of the type (change_maa_regiment_size size = 0 + reinforce = yes, the
# form one_trait_sanctum_run_levies uses). That also tops up other unraised regiments of the type: harmless in a
# cheat mod, and the new one is always among them.
one_trait_ring_create_effect = {
	one_trait_ring_create_sized_effect = { TYPE = $TYPE$ }
	every_maa_regiment = {
		limit = {
			is_maa_type = $TYPE$
			is_raised = no
		}
		change_maa_regiment_size = {
			size = 0
			reinforce = yes
		}
	}
}

# ---- a grant (character scope): the current level's type, the Count's companies ----------------------------------
# ROLE = legion | bowmen | riders. one_trait_ring_companies_value is rooted on the player (root.var:one_trait_ring_count).
one_trait_ring_grant_effect = {
	save_temporary_scope_value_as = {
		name = ring_n
		value = one_trait_ring_companies_value
	}
	if = {
		limit = { one_trait_ring_is_godlike_trigger = yes }
		one_trait_ring_create_effect = { TYPE = one_trait_$ROLE$ }
	}
	else_if = {
		limit = { one_trait_ring_is_elite_trigger = yes }
		one_trait_ring_create_effect = { TYPE = one_trait_$ROLE$_elite }
	}
	else = {
		one_trait_ring_create_effect = { TYPE = one_trait_$ROLE$_regular }
	}
}

# ---- conversion (character scope) --------------------------------------------------------------------------------
# One type to another, regiment by regiment: new companies = ceil(maa_size x old stack / new stack), at least 1
# (R4: "men kept" means CAPACITY; the new regiment is refilled to full by the create effect). The new regiment is
# created in the owner's scope first, then the old one is destroyed. Called for old types only, so the loop never
# meets what it creates.
one_trait_ring_convert_type_effect = {
	every_maa_regiment = {
		limit = { is_maa_type = $OLD$ }
		save_temporary_scope_value_as = {
			name = ring_n
			value = {
				value = maa_size
				multiply = $OLD_STACK$
				divide = $NEW_STACK$
				ceiling = yes
				min = 1
			}
		}
		regiment_owner = { one_trait_ring_create_effect = { TYPE = $NEW$ } }
		destroy_maa_regiment = yes
	}
}

# Every ring regiment not of the character's current level becomes the same role at the current level.
one_trait_ring_convert_effect = {
	if = {
		limit = { one_trait_ring_is_godlike_trigger = yes }
		one_trait_ring_convert_type_effect = { OLD = one_trait_legion_elite OLD_STACK = 200 NEW = one_trait_legion NEW_STACK = 500 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_bowmen_elite OLD_STACK = 200 NEW = one_trait_bowmen NEW_STACK = 500 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_riders_elite OLD_STACK = 200 NEW = one_trait_riders NEW_STACK = 500 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_legion_regular OLD_STACK = 100 NEW = one_trait_legion NEW_STACK = 500 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_bowmen_regular OLD_STACK = 100 NEW = one_trait_bowmen NEW_STACK = 500 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_riders_regular OLD_STACK = 100 NEW = one_trait_riders NEW_STACK = 500 }
	}
	else_if = {
		limit = { one_trait_ring_is_elite_trigger = yes }
		one_trait_ring_convert_type_effect = { OLD = one_trait_legion OLD_STACK = 500 NEW = one_trait_legion_elite NEW_STACK = 200 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_bowmen OLD_STACK = 500 NEW = one_trait_bowmen_elite NEW_STACK = 200 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_riders OLD_STACK = 500 NEW = one_trait_riders_elite NEW_STACK = 200 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_legion_regular OLD_STACK = 100 NEW = one_trait_legion_elite NEW_STACK = 200 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_bowmen_regular OLD_STACK = 100 NEW = one_trait_bowmen_elite NEW_STACK = 200 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_riders_regular OLD_STACK = 100 NEW = one_trait_riders_elite NEW_STACK = 200 }
	}
	else = {
		one_trait_ring_convert_type_effect = { OLD = one_trait_legion OLD_STACK = 500 NEW = one_trait_legion_regular NEW_STACK = 100 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_bowmen OLD_STACK = 500 NEW = one_trait_bowmen_regular NEW_STACK = 100 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_riders OLD_STACK = 500 NEW = one_trait_riders_regular NEW_STACK = 100 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_legion_elite OLD_STACK = 200 NEW = one_trait_legion_regular NEW_STACK = 100 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_bowmen_elite OLD_STACK = 200 NEW = one_trait_bowmen_regular NEW_STACK = 100 }
		one_trait_ring_convert_type_effect = { OLD = one_trait_riders_elite OLD_STACK = 200 NEW = one_trait_riders_regular NEW_STACK = 100 }
	}
}

# ---- Disband (character scope): every ring regiment of all nine types ---------------------------------------------
one_trait_ring_disband_effect = {
	every_maa_regiment = {
		limit = { one_trait_ring_regiment_trigger = yes }
		destroy_maa_regiment = yes
	}
}
