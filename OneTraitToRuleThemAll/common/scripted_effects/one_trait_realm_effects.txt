# One Trait 2.7 Task 2 (batch-2 handoff): the Realm tab's hand-written effects, named
# one_trait_realm_<key>_effect for each tools/build_one_trait_panel.py REALM_ACTIONS key, and
# one_trait_realm_<list>_<button>_effect for each REALM_LISTS button. Spec:
# docs/superpowers/specs/2026-10-14-one-trait-2.7-realm-pages-design.md section 2.
#
# Scope rule (R1, plan docs/superpowers/plans/2026-10-14-one-trait-2.7-realm-pages.md "Revisions"): root is
# the player throughout (the generated click effect saves scope:one_trait_target from
# var:one_trait_ui_target just before calling one of these, the same idiom
# OneTraitRealmProto/common/scripted_guis/one_trait_realm_proto_guis.txt:87 proved). Succession's six
# effects below each wrap their WHOLE moved body in "scope:one_trait_target = { ... }" as their only
# top-level statement, because the moved option bodies (one_trait_menu_events.txt:1221-1314) act on the
# implicit current scope, which used to be root (the player, since the menu always ran Succession on
# itself) and must now be the panel's character instead; "holder = root" in create_titles becomes
# "holder = scope:one_trait_target" for the same reason. Titles/Culture/Buildings/Realm events (Task 3)
# already act on scope:one_trait_target/root as two characters (the spec's "you and them"), so those
# bodies move unchanged, with no such wrapper, when Task 3 fills them in.
#
# Titles (1200), Culture (0430), Buildings (0600/0610) and Realm events (0420), plus the three R11/R12 list
# buttons, are filled in by Batch 3 (Task 3) below: every body is the retired option's, minus ONLY its
# trigger_event = one_trait_menu.<page> navigation line (R3 keeps conqueror.0001 and the delayed 0421).
#
# Re-check on every CK3 patch -- carried over from one_trait_menu_events.txt:5-15, the domicile and struggle
# effects below fail silently rather than erroring if the game changes underneath them:
#   - The five hardcoded top-tier domicile building ids on realm_events_domicile (camp_main_04,
#     estate_main_05, east_asian_estate_main_05, yurt_main_06, japanese_manor_main_06) may stop being the
#     top tier if a patch adds a higher level.
#   - The is_domicile_type if/else_if chain on realm_events_domicile has no final else, so a new domicile
#     type added by a patch gets no building added and no warning.
#   - The two hardcoded struggle ids on realm_events_struggle (iberian_struggle, persian_struggle) would not
#     include a third struggle if one is added.

# Moved from one_trait_menu_events.txt:1180-1182 (option .destroy_primary), minus its trigger (:1177-1179,
# moved to one_trait_realm_titles_destroy_primary_valid_trigger below) and trigger_event (:1183).
one_trait_realm_titles_destroy_primary_effect = {
	scope:one_trait_target = {
		if = {
			limit = { primary_title.tier >= tier_duchy }
			destroy_title = primary_title
		}
		else = {
			# 2026-10-14 (user): a county is land-bound and cannot be destroyed (destroy_title silently does
			# nothing), so a count's primary county is revoked instead: to their liege, or to the player when
			# they are independent. A count with other counties keeps those (the game picks a new primary);
			# a count with only this one becomes landless. Revoke form: 00_diarch_interactions.txt:9730-9733.
			if = { limit = { exists = scope:one_trait_county_heir } clear_saved_scope = one_trait_county_heir }
			if = {
				limit = { is_independent_ruler = yes }
				root = { save_scope_as = one_trait_county_heir }
			}
			else = {
				liege = { save_scope_as = one_trait_county_heir }
			}
			create_title_and_vassal_change = {
				type = revoked
				save_scope_as = change
				add_claim_on_loss = no
			}
			primary_title = {
				change_title_holder_include_vassals = {
					holder = scope:one_trait_county_heir
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
	}
}

# Moved from one_trait_menu_events.txt:267-298 (option .domicile), minus its trigger (:264-266, moved to
# one_trait_realm_realm_events_domicile_shown_trigger below) and trigger_event (:299).
# Item 2 (batch-4r2 repair): error.log -- "add_domicile_building effect [ Cannot construct an upgrade when
# previous building has not been built: 'east_asian_estate_main_05' ]". A single add_domicile_building call
# for the top tier only works if every lower tier is already built; build each tier that isn't built yet,
# lowest first, up to the top tier below, using vanilla's own gating idiom (has_domicile_building = the
# previous tier, NOT = { has_domicile_building_or_higher = this tier }), e.g.
# game/common/scripted_effects/00_administrative_effects.txt:404-421 (set_up_domicile_estate_effect).
# Chains (game/common/domiciles/buildings/, tier 01 up to the top tier named below):
#   camp: 00_camp_buildings.txt (camp_main_01..04); estate: 00_estate_buildings.txt (estate_main_01..05);
#   east_asian_estate: 00_chinese_estate_buildings.txt (east_asian_estate_main_01..05);
#   yurt: 00_yurt_buildings.txt (yurt_main_01..06);
#   japanese_manor: 10_japanese_manor_buildings.txt (japanese_manor_main_01..06).
# Item 1 (batch-4r3 repair): error.log -- "add_domicile_building effect [ Domicile owner failed to meet
# triggered requirements: 'east_asian_estate_main_04' ... ]" (run 3). The previous-tier gate above is not
# enough above tier 03: tiers 04/05(/06) carry their own owner-culture `can_construct` requirement that the
# chain never tested before calling add_domicile_building. Found the vanilla idiom for testing this BEFORE
# adding: game/common/scripted_effects/00_administrative_effects.txt:403-421 (estate), :441-459
# (japanese_manor), :475-493 (east_asian_estate) -- set_up_domicile_estate_effect, called from DOMICILE
# scope exactly like this effect, copies each tier's own `can_construct` requirement into its own gate as
# `owner.culture ?= { has_innovation = ... }` (the building's own can_construct block runs with the OWNER as
# its implicit scope -- the error says "Domicile owner failed..." -- so from domicile scope it needs the
# `owner.` link; vanilla's `culture ?= {...}` bare form in the can_construct block itself is the same
# requirement with that implicit owner scope already current). set_up_domicile_estate_effect only climbs to
# tier 03 ("intentionally one level lower than what you can get"); tiers 04/05/06 are not in that function,
# so their own `can_construct` blocks are copied directly from the building files (cited per tier below) in
# the same `owner.culture ?=` form, per the handoff's fallback ("if no script-side test exists, copy each
# tier's requirement block into the gate"):
#   estate_main_04 can_construct: game/common/domiciles/buildings/00_estate_buildings.txt:251-252
#   estate_main_05 can_construct: game/common/domiciles/buildings/00_estate_buildings.txt:336-337
#   east_asian_estate_main_04 can_construct: game/common/domiciles/buildings/00_chinese_estate_buildings.txt:253-255
#   east_asian_estate_main_05 can_construct: game/common/domiciles/buildings/00_chinese_estate_buildings.txt:291-293
#   japanese_manor_main_04 can_construct: game/common/domiciles/buildings/10_japanese_manor_buildings.txt:128-129
#   japanese_manor_main_05 can_construct: game/common/domiciles/buildings/10_japanese_manor_buildings.txt:171-172
#   japanese_manor_main_06 can_construct: game/common/domiciles/buildings/10_japanese_manor_buildings.txt:213-214
# camp (00_camp_buildings.txt, main_01..04) has NO can_construct block on any main tier -- checked directly,
# no gate needed there. yurt's can_construct blocks (00_yurt_buildings.txt) are an OR of `has_realm_law`
# instead of culture (main_03: :77; main_04: :132; main_05: :187; main_06: :242). Vanilla never writes
# `owner.has_realm_law = X`, so the gate enters the owner's scope instead: `owner = { OR = { has_realm_law = ...
# } }` (a scope block followed by an ordinary character trigger).
one_trait_realm_realm_events_domicile_effect = {
	scope:one_trait_target.domicile = {
		if = {
			limit = {
				is_domicile_type = camp
			}
			if = { limit = { NOT = { has_domicile_building_or_higher = camp_main_01 } } add_domicile_building = camp_main_01 }
			if = { limit = { has_domicile_building = camp_main_01 NOT = { has_domicile_building_or_higher = camp_main_02 } } add_domicile_building = camp_main_02 }
			if = { limit = { has_domicile_building = camp_main_02 NOT = { has_domicile_building_or_higher = camp_main_03 } } add_domicile_building = camp_main_03 }
			if = { limit = { has_domicile_building = camp_main_03 NOT = { has_domicile_building_or_higher = camp_main_04 } } add_domicile_building = camp_main_04 }
		}
		else_if = {
			limit = {
				is_domicile_type = estate
			}
			if = { limit = { NOT = { has_domicile_building_or_higher = estate_main_01 } } add_domicile_building = estate_main_01 }
			if = { limit = { has_domicile_building = estate_main_01 NOT = { has_domicile_building_or_higher = estate_main_02 } owner.culture ?= { has_innovation = innovation_city_planning } } add_domicile_building = estate_main_02 }
			if = { limit = { has_domicile_building = estate_main_02 NOT = { has_domicile_building_or_higher = estate_main_03 } owner.culture ?= { has_innovation = innovation_manorialism } } add_domicile_building = estate_main_03 }
			if = { limit = { has_domicile_building = estate_main_03 NOT = { has_domicile_building_or_higher = estate_main_04 } owner.culture ?= { has_innovation = innovation_development_03 } } add_domicile_building = estate_main_04 }
			if = { limit = { has_domicile_building = estate_main_04 NOT = { has_domicile_building_or_higher = estate_main_05 } owner.culture ?= { has_innovation = innovation_cranes } } add_domicile_building = estate_main_05 }
		}
		else_if = {
			limit = {
				is_domicile_type = east_asian_estate
			}
			if = { limit = { NOT = { has_domicile_building_or_higher = east_asian_estate_main_01 } } add_domicile_building = east_asian_estate_main_01 }
			if = { limit = { has_domicile_building = east_asian_estate_main_01 NOT = { has_domicile_building_or_higher = east_asian_estate_main_02 } owner.culture ?= { has_innovation = innovation_city_planning } } add_domicile_building = east_asian_estate_main_02 }
			if = { limit = { has_domicile_building = east_asian_estate_main_02 NOT = { has_domicile_building_or_higher = east_asian_estate_main_03 } owner.culture ?= { has_innovation = innovation_manorialism } } add_domicile_building = east_asian_estate_main_03 }
			if = { limit = { has_domicile_building = east_asian_estate_main_03 NOT = { has_domicile_building_or_higher = east_asian_estate_main_04 } owner.culture ?= { has_innovation = innovation_development_03 } } add_domicile_building = east_asian_estate_main_04 }
			if = { limit = { has_domicile_building = east_asian_estate_main_04 NOT = { has_domicile_building_or_higher = east_asian_estate_main_05 } owner.culture ?= { has_innovation = innovation_cranes } } add_domicile_building = east_asian_estate_main_05 }
		}
		else_if = {
			limit = {
				is_domicile_type = yurt
			}
			if = { limit = { NOT = { has_domicile_building_or_higher = yurt_main_01 } } add_domicile_building = yurt_main_01 }
			if = { limit = { has_domicile_building = yurt_main_01 NOT = { has_domicile_building_or_higher = yurt_main_02 } } add_domicile_building = yurt_main_02 }
			if = {
				limit = {
					has_domicile_building = yurt_main_02
					NOT = { has_domicile_building_or_higher = yurt_main_03 }
					owner = { OR = { has_realm_law = nomadic_authority_2 has_realm_law = nomadic_authority_3 has_realm_law = nomadic_authority_4 has_realm_law = nomadic_authority_5 } }
				}
				add_domicile_building = yurt_main_03
			}
			if = {
				limit = {

# UI checklist 2026-10-15 (user): destroy one title -- destroy_primary's body (above) for
# scope:landed_title: a duchy or higher is destroyed; a county is land-bound (destroy_title does nothing),
# so it is revoked to their liege, or to the player when they are independent.
one_trait_realm_titles_take_one_destroy_effect = {
	if = {
		limit = { scope:landed_title.tier >= tier_duchy }
		destroy_title = scope:landed_title
	}
	else = {
		if = { limit = { exists = scope:one_trait_county_heir } clear_saved_scope = one_trait_county_heir }
		scope:one_trait_target = {
			if = {
				limit = { is_independent_ruler = yes }
				root = { save_scope_as = one_trait_county_heir }
			}
			else = {
				liege = { save_scope_as = one_trait_county_heir }
			}
		}
		create_title_and_vassal_change = {
			type = revoked
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:landed_title = {
			change_title_holder_include_vassals = {
				holder = scope:one_trait_county_heir
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}
