# One Trait 2.11: Break their army. The Diplomacy -> Hostility row's four buttons (25% / 50% / 75% / All; the
# hand-written one_trait_diplomacy_hostility_break_<x>_effect in one_trait_diplomacy_effects.txt) call this effect in
# the panel target's scope. Spec: docs/superpowers/specs/2026-11-06-one-trait-2.11-break-their-army-design.md; plan:
# docs/superpowers/plans/2026-11-06-one-trait-2.11-break-their-army.md ("What one_trait_break_army_effect does" and
# Revisions R1-R11).
#
# Scope rule: this = scope:one_trait_target (saved by the generated click effect), root = the player.
#
# Engine names (docs/reference/ck3-1.20.1.1-*.log):
#   effects.log every_army :4221, every_maa_regiment :4757, save_temporary_scope_as :2693,
#     send_interface_toast :10037, change_maa_troops_count :13341 ("can't exceed max troops count"),
#     destroy_maa_regiment :13347 ("Doesn't work on regiments that belong to mercenary companies and holy orders"),
#     deplete_army_by_percentage :15134;
#   triggers.log is_event_maa_regiment :9333, is_hired_maa_regiment :9339, is_maa_in_combat :9345, is_raised :9364,
#     maa_current_troops_count :9388;
#   event_targets.log army_owner :1386.
#
# Rules this file keeps (plan Global Constraints, R1):
# - never change_maa_regiment_size (it shrinks the regiment's maximum for good);
# - never deplete_army_by_percentage = 1 (the engine ignores it; Pantheon 2.4): "All" on an army is two 0.95 passes,
#   as PantheonOfOne/common/scripted_effects/pantheon_raid_effects.txt:21-25 and :38 do;
# - one click cuts each soldier at most once: a RAISED regiment is cut only through its army (step 1), a regiment
#   at home only by the men-at-arms pass (step 2). is_raised splits them.
#
# $SHARE$ = 0.25 / 0.5 / 0.75 (ALL = no) or 0.95 (ALL = yes). The toast's desc key is built from $SHARE$
# (one_trait_break_army_toast_<share>_desc, written by tools/build_one_trait_panel.py), so pass it exactly as the four
# row effects do.
one_trait_break_army_effect = {
	# R4: click effects do not check the valid trigger, so the effect refuses the player itself (the row is hidden
	# on yourself too). Nothing happens and no toast is sent.
	if = {
		limit = { NOT = { scope:one_trait_target = root } }

		# 1. Armies in the field: raised levies, and raised men-at-arms with them (vassal levies marching in the
		#    target's army too: the engine cannot separate them). Owner-filtered (R11; vanilla form
		#    game/events/game_rule_events.txt:576, limit = { army_owner = scope:winner }), so an ally's or a vassal's
		#    own army is never touched. All adds a second 0.95 pass under ?= (the army may be gone after the first).
		every_army = {
			limit = { army_owner = scope:one_trait_target }
			save_temporary_scope_as = one_trait_break_army
			deplete_army_by_percentage = $SHARE$
			if = {
				limit = { always = $ALL$ }
				scope:one_trait_break_army ?= { deplete_army_by_percentage = 0.95 }
			}
		}

		if = {
			limit = { always = $ALL$ }
			# 2. All, men-at-arms at home: destroyed (vanilla every_maa_regiment + destroy_maa_regiment,
			#    game/common/decisions/dlc_decisions/tgp/tgp_steppe_decisions.txt:339-343). Event troops are story
			#    troops and stay. Hired mercenaries and holy orders cannot be destroyed (effects.log:13347): they are
			#    emptied instead.
			every_maa_regiment = {
				limit = {
					is_raised = no
					is_event_maa_regiment = no
				}
				if = {
					limit = { is_hired_maa_regiment = no }
					destroy_maa_regiment = yes
				}
				else = {
					change_maa_troops_count = {
						value = maa_current_troops_count
						multiply = -1
					}
				}
			}
			# 3. All, raised regiments: step 1 already emptied them; now the regiment itself goes. Not hired ones (the
			#    engine refuses; a band the target has hired out sits in someone else's army, which step 1 never
			#    reaches), not event troops, and not one in combat (left as step 1 emptied it).
			every_maa_regiment = {
				limit = {
					is_raised = yes
					is_event_maa_regiment = no
					is_hired_maa_regiment = no
					is_maa_in_combat = no
				}
				destroy_maa_regiment = yes
			}
		}
		else = {
			# 2. 25 / 50 / 75%, men-at-arms at home: each regiment loses that share of its CURRENT soldiers. Vanilla
			#    removes soldiers with change_maa_troops_count = { value = maa_current_troops_count multiply = <negative> }
			#    (game/events/scheme_events/laamp_base_contract_scheme_events.txt:10111, values -0.6 to -0.2 at
			#    game/common/script_values/07_ep3_values.txt:5803). Here the positive loss is rounded down first
			#    (floor = yes inside an inline effect value: game/events/activities/coronation_activity/
			#    coronation_events_klank.txt:3790-3793), then negated, so the cut never exceeds the share
			#    (3 soldiers at 25% lose 0).
			every_maa_regiment = {
				limit = {
					is_raised = no
					is_event_maa_regiment = no
				}
				change_maa_troops_count = {
					value = maa_current_troops_count
					multiply = $SHARE$
					floor = yes
					multiply = -1
				}
			}
		}

		# 4. Feedback to the player, as Pantheon's raid strike does (pantheon_raid_effects.txt:41-45).
		root = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = one_trait_break_army_toast_$SHARE$_t
				desc = one_trait_break_army_toast_$SHARE$_desc
				left_icon = scope:one_trait_target
			}
		}
	}
}
