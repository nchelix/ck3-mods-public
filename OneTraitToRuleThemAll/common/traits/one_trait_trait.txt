one_trait_scheme = {
	# CK3 rebuilt schemes around success chance and phase duration; the whole
	# "scheme power" vocabulary this trait used was removed. The success chance
	# cap is 95% by default (see HOSTILE/PERSONAL/POLITICAL_SCHEME_MAX_SUCCESS_CHANCE),
	# so the _max_add lines are what actually make schemes certain rather than
	# merely very likely.

	# Your schemes always succeed
	owned_personal_scheme_success_chance_add = 100
	owned_personal_scheme_success_chance_max_add = 100
	owned_hostile_scheme_success_chance_add = 100
	owned_hostile_scheme_success_chance_max_add = 100
	owned_political_scheme_success_chance_add = 100
	owned_political_scheme_success_chance_max_add = 100
	owned_contract_scheme_success_chance_add = 100

	# ...immediately
	personal_scheme_phase_duration_add = -100
	hostile_scheme_phase_duration_add = -100

	# ...and unseen
	owned_scheme_secrecy_add = 100

	# Nobody schemes against you
	enemy_personal_scheme_success_chance_add = -100
	enemy_hostile_scheme_success_chance_add = -100
	enemy_hostile_scheme_success_chance_growth_add = -100
	enemy_scheme_secrecy_add = -100
	enemy_hostile_scheme_phase_duration_add = 1000
	enemy_political_scheme_phase_duration_add = 1000

	# Run as many at once as you like
	max_personal_schemes_add = 10
	max_hostile_schemes_add = 10
	max_political_schemes_add = 10

	# Contracts and tasks (Wandering Nobles)
	max_contract_schemes_add = 10
	contract_scheme_phase_duration_add = -100
	political_scheme_phase_duration_add = -100

	# enemy_hostile_scheme_success_chance_add lowers enemy chances; this lowers
	# their cap, so no enemy scheme can succeed at all.

	# unlock. Vanilla checks these as has_trait_with_flag.
