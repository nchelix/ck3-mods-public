namespace = one_trait_panel

# One Trait 2.5: the work behind Give a title and Marry. Hidden events on the player; the
# interactions pass everything in variables.

# Give a title: the effect of the retired one_trait_menu.1200 give options
one_trait_panel.0001 = {
	type = character_event
	hidden = yes
	immediate = {
		var:one_trait_give_to = { save_scope_as = one_trait_target }
		var:one_trait_give_title = { save_scope_as = one_trait_given_title }
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:one_trait_given_title = {
			change_title_holder_include_vassals = {
				holder = scope:one_trait_target
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		if = { limit = { has_variable = one_trait_give_title } remove_variable = one_trait_give_title }
		if = { limit = { has_variable = one_trait_give_to } remove_variable = one_trait_give_to }
		if = { limit = { has_variable = one_trait_wed_a } remove_variable = one_trait_wed_a }
		if = { limit = { has_variable = one_trait_wed_b } remove_variable = one_trait_wed_b }
		if = { limit = { has_variable = one_trait_wed_matrilineal } remove_variable = one_trait_wed_matrilineal }
	}
}

# Marry, or betroth when either is a child; matrilineally if chosen
one_trait_panel.0002 = {
	type = character_event
	hidden = yes
	immediate = {
		var:one_trait_wed_a = { save_scope_as = one_trait_wed_a }
		var:one_trait_wed_b = { save_scope_as = one_trait_wed_b }
		if = {
			limit = { OR = { scope:one_trait_wed_a = { is_adult = no } scope:one_trait_wed_b = { is_adult = no } } }
			if = { limit = { has_variable = one_trait_wed_matrilineal } scope:one_trait_wed_a = { create_betrothal_matrilineal = scope:one_trait_wed_b } }
			else = { scope:one_trait_wed_a = { create_betrothal = scope:one_trait_wed_b } }
		}
		else = {
			if = { limit = { has_variable = one_trait_wed_matrilineal } scope:one_trait_wed_a = { marry_matrilineal = scope:one_trait_wed_b } }
			else = { scope:one_trait_wed_a = { marry = scope:one_trait_wed_b } }
		}
		if = { limit = { has_variable = one_trait_give_title } remove_variable = one_trait_give_title }
		if = { limit = { has_variable = one_trait_give_to } remove_variable = one_trait_give_to }
		if = { limit = { has_variable = one_trait_wed_a } remove_variable = one_trait_wed_a }
		if = { limit = { has_variable = one_trait_wed_b } remove_variable = one_trait_wed_b }
		if = { limit = { has_variable = one_trait_wed_matrilineal } remove_variable = one_trait_wed_matrilineal }
	}
}
