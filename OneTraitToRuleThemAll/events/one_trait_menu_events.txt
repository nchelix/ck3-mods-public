# One Trait menu graph. Entered from one_trait_menu_interaction.
# scope:one_trait_target is the right-clicked character and is available in
# every event below.
#
# Re-check on every CK3 patch -- all of the following fail silently rather
# than erroring if the game changes underneath them:
#   - The five hardcoded top-tier domicile building ids on 0420.domicile
#     (camp_main_04, estate_main_05, east_asian_estate_main_05, yurt_main_06,
#     japanese_manor_main_06) may stop being the top tier if a patch adds a
#     higher level.
#   - The is_domicile_type if/else_if chain on 0420.domicile has no final
#     else, so a new domicile type added by a patch gets no building added
#     and no warning.
#   - The two hardcoded struggle ids on 0420.struggle (iberian_struggle,
#     persian_struggle) would not include a third struggle if one is added.

namespace = one_trait_menu

# Modern Systems: the realm.
one_trait_menu.0420 = {
	type = character_event
	title = one_trait_menu.0420.t
	desc = one_trait_menu.0420.desc
	theme = default

	left_portrait = {
		character = scope:one_trait_target
		animation = personality_bold
	}

	option = {
		name = one_trait_menu.0420.herd
		trigger = {
			scope:one_trait_target = {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:one_trait_target.domicile = {
			change_herd = 10000
		}
		trigger_event = one_trait_menu.0420
	}

	option = {
		name = one_trait_menu.0420.herd_drain
		trigger = {
			scope:one_trait_target = {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:one_trait_target.domicile = {
			change_herd = -10000
		}
		trigger_event = one_trait_menu.0420
	}

	option = {
		name = one_trait_menu.0420.plague
		trigger = {
			exists = scope:one_trait_target.capital_province
		}
		scope:one_trait_target.capital_province = {
			create_epidemic_outbreak = {
				type = bubonic_plague
				intensity = major
			}
		}
		trigger_event = one_trait_menu.0420
	}

	option = {
		name = one_trait_menu.0420.domicile
		trigger = {
			exists = scope:one_trait_target.domicile
		}
		scope:one_trait_target.domicile = {
			if = {
				limit = {
					is_domicile_type = camp
				}
				add_domicile_building = camp_main_04
			}
			else_if = {
				limit = {
					is_domicile_type = estate
				}
				add_domicile_building = estate_main_05
			}
			else_if = {
				limit = {
					is_domicile_type = east_asian_estate
				}
				add_domicile_building = east_asian_estate_main_05
			}
			else_if = {
				limit = {
					is_domicile_type = yurt
				}
				add_domicile_building = yurt_main_06
			}
			else_if = {
				limit = {
					is_domicile_type = japanese_manor
				}
				add_domicile_building = japanese_manor_main_06
			}
		}
		trigger_event = one_trait_menu.0420
	}

	option = {
		name = one_trait_menu.0420.struggle
		trigger = {
			scope:one_trait_target = {
				any_character_struggle = {
					always = yes
				}
			}
		}
		struggle:iberian_struggle ?= {
			end_struggle = yes
		}
		struggle:persian_struggle ?= {
			end_struggle = yes
		}
		trigger_event = one_trait_menu.0420
	}

	option = {
		name = one_trait_menu.0420.tributary
		trigger = {
			scope:one_trait_target = {
				any_tributary = {
					always = yes
				}
			}
		}
		scope:one_trait_target = {
			every_tributary = {
				end_tributary = yes
			}
		}
		trigger_event = one_trait_menu.0420
	}

	# Starts vanilla's conquest: conqueror.0001 adds the trait and the
	# story_conqueror story, whose variable unlocks the armies, gifts and CBs.
	# Its trigger needs an independent landed ruler, so this option does too.
	# It fires on the target, because the event adds the trait to its root, and
	# does not re-fire the menu: the conqueror window replaces it.
	option = {
		name = one_trait_menu.0420.conqueror
		trigger = {
			scope:one_trait_target = {
				top_liege = this
				is_landed = yes
				NOR = {
					has_trait = conqueror
					any_owned_story = {
						story_type = story_conqueror
					}
				}
			}
		}
		scope:one_trait_target = {
			add_character_flag = {
				flag = one_trait_scourge_roll
				days = 5
			}
			trigger_event = conqueror.0001
			# Scourge of the Gods a day later, after vanilla's story setup (2.4.1)
			trigger_event = { id = one_trait_menu.0421 days = 1 }
		}
	}

	# 2.4.1: as Become a Conqueror, with Scourge of the Gods guaranteed. Vanilla grants
	# ai_extreme_conqueror_modifier only to AI conquerors.
	option = {
		name = one_trait_menu.0420.scourge
		trigger = {
			scope:one_trait_target = {
				top_liege = this
				is_landed = yes
				NOR = {
					has_trait = conqueror
					any_owned_story = {
						story_type = story_conqueror
					}
				}
			}
		}
		scope:one_trait_target = {
			add_character_flag = {
				flag = one_trait_scourge_sure
				days = 5
			}
			trigger_event = conqueror.0001
			# Scourge of the Gods a day later, after vanilla's story setup (2.4.1)
			trigger_event = { id = one_trait_menu.0421 days = 1 }
		}
	}

	# Ends the conquest with no penalty. Vanilla's conqueror.0002 is the FAILED
	# ending (-2 prestige levels, -50 opinion from glory hounds); ending the
	# story directly runs its on_end, which removes the trait and modifiers.
	option = {
		name = one_trait_menu.0420.conqueror_end
		trigger = {
			scope:one_trait_target = {
				any_owned_story = {
					story_type = story_conqueror
				}
			}
		}
		scope:one_trait_target = {
			every_owned_story = {
				limit = {
					story_type = story_conqueror
				}
				end_story = yes
			}
		}
		trigger_event = one_trait_menu.0420
	}

	option = {
		name = one_trait_menu.0420.back
		trigger_event = one_trait_menu.0400
	}
}

# 2.4.1: a day after a conquest starts, Scourge of the Gods (ai_extreme_conqueror_modifier). Root is the
# conqueror. Guaranteed from the Scourge option; otherwise, for a player, the Conqueror Bonuses rule's odds as
# vanilla rolls them for AI conquerors (an AI target already had vanilla's own roll) (default and none: never; random 5%; reduced 0.5%; extreme: always).
one_trait_menu.0421 = {
	type = character_event
	hidden = yes
	trigger = {
		any_owned_story = { story_type = story_conqueror }
		NOT = { has_character_modifier = ai_extreme_conqueror_modifier }
	}
	immediate = {
		if = {
			limit = {
				OR = {
					has_character_flag = one_trait_scourge_sure
					AND = {
						is_ai = no
						has_game_rule = extreme_conquerors_bonuses
					}
				}
			}
			save_scope_value_as = { name = one_trait_scourge value = yes }
		}
		else_if = {
			limit = { is_ai = no has_game_rule = random_conquerors_bonuses }
			random_list = {
				95 = { }
				5 = { save_scope_value_as = { name = one_trait_scourge value = yes } }
			}
		}
		else_if = {
			limit = { is_ai = no has_game_rule = reduced_random_conquerors_bonuses }
			random_list = {
				995 = { }
				5 = { save_scope_value_as = { name = one_trait_scourge value = yes } }
			}
		}
		if = {
			limit = { exists = scope:one_trait_scourge }
			add_character_modifier = {
				modifier = ai_extreme_conqueror_modifier
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = one_trait_scourge_toast
				left_icon = root
			}
		}
		remove_character_flag = one_trait_scourge_roll
		remove_character_flag = one_trait_scourge_sure
	}
}

# Character editor: body. set_age bypasses birthday on-actions (engine doc

	desc = one_trait_menu.0700.desc
	theme = default

	left_portrait = {
		character = scope:one_trait_target
		animation = personality_bold
	}

	# Decided here, not in Remove-every's trigger: a trigger that checks

	immediate = {
		if = {
			limit = {
				scope:one_trait_target = {
					OR = {
						has_trait = one_trait_health
						has_trait = one_trait_immortality
						has_trait = one_trait_giant
						has_trait = one_trait_fertility
						has_trait = one_trait_celibate
						has_trait = one_trait_stress
						has_trait = one_trait_stats
						has_trait = one_trait_xp
						has_trait = one_trait
						has_trait = one_trait_wealth
						has_trait = one_trait_fame
						has_trait = one_trait_faith
						has_trait = one_trait_charisma
						has_trait = one_trait_dread
						has_trait = one_trait_manager
						has_trait = one_trait_builder
						has_trait = one_trait_domain
						has_trait = one_trait_commander
						has_trait = one_trait_scheme
						has_trait = one_trait_steppe
						has_trait = one_trait_wanderer
					}
				}
			}
			save_scope_value_as = {
				name = one_trait_target_has_any

			}
		}
		# Always set, never just when true: saved values travel down the chain

		name = one_trait_menu.0700.body
		trigger_event = one_trait_menu.0710
