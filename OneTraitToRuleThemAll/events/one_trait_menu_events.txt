# 2.7 Task 4 (batch-3 handoff): one_trait_menu.0420 (Modern Systems: the Realm) retired -- its options
# moved into the panel's Realm > Realm events sub-tab (one_trait_realm_effects.txt /
# one_trait_realm_triggers.txt, Batches 2-3). one_trait_menu.0421 stays: it is a hidden, delayed follow-up
# (R3), not a page, and the conqueror/scourge effects below still schedule it exactly as this page did.

# 2.4.1: a day after a conquest starts, Scourge of the Gods (ai_extreme_conqueror_modifier). Root is the
# conqueror. Guaranteed from the Scourge option; otherwise, for a player, the Conqueror Bonuses rule's odds as
# vanilla rolls them for AI conquerors (an AI target already had vanilla's own roll) (default and none: never; random 5%; reduced 0.5%; extreme: always).
one_trait_menu.0421 = {
	type = character_event
	hidden = yes
	trigger = {
		any_owned_story = { story_type = story_conqueror }
		NOT = { has_character_modifier = ai_extreme_conqueror_modifier }
	}
	immediate = {
		if = {
			limit = {
				OR = {
					has_character_flag = one_trait_scourge_sure
					AND = {
						is_ai = no
						has_game_rule = extreme_conquerors_bonuses
					}
				}
			}
			save_scope_value_as = { name = one_trait_scourge value = yes }
		}
		else_if = {
			limit = { is_ai = no has_game_rule = random_conquerors_bonuses }
			random_list = {
				95 = { }
				5 = { save_scope_value_as = { name = one_trait_scourge value = yes } }
			}
		}
		else_if = {
			limit = { is_ai = no has_game_rule = reduced_random_conquerors_bonuses }
			random_list = {
				995 = { }
				5 = { save_scope_value_as = { name = one_trait_scourge value = yes } }
			}
		}
		if = {
			limit = { exists = scope:one_trait_scourge }
			add_character_modifier = {
				modifier = ai_extreme_conqueror_modifier
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = one_trait_scourge_toast
				left_icon = root
			}
		}
		remove_character_flag = one_trait_scourge_roll
		remove_character_flag = one_trait_scourge_sure
	}
}

			every_ally = {
				limit = {
					NOR = {

				add_opinion = {
					modifier = perk_negotiated_alliance_opinion
					target = root
				}
				root = {
					allow_alliance = scope:one_trait_inherited_ally
					create_alliance = {
						target = scope:one_trait_inherited_ally
						allied_through_owner = root
						allied_through_target = scope:one_trait_inherited_ally
					}
				}
			}
		}
		trigger_event = one_trait_menu.1410
	}

	option = {
		name = one_trait_menu.1410.isolate
		trigger = {
			scope:one_trait_target = { any_ally = { always = yes } }
		}
		scope:one_trait_target = {
			every_ally = {
				break_alliance = scope:one_trait_target
			}
